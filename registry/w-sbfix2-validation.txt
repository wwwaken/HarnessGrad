from pathlib import Path
import shlex
import shutil
import subprocess
import sys

label, *command = sys.argv[1:]
tmp = Path('/tmp/sbfix2-' + label)
tmp.mkdir(exist_ok=False)
argv = ['env', 'TMPDIR=' + str(tmp), 'PYTHONDONTWRITEBYTECODE=1', *command]
try:
    result = subprocess.run(argv, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
finally:
    shutil.rmtree(tmp)
transcript = ('$ ' + shlex.join(argv) + '\n' + result.stdout
              + f'[exit_code={result.returncode}]\n'
              + f'Cleanup: shutil.rmtree({str(tmp)!r}); exists={tmp.exists()}\n')
Path('/mnt/20t/xubuqiang/hg-work/sbfix2/' + label + '.txt').write_text(transcript)
