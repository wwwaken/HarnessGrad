env = dict(os.environ, TMPDIR=str(temporary), PYTHONDONTWRITEBYTECODE='1')
try:
    proc = subprocess.run(command, shell=True, executable='/bin/bash', env=env,
                          stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                          text=True, errors='replace')
    output = (f'$ TMPDIR={temporary} PYTHONDONTWRITEBYTECODE=1 {command}\n'
              + proc.stdout + f'[exit_code={proc.returncode}]\n')

(root / (name + '.log')).write_text(output)
print(output, end='')
sys.exit(proc.returncode)
