"""Observe selected/spec/receipt identities without changing verifier results."""
import json
import os
from pathlib import Path


def pytest_configure(config):
    from eval.run import runner
    from eval.verification import verifier
    original = verifier._verify

    def observed(workdir, spec, answer, **kwargs):
        result = original(workdir, spec, answer, **kwargs)
        binding = (spec or {}).get('scorer_input') or {}
        receipt = (result.get('evidence') or {}).get('scorer_input') or {}
        row = {'selected_task_id': kwargs.get('task_id'),
               'spec_task_id': binding.get('task_id'),
               'identity_equal': binding.get('task_id') == kwargs.get('task_id'),
               'receipt_task_id': receipt.get('task_id'),
               'receipt_verified': receipt.get('verified'),
               'unmeasured': result.get('unmeasured'), 'score': result.get('score')}
        with Path(os.environ['IDENTCALL_AUDIT_LOG']).open('a') as out:
            out.write(json.dumps(row, sort_keys=True) + '\n')
        return result

    verifier._verify = observed
    if runner._verify is original:
        runner._verify = observed
