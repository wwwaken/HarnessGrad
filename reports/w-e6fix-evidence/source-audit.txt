    1 """Run a task's harness and its check inside a container. INTERFACE.md §2.5.
    2 
    3 This is the `exec` environment kind. It is a **second, independent boundary**, not a
    4 replacement for `eval/containers/sandbox.py`:
    5 
    6 * `files` -- the harness runs on the host inside a bwrap namespace. The platform is
    7   hidden by not being mounted (`eval/containers/sandbox.py`), and the runtime is the host's.
    8 * `exec`  -- the harness runs **inside a container**. The platform is hidden the same
    9   way (it is not mounted), but the runtime, the interpreter and the tools are the
   10   image's, and `network: none` becomes available because the container has its own
   11   network namespace rather than borrowing the host's.
   12 
   13 Three properties are load-bearing and each is a decision rather than an accident:
   14 
   15 **The container is mounted from a digest, never a tag.** `:latest` moves, and two runs
   16 under one tag are not two measurements of the same thing -- the same failure this
   17 platform already shipped once for `agent_model`. `resolve_digest` is therefore not a
   18 convenience: a run whose digest cannot be resolved **refuses to start** (§2.5.4).
   19 
   20 **Nothing is built here.** A task image is built by `tools/import_env.py`, once, by a
   21 human, at import time (§2.5.5). `docker run` on a missing image would silently pull
   22 whatever the registry served that day; this module does not pull and does not build.
   23 If the image is not on the local daemon, that is a refusal with instructions.
   24 
   25 **The container is not given a way out of itself.** No docker socket, no host home, no
   26 platform. It gets exactly three mounts (§2.5.3), and the order they are applied in is
   27 load-bearing -- see `_mounts`.
   28 """

   76 # A snapshot is a Docker image layer. Bound both the writable layer and the space Docker
   77 # can consume while committing it; otherwise a large GAIA attachment or JVM cache can turn
   78 # commit into a host OOM/disk failure and the task is incorrectly reported as a harness score.
   79 SNAPSHOT_MAX_BYTES = 16 << 30
   80 SNAPSHOT_RESERVE_BYTES = 1 << 30
   81 SNAPSHOT_INSPECT_TIMEOUT_S = 60
   82 SNAPSHOT_COMMIT_TIMEOUT_S = 1800

   85 def _snapshot_limit(key: str, default: int) -> int:
   86     try:
   87         value = int(os.environ.get(key, default))
   88         if not 0 < value <= 2**63 - 1:
   89             raise ValueError("outside positive signed 64-bit range")
   90         return value
   91     except (TypeError, ValueError) as exc:
   92         raise SnapshotUnavailable(f"snapshot admission failed kind=resource: invalid {key}") from exc

   95 def _snapshot_failure(stage: str, proc: subprocess.CompletedProcess | None = None,
   96                       exc: BaseException | None = None) -> SnapshotUnavailable:
   97     text = " ".join(filter(None, [
   98         str(exc) if exc else "",
   99         (proc.stderr if proc else "") if proc else "",
  100         (proc.stdout if proc else "") if proc else "",
  101     ])).strip()
  102     lowered = text.lower()
  103     if ((proc is not None and proc.returncode == 137)
  104             or "out of memory" in lowered or "cannot allocate memory" in lowered
  105             or "oom" in lowered):
  106         kind = "oom"
  107     elif any(marker in lowered for marker in
  108              ("no space left", "disk quota", "quota exceeded", "insufficient space")):
  109         kind = "disk"
  110     elif any(marker in lowered for marker in
  111              ("resource temporarily unavailable", "too many processes", "pids limit")):
  112         kind = "resource"
  113     else:
  114         kind = "docker"
  115     code = f" exit={proc.returncode}" if proc is not None else ""
  116     return SnapshotUnavailable(f"snapshot {stage} failed kind={kind}{code}: {text[-1200:]}")

  119 def snapshot_admission(name: str) -> dict[str, int | str | bool]:
  120     """Check bounded writable size and Docker-root headroom before ``docker commit``.
  121 
  122     This is byte-capacity preflight, not a filesystem quota or inode/resource gate.
  123     A commit killed by the daemon can leave an incomplete image and used to surface
  124     as an unexplained exit 137. The shared runner admission contract covers the
  125     additional observations a backend must supply for those stronger gates.
  126     """
  127     maximum = _snapshot_limit("HG_SNAPSHOT_MAX_BYTES", SNAPSHOT_MAX_BYTES)
  128     reserve = _snapshot_limit("HG_SNAPSHOT_RESERVE_BYTES", SNAPSHOT_RESERVE_BYTES)
  129     try:
  130         proc = docker("inspect", "--size", "-f", "{{.SizeRw}}", name,
  131                       timeout=SNAPSHOT_INSPECT_TIMEOUT_S)
  132     except (subprocess.TimeoutExpired, OSError) as exc:
  133         raise _snapshot_failure("admission inspect", exc=exc) from exc
  134     if proc.returncode != 0:
  135         raise _snapshot_failure("admission inspect", proc)
  136     try:
  137         writable = int(proc.stdout.strip())
  138     except ValueError:
  139         raise SnapshotUnavailable(
  140             "snapshot admission failed kind=docker: invalid writable size")
  141     if not 0 <= writable <= 2**63 - 1:
  142         raise SnapshotUnavailable("snapshot admission failed kind=docker: invalid writable size")
  143     if writable > maximum:
  144         raise SnapshotUnavailable(
  145             f"snapshot admission rejected kind=disk writable_bytes={writable} "
  146             f"max_bytes={maximum}")
  147 
  148     try:
  149         info = docker("info", "--format", "{{.DockerRootDir}}",
  150                       timeout=SNAPSHOT_INSPECT_TIMEOUT_S)
  151     except (subprocess.TimeoutExpired, OSError) as exc:
  152         raise _snapshot_failure("admission docker-info", exc=exc) from exc
  153     if info.returncode != 0:
  154         raise _snapshot_failure("admission docker-info", info)
  155     root_text = info.stdout.strip()
  156     root = Path(root_text)
  157     if (not root.is_absolute() or len(root_text) > 1024
  158             or any(ord(c) < 32 for c in root_text)):
  159         raise SnapshotUnavailable("snapshot admission failed kind=docker: invalid Docker root")
  160     try:
  161         free = shutil.disk_usage(root).free
  162     except OSError as exc:
  163         raise _snapshot_failure("admission disk-usage", exc=exc) from exc
  164     if type(free) is not int or not 0 <= free <= 2**63 - 1:
  165         raise SnapshotUnavailable("snapshot admission failed kind=disk: invalid free bytes")
  166     required = writable + reserve
  167     if free < required:
  168         raise SnapshotUnavailable(
  169             f"snapshot admission rejected kind=disk docker_root={root} free_bytes={free} "
  170             f"required_bytes={required} writable_bytes={writable}")
  171     return {"writable_bytes": writable, "free_bytes": free, "required_bytes": required,
  172             "docker_root": str(root), "max_bytes": maximum, "reserve_bytes": reserve,
  173             "inspect_timeout_s": SNAPSHOT_INSPECT_TIMEOUT_S,
  174             "commit_timeout_s": SNAPSHOT_COMMIT_TIMEOUT_S,
  175             "preflight_only": True, "quota_enforced": False}

  180 #: How much of a streamed command's output is kept. `docker exec` on a long task can
  181 #: print for hours, and only the tail is ever read (the runner keeps `stderr[-2000:]`).
  182 #: A cap rather than a spool to disk: the point of the cap is that a chatty subject
  183 #: cannot fill the machine's memory with output nobody will read.
  184 _STREAM_KEEP = 4 << 20

  187 def _drain(pipe, sink: list[str], limit: int = _STREAM_KEEP) -> None:
  188     """Read a pipe to EOF, keeping the last `limit` bytes. Runs on its own thread.
  189 
  190     Interleaving stdout and stderr is what `subprocess.run` would not do, and it is
  191     what makes a live log readable: `docker exec` writes its own errors and the
  192     subject's output to two different pipes, and a reader wants them in the order they
  193     were said.
  194     """
  195     try:
  196         while True:
  197             chunk = pipe.readline()
  198             if not chunk:
  199                 break
  200             sink.append(chunk)
  201             if sum(len(c) for c in sink) > limit:
  202                 del sink[:max(1, len(sink) // 2)]
  203     except (ValueError, OSError):
  204         pass
  205     finally:
  206         try:
  207             pipe.close()
  208         except OSError:
  209             pass

  212 def _run_streaming(argv: list[str], timeout: int, _label: str = ""):
  213     """`subprocess.run` shape, but in slices so this thread stays interruptible.
  214 
  215     Only used while a task is in flight (`eval/observe/progress.py`). The beats themselves come
  216     from the *phase*, not from here -- one thread that beats every 10s regardless of
  217     which call is blocking, which is why this loop only has to wait and check a
  218     deadline. The first version beat here as well and produced two heartbeats per
  219     interval, each with a different elapsed time, which is a log that contradicts
  220     itself: measured, `elapsed_s: 10.0` twice in a row.
  221 
  222     The cost -- two reader threads, merged output -- is paid only inside a task, so the
  223     30-odd short docker calls a task makes (`create`, `cp`, `commit`) are untouched.
  224     """
  225     if timeout:
  226         deadline = time.monotonic() + timeout
  227     else:
  228         deadline = None
  229     proc = subprocess.Popen(argv, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
  230                             text=True, bufsize=1)
  231     out_chunks: list[str] = []
  232     err_chunks: list[str] = []
  233     readers = [threading.Thread(target=_drain, args=(proc.stdout, out_chunks),
  234                                 daemon=True),
  235                threading.Thread(target=_drain, args=(proc.stderr, err_chunks),
  236                                 daemon=True)]
  237     for r in readers:
  238         r.start()
  239     #: One second, so that a timeout is enforced to within about a second -- the same
  240     #: promise `subprocess.run`'s own timeout makes -- and short enough that a stopping
  241     #: driver is noticed promptly.
  242     SLICE = 1.0
  243     while True:
  244         try:
  245             proc.wait(timeout=SLICE)
  246             break
  247         except subprocess.TimeoutExpired:
  248             if deadline is not None and time.monotonic() > deadline:
  249                 proc.kill()
  250                 proc.wait()
  251                 for r in readers:
  252                     r.join(timeout=2)
  253                 raise subprocess.TimeoutExpired(argv, timeout)
  254     for r in readers:
  255         r.join(timeout=5)
  256     return subprocess.CompletedProcess(argv, proc.returncode, "".join(out_chunks),
  257                                        "".join(err_chunks))

  260 def docker(*args: str, timeout: int = 120) -> subprocess.CompletedProcess:
  261     """Run the docker client.
  262 
  263     Public because `eval/service/services.py` needs the same client, and two modules each
  264     shelling out to `docker` with their own idea of how to report a failure is two
  265     error surfaces to keep in step.
  266 
  267     Long calls made *inside a task* stream and beat; everything else is one
  268     `subprocess.run`. The branch is on the presence of a task rather than on a flag per
  269     call site, because the calls that need it are the two that block for minutes
  270     (`exec` a harness, `exec` a check) and they are made from three places, none of
  271     which knows whether a watcher exists.
  272     """
  273     from eval.observe import progress as progress_mod
  274 
  275     if progress_mod.task_of() is None:
  276         return subprocess.run(["docker", *args], capture_output=True, text=True,
  277                               timeout=timeout)
  278     return _run_streaming(["docker", *args], timeout, f"docker {args[0]}")
  279

  296 def looks_pinned(image: str) -> bool:
  297     """True for `repo@sha256:<64 hex>` and for a bare `sha256:<64 hex>` image id."""
  298     if "@sha256:" in image:
  299         return _is_hex(image.rsplit("@sha256:", 1)[1])
  300     if image.startswith("sha256:"):
  301         return _is_hex(image[len("sha256:"):])
  302     return False

  309 def resolve_digest(image: str) -> str:
  310     """Pin `image` to a content address, locally, without pulling or building.
  311 
  312     Returns a reference `docker run` accepts -- `repo@sha256:...` when the image
  313     carries a registry digest, else the bare image ID, which is itself a content hash
  314     of the local image and therefore pinnable.
  315 
  316     Refuses rather than pulling. A pull during a run is a network dependency that
  317     makes the run depend on what the registry served that day; importing the image is
  318     a one-time step (`tools/import_env.py`), which is where §2.5.5 puts it.
  319     """
  320     if looks_pinned(image):
  321         return image
  322     if image in _DIGEST_CACHE:
  323         return _DIGEST_CACHE[image]
  324 
  325     try:
  326         proc = _docker("image", "inspect", "--format",
  327                        "{{if .RepoDigests}}{{index .RepoDigests 0}}{{else}}{{.Id}}{{end}}",
  328                        image, timeout=60)
  329     except subprocess.TimeoutExpired as exc:
  330         raise ContainerUnavailable(f"timed out inspecting image {image!r}") from exc
  331     if proc.returncode != 0:
  332         raise ContainerUnavailable(
  333             f"image {image!r} is not on the local daemon, and this platform does not "
  334             f"pull during a run. Import it first:\n"
  335             f"    python tools/import_env.py --image {image}\n"
  336             f"docker said: {proc.stderr.strip()[:300]}")
  337 
  338     pinned = proc.stdout.strip()
  339     if not looks_pinned(pinned):
  340         raise ContainerUnavailable(
  341             f"could not pin image {image!r} to a content address; "
  342             f"`docker image inspect` returned {pinned!r}")
  343     _DIGEST_CACHE[image] = pinned
  344     return pinned
  345 
  346 
  347 #: One `docker image inspect` per distinct reference, not one per task: resolution is
  348 #: a startup step, and a 400-task benchmark should not fork the docker client 400
  349 #: times to learn something that cannot change during a run.

  362 def interpreter_paths() -> tuple[Path, ...]:
  363     """The trees the platform's own interpreter needs, so a container can run it.
  364 
  365     A harness is a Python program -- the platform already assumes that when it invokes
  366     `python <entry>` -- and **a task image need not have a Python at all**. Measured: the
  367     real Terminal-Bench image is `ubuntu:24.04` plus a `COPY`, with no `python3`, no `R`
  368     and no `gcc`; the task explicitly expects the agent to `apt-get install` what it
  369     needs. So assuming the image provides the interpreter is assuming something the
  370     benchmark does not promise.
  371 
  372     Both `sys.prefix` and `sys.base_prefix` are needed, and that is measured too: a
  373     virtualenv's `bin/python3` resolves to its base prefix, so mounting the venv alone
  374     gives `not found` inside the container. Together they give a working interpreter
  375     **and** the packages the platform installed for its harnesses.
  376     """
  377     import sys as _sys
  378     out: list[Path] = []
  379     for candidate in (_sys.prefix, _sys.base_prefix):
  380         path = Path(candidate)
  381         if path.is_dir() and path not in out:
  382             out.append(path)
  383     return tuple(out)

  390 def image_python(image: str) -> str | None:
  391     """The interpreter the image itself provides, or None if it has none.
  392 
  393     The platform injects its own interpreter into a container, because a task image need
  394     not have one -- measured: the real Terminal-Bench image is `ubuntu:24.04` plus a
  395     `COPY`, with no `python3` at all. But injecting it unconditionally would contradict
  396     §2.5.3: the reason the harness runs *inside* the environment is so that "`bash`, the
  397     file editor and **the interpreter** are the image's".
  398 
  399     So the image wins when it has an interpreter, and the platform's is the fallback.
  400     Both halves are measured: a `files`-style image with its own Python keeps it (and
  401     keeps its own installed packages), and an image with none still runs the harness.
  402     """
  403     if image in _PYTHON_CACHE:
  404         return _PYTHON_CACHE[image]
  405     found: str | None = None
  406     try:
  407         proc = docker("run", "--rm", "--network", "none", "--entrypoint", "sh", image,
  408                       "-c", "command -v python3 || command -v python || true",
  409                       timeout=180)
  410         out = proc.stdout.strip()
  411         found = out.splitlines()[-1].strip() if out else None
  412     except (subprocess.TimeoutExpired, OSError):
  413         found = None
  414     _PYTHON_CACHE[image] = found or None
  415     return _PYTHON_CACHE[image]

  429 def container_env(task_mount: str, deadline_s: int | None = None) -> dict[str, str]:
  430     """Everything a harness sees in its environment inside the container.
  431 
  432     Built from an allowlist rather than forwarded, and with `PATH` deliberately absent:
  433     the container has its own, the host's names directories that do not exist inside,
  434     and forwarding it would let a harness pick up a host binary if the two ever
  435     overlapped.
  436 
  437     **`HG_AGENT_DEADLINE_S` is the environment's own `agent_timeout_s`.** It is not a
  438     hint: the platform kills the container at that second, and until this existed a
  439     harness that spawns a subprocess had no way to know the number it was being measured
  440     against. Measured (`codex-task-dgm`, 2026-10-05): an adapter that gave its own child a
  441     900 s timeout raced the platform's 900 s kill -- the container died first and the
  442     curve recorded `exit code -1 -- timeout after 900s`, a killed container instead of a
  443     harness that decided to stop. The budget a harness is subject to is a fact about the
  444     measurement, so it is stated where the harness can read it (INTERFACE.md §3.3).
  445     """
  446     env = {k: v for k, v in os.environ.items()
  447            if k.startswith(PASS_THROUGH_PREFIXES) or k in PASS_THROUGH_EXACT}
  448     home = env_home(task_mount)
  449     env.update({
  450         "HOME": home,
  451         "XDG_CONFIG_HOME": f"{home}/.config",
  452         "XDG_DATA_HOME": f"{home}/.local/share",
  453         "XDG_CACHE_HOME": f"{home}/.cache",
  454         # A CLI's scratch files should not land in the directory the task is about,
  455         # where a check would then be reading them.
  456         "TMPDIR": f"{home}/tmp",
  457     })
  458     if deadline_s:
  459         env["HG_AGENT_DEADLINE_S"] = str(int(deadline_s))
  460     return env
  461

  463 #: The `container_env` entries whose value is a **directory a harness is invited to write
  464 #: into**. `PATH` and the model variables are deliberately absent: a name belongs here only
  465 #: when something is expected to put a file underneath it.
  466 SCRATCH_ENV_VARS = ("HOME", "XDG_CONFIG_HOME", "XDG_DATA_HOME", "XDG_CACHE_HOME", "TMPDIR")

  469 def writable_env_dirs(task_mount: str) -> tuple[str, ...]:
  470     """Every directory `container_env` names that a harness may write into.
  471 
  472     **A variable that names a directory is not a directory.** `docker create --tmpfs
  473     <home>` brings its own mount point and nothing below it, so `HOME` exists while
  474     `HOME/tmp` -- the value this module hands out as `TMPDIR` -- does not. A harness is
  475     entitled to believe the environment it was given: `mkdtemp(TMPDIR + "/scratch-XXXXXX")`
  476     needs the *parent* to be there, and `ENOENT` when it is not is a failure caused by the
  477     platform, not by the harness.
  478 
  479     Measured 2026-10-06, on the `dsh_agent` harness: DSH's spill store died on exactly that
  480     `mkdtemp` and took its whole plugin tree with it, reporting `dsh: plugin tree failed to
  481     load: ... failed to apply loader entry spill-local` -- a message naming neither `TMPDIR`
  482     nor this module, at the end of a four-layer hunt. Returning the list instead of doing
  483     the `mkdir` here is deliberate: `mkdir_in` is a `docker exec`, so the container has to
  484     be **running**, and the caller that owns the start/stop order is the only one that knows
  485     when that is true.
  486     """
  487     env = container_env(task_mount)
  488     seen: list[str] = []
  489     for var in SCRATCH_ENV_VARS:
  490         value = env.get(var)
  491         if value and value not in seen:
  492             seen.append(value)
  493     return tuple(seen)

  496 def _mounts(task_dir: Path, task_mount: str, harness_root: Path | None,
  497             state_dirs: tuple[Path, ...], task_dir_writable: bool = True,
  498             readonly_paths: tuple[Path, ...] = ()) -> list[str]:
  499     """The mount plan. Parent before child, so a reader can follow it.
  500 
  501     **The order is not what makes the `.state/` hole work here, and this docstring
  502     used to claim it was.** `eval/containers/sandbox.py` needs its holes bound after the
  503     read-only parent, because bwrap's binds are last-one-wins and the reverse order
  504     silently produced a `Read-only file system` on the harness's own scratch space.
  505     Docker is not bwrap: it *sorts* nested mounts by destination depth, so the same
  506     plan in the opposite CLI order gives the same result. Measured, both orders:
  507     `state_writable=yes`, with the `readonly` parent bind given last.
  508 
  509     So the property comes from Docker's nesting-aware sort, not from this list, and
  510     the ordering below is kept only because a deterministic plan is easier to read and
  511     to test than an arbitrary one. It is stated this way rather than copied over from
  512     the other module because "it worked there for this reason" is exactly how a
  513     correct-looking plan acquires a false explanation.
  514 
  515     The task directory is mounted first of all: it is the one tree that must always be
  516     writable, because it is what the harness is being asked to change.
  517     """
  518     plan = ["--mount",
  519             f"type=bind,src={task_dir},dst={task_mount}"
  520             + ("" if task_dir_writable else ",readonly")]
  521     if harness_root is not None:
  522         plan += ["--mount", f"type=bind,src={harness_root},dst={harness_root},readonly"]
  523     for state in state_dirs:
  524         plan += ["--mount", f"type=bind,src={state},dst={state}"]
  525     # Trees the harness needs to run at all: a CLI it drives, the interpreter it was
  526     # written against. Read-only, and **also for `exec`** -- an earlier version applied
  527     # them only to `files` on the theory that "inside a container the runtime is the
  528     # image's", which is false for a task image that ships no runtime.
  529     for extra in readonly_paths:
  530         plan += ["--mount", f"type=bind,src={extra},dst={extra},readonly"]
  531     return plan
  532

  534 def command(argv: list[str], *, image: str, task_dir: Path, task_mount: str,
  535             harness_root: Path | None = None, state_dirs: tuple[Path, ...] = (),
  536             cwd: str | None = None, env: dict[str, str] | None = None,
  537             network: str = "none", read_only: bool = True,
  538             task_dir_writable: bool = True,
  539             extra_mounts: tuple[str, ...] = (),
  540             readonly_paths: tuple[Path, ...] = (),
  541             limits: dict | None = None) -> list[str]:
  542     """The `docker run` argv. Split out so `self_check` and the tests can read it."""
  543     limits = limits or {}
  544     cmd = ["docker", "run", "--rm", "--network", network]
  545 
  546     if read_only:
  547         # The environment is the image (§2.5.3). A rootfs the harness can write to
  548         # would make the check's world depend on what the harness did to it, and two
  549         # runs of one task would stop being comparable. `/tmp` is the exception
  550         # because enough software hardcodes it that refusing would only produce
  551         # confusing failures; it is a tmpfs, so it dies with the container.
  552         cmd += ["--read-only", "--tmpfs", "/tmp:rw,exec,nosuid,size=512m,mode=1777"]
  553 
  554     cmd += ["--pids-limit", str(limits.get("pids", 1024))]
  555     cmd += ["--memory", str(limits.get("memory", "4g"))]
  556     cmd += ["--cpus", str(limits.get("cpus", "2"))]
  557 
  558     cmd += _mounts(task_dir, task_mount, harness_root, tuple(state_dirs),
  559                    task_dir_writable, tuple(readonly_paths))
  560     # Mounts this module does not know the meaning of: a service's recording, whose
  561     # audience (`eval/service/services.py`) is the design decision and not this layer's to
  562     # make. Appended after the task mount so a recording can never shadow it.
  563     cmd += list(extra_mounts)
  564 
  565     # Run as the invoking user rather than root, for two reasons that are both
  566     # correctness rather than hygiene: files the harness creates would otherwise land
  567     # on the host owned by root, which the platform then cannot delete (its cleanup is
  568     # `shutil.rmtree` as this user), and root-in-container is a strictly larger
  569     # privilege set than the task needs.
  570     cmd += ["--user", f"{os.getuid()}:{os.getgid()}"]
  571     for key, value in sorted((env or {}).items()):
  572         cmd += ["-e", f"{key}={value}"]
  573     cmd += ["-w", cwd or task_mount]
  574     cmd += [image, *argv]
  575     return cmd
  576

  578 def run(argv: list[str], *, image: str, task_dir: Path, task_mount: str,
  579         harness_root: Path | None = None, state_dirs: tuple[Path, ...] = (),
  580         cwd: str | None = None, env: dict[str, str] | None = None,
  581         timeout_s: int = 3600, network: str = "none",
  582         read_only: bool = True, task_dir_writable: bool = True,
  583         extra_mounts: tuple[str, ...] = (),
  584         readonly_paths: tuple[Path, ...] = (),
  585         ) -> subprocess.CompletedProcess:
  586     """`subprocess.run` shape, inside a **fresh** container. Same return type.
  587 
  588     Fresh per call, not one container reused for the harness and then the check. The
  589     property §2.5.6 asks for -- that nothing the harness started is still running when
  590     the check runs -- cannot be *guaranteed* by killing the process tree of a
  591     `docker exec`: anything the harness backgrounded is reparented to the container's
  592     PID 1 and survives the exec. Starting the check in a new container from the same
  593     image makes the guarantee structural instead of best-effort, and it costs one
  594     container start.
  595 
  596     What survives between the two is the task directory, because it is on a bind mount
  597     -- which is exactly the set of things a check is supposed to grade.
  598     """
  599     # The safety net, not the mechanism. The driver resolves every dataset image once
  600     # at startup (§2.5.4: no digest, no start); this refuses anyway, because the cost
  601     # of being wrong is a silent pull and the cost of checking is a string compare.
  602     if not looks_pinned(image):
  603         raise ContainerUnavailable(
  604             f"refusing to run {image!r}: it is not pinned to a content address, and "
  605             f"`docker run` on a tag would pull whatever the registry served today. "
  606             f"Resolve it with `resolve_digest` first (INTERFACE.md §2.5.4).")
  607     cmd = command(argv, image=image, task_dir=task_dir, task_mount=task_mount,
  608                   harness_root=harness_root, state_dirs=state_dirs, cwd=cwd,
  609                   env=env, network=network, read_only=read_only,
  610                   task_dir_writable=task_dir_writable, extra_mounts=extra_mounts,
  611                   readonly_paths=readonly_paths)
  612     # A client timeout kills `docker run`, not the daemon-side container. `--rm`
  613     # only takes effect when that container exits, so retain an exact per-call
  614     # identity and reap it before callers tear down recordings or their network.
  615     name = f"hg-run-{uuid.uuid4().hex}"
  616     cmd[2:2] = ["--name", name]
  617     try:
  618         return subprocess.run(cmd, capture_output=True, text=True, timeout=timeout_s)
  619     finally:
  620         remove_container(name)
  621

  623 def to_container(path: Path | str, task_dir: Path, task_mount: str) -> str:
  624     """Translate a host path under `task_dir` to its container path.
  625 
  626     Only the task directory is translated. The harness tree is mounted at its own host
  627     path precisely so that the entrypoint argv needs no rewriting, and a translation
  628     is the one place a harness's own record can stop matching the filesystem it ran on.
  629     """
  630     path = Path(path)
  631     try:
  632         rel = path.relative_to(task_dir)
  633     except ValueError:
  634         return str(path)
  635     return task_mount if str(rel) == "." else f"{task_mount.rstrip('/')}/{rel}"

  638 # --------------------------------------------------------------- self check ---
  639 
  640 #: The probe the self-check runs inside a throwaway container. `sh` rather than
  641 #: Python: the image is the task's, and a task image need not have a Python we can
  642 #: rely on. Every line is `key=value` so the result can be read without a parser.
  643 #:
  644 #: `rootfs_writable` writes to `/var/tmp` and not `/usr`, and that is not arbitrary:
  645 #: the container runs as a non-root uid, which cannot write `/usr` in *any* image, so
  646 #: a `/usr` probe reports "read-only" whether or not `--read-only` was passed -- a
  647 #: check that cannot fail. `/var/tmp` is 1777 in a Debian-family image, so it is
  648 #: writable by that uid exactly when the rootfs is not read-only. Measured both ways.
  649 _PROBE = r"""
  650 p() { if [ -e "$1" ]; then echo "$2=yes"; else echo "$2=no"; fi; }
  651 p __PLATFORM__ platform_visible
  652 p __SIBLING__ sibling_visible
  653 p /var/run/docker.sock docker_socket
  654 p __HARNESS__/agent.py harness_readable
  655 if ( echo x > __HARNESS__/probe 2>/dev/null ); then echo harness_writable=yes; else echo harness_writable=no; fi
  656 if ( echo x > __HARNESS__/.state/probe 2>/dev/null ); then echo state_writable=yes; else echo state_writable=no; fi
  657 if ( echo x > __TASK__/written 2>/dev/null ); then echo task_writable=yes; else echo task_writable=no; fi
  658 if ( echo x > /var/tmp/probe 2>/dev/null ); then echo rootfs_writable=yes; else echo rootfs_writable=no; fi
  659 if ( echo x > /tmp/probe 2>/dev/null ); then echo tmp_writable=yes; else echo tmp_writable=no; fi
  660 echo uid=$(id -u)
  661 """
  662

  664 def self_check(platform: Path, image: str) -> dict:
  665     """Prove the container boundary holds before a paid run depends on it.
  666 
  667     Returns {"ok": bool, "detail": str}. Called at startup for a dataset that needs
  668     `exec`: a boundary that cannot be built must stop the run at the door, not once
  669     per task, and never by falling back to running on the host.
  670 
  671     Two of the probes exist because the obvious version of this check passes for the
  672     wrong reason:
  673 
  674     * `platform_visible` probes the **real platform path**, not a marker under `/tmp`.
  675       The container gets a tmpfs on `/tmp`, so a marker placed there would be shadowed
  676       and reported invisible whether or not anything was mounted -- a check that
  677       cannot fail. The real path is also the fact we actually care about.
  678     * `sibling_visible` probes a file next to the harness mount. Without it, "the
  679       platform is invisible" is satisfied by a plan that mounts the harness's whole
  680       parent directory, which would be a far larger hole than intended.
  681     """
  682     try:
  683         pinned = resolve_digest(image)
  684     except ContainerUnavailable as exc:
  685         return {"ok": False, "detail": str(exc)}
  686 
  687     # `/var/tmp`, not `/tmp`: everything under `/tmp` is behind the container's tmpfs.
  688     with tempfile.TemporaryDirectory(prefix="hg-container-check-", dir="/var/tmp") as tmp:
  689         tmp_path = Path(tmp)
  690         task_dir = tmp_path / "task"
  691         task_dir.mkdir()
  692 
  693         harness = tmp_path / "harness"
  694         state = harness / STATE_DIRNAME
  695         state.mkdir(parents=True)
  696         (harness / "agent.py").write_text("# a harness\n")
  697         # A file the harness mount must NOT expose: its sibling.
  698         sibling = tmp_path / "sibling.txt"
  699         sibling.write_text("not the harness\n")
  700 
  701         probe = (_PROBE
  702                  .replace("__PLATFORM__", str(platform / "driver.py"))
  703                  .replace("__SIBLING__", str(sibling))
  704                  .replace("__HARNESS__", str(harness))
  705                  .replace("__TASK__", str(task_dir)))
  706         # The task is mounted at its own host path here, unlike a real run where it is
  707         # mounted at the dataset's `workdir`: the probe asserts host paths that the
  708         # module did not choose.
  709         try:
  710             proc = run(["sh", "-c", probe], image=pinned, task_dir=task_dir,
  711                        task_mount=str(task_dir), harness_root=harness,
  712                        state_dirs=(state,), cwd=str(task_dir), timeout_s=180)
  713         except (subprocess.TimeoutExpired, OSError) as exc:
  714             return {"ok": False, "detail": f"the probe container did not run: {exc}"}
  715 
  716         if proc.returncode != 0:
  717             return {"ok": False,
  718                     "detail": (f"the probe container exited {proc.returncode}: "
  719                                f"{(proc.stderr or proc.stdout).strip()[:400]}")}
  720 
  721     facts = {}
  722     for line in proc.stdout.splitlines():
  723         if "=" in line:
  724             key, _, value = line.partition("=")
  725             facts[key.strip()] = value.strip()
  726 
  727     problems = []
  728     if facts.get("platform_visible") != "no":
  729         problems.append("the platform is visible inside the container")
  730     if facts.get("sibling_visible") != "no":
  731         problems.append("a file beside the harness mount is visible, so the plan "
  732                         "mounts more than the harness")
  733     if facts.get("docker_socket") != "no":
  734         problems.append("the docker socket is visible inside the container, so the "
  735                         "harness can start a sibling container with any mount it likes")
  736     if facts.get("harness_readable") != "yes":
  737         problems.append("the harness tree is not mounted, so the harness cannot run")
  738     if facts.get("harness_writable") != "no":
  739         problems.append("the harness tree is writable, so tamper detection is void")
  740     if facts.get("state_writable") != "yes":
  741         problems.append("the harness's .state/ is not writable: the read-only bind of "
  742                         "its parent won over the writable bind of the hole")
  743     if facts.get("task_writable") != "yes":
  744         problems.append("the task directory is not writable")
  745     if facts.get("rootfs_writable") != "no":
  746         problems.append("the container rootfs is writable, so the environment is not "
  747                         "the image and two runs of one task need not agree")
  748     if facts.get("tmp_writable") != "yes":
  749         problems.append("/tmp is not writable, which several agent CLIs require")
  750 
  751     return {"ok": not problems,
  752             "detail": ("; ".join(problems) if problems
  753                        else f"image {digest_of(pinned)}: " + ", ".join(
  754                            f"{k}={v}" for k, v in sorted(facts.items()))),
  755             "facts": facts, "image_digest": digest_of(pinned)}
  756

  758 # ------------------------------------------------- container-state lifecycle ---
  759 #
  760 # INTERFACE.md §2.5.9. Some tasks put their state in the container's own filesystem
  761 # rather than in a bind mount -- measured on the real Terminal-Bench images: `WORKDIR
  762 # /app` is set by the image, `/app` ships *empty*, and the check reads `/app`. A bind
  763 # mount would put the task somewhere the task itself did not put it.
  764 #
  765 # So the platform creates the container, copies the task in, runs the harness inside it,
  766 # **commits** the result, and verifies against a fresh container from that snapshot.
  767 # Every function here is one step of that, and `eval/run/runner.py` owns the order.

  788 def _require_static_elf(path: Path) -> None:
  789     """Reject loaders and dynamic dependencies in the platform's idle executable.
  790 
  791     An absolute image path alone is unsafe: its executable, ELF loader and preloads
  792     can all belong to the actor. Validate the copied host binary, not image tools.
  793     """
  794     raw = path.read_bytes()
  795     try:
  796         if raw[:4] != b"\x7fELF" or raw[4] not in (1, 2) or raw[5] not in (1, 2):
  797             raise ValueError("not ELF")
  798         endian = "<" if raw[5] == 1 else ">"
  799         layout = "16sHHIIIIIHHHHHH" if raw[4] == 1 else "16sHHIQQQIHHHHHH"
  800         header = struct.unpack_from(endian + layout, raw)
  801         offset, stride, count = header[5], header[9], header[10]
  802         if (header[1] != 2 or not 0 < count <= 4096
  803                 or stride < (32 if raw[4] == 1 else 56)
  804                 or offset + stride * count > len(raw)):
  805             raise ValueError("invalid executable headers")
  806         for i in range(count):
  807             kind, = struct.unpack_from(endian + "I", raw, offset + i * stride)
  808             if kind in (2, 3):  # PT_DYNAMIC / PT_INTERP
  809                 raise ValueError("dynamic executable")
  810     except (IndexError, struct.error, ValueError) as exc:
  811         raise SnapshotUnavailable("snapshot startup requires a static ELF BusyBox") from exc

  814 @contextmanager
  815 def snapshot_startup():
  816     """A trusted idle PID 1; no actor entrypoint, PATH, loader or healthcheck.
  817 
  818     Requires a static BusyBox on the trusted host; absence fails closed. The private
  819     copy remains mounted read-only until verifier removal. Scorer execution itself
  820     still uses the snapshot runtime and is outside this startup guarantee.
  821     """
  822     source = shutil.which("busybox")
  823     if not source:
  824         raise SnapshotUnavailable("snapshot startup requires host static BusyBox")
  825     with tempfile.TemporaryDirectory(prefix="hg-snapshot-idle-") as tmp:
  826         binary = Path(tmp) / "busybox"
  827         try:
  828             shutil.copyfile(source, binary)
  829             _require_static_elf(binary)
  830             binary.chmod(0o755)
  831         except OSError as exc:
  832             raise SnapshotUnavailable(f"snapshot startup unavailable: {exc}") from exc
  833         target = "/" + Path(tmp).name
  834         yield target + "/busybox", ("--mount", f"type=bind,src={tmp},dst={target},readonly")

  837 def create(name: str, *, image: str, limits: dict | None = None,
  838            network: str = "none", extra_mounts: tuple[str, ...] = (),
  839            tmpfs: tuple[str, ...] = (), run_id: str = "", task_id: str = "",
  840            startup_entrypoint: str | None = None) -> str:
  841     """Create a stopped container that stays behind, so it can be committed.
  842 
  843     `sleep infinity` as the command because the image is the task's, and a task image
  844     need not have a long-running entrypoint -- measured: the Terminal-Bench images have
  845     no `CMD` at all, only `FROM`, `WORKDIR` and a `COPY`. The platform brings its own
  846     process to keep the container alive while it copies the task in.
  847 
  848     **Root, unlike everything else here.** These images and their checks need it: the
  849     real Terminal-Bench `test.sh` opens with `apt-get update && apt-get install -y curl`.
  850     The two host-ownership problems that pushed every other container to the invoking uid
  851     are avoided instead of ignored -- the harness tree is mounted read-only, and its
  852     scratch space is a tmpfs that never reaches the host.
  853     """
  854     limits = limits or {}
  855     # **Labelled, because a name is not findable.** A hard kill (SIGKILL, or the harness
  856     # process tree being reaped) skips every `finally`, so these containers survive it --
  857     # and `services.reap` searches by `label=harnessgrad.run=<id>`, which is how the
  858     # *service* containers and the task networks were already findable. The harness
  859     # container had only a name, so the one leak that actually happened was the one a
  860     # reaper could not see. Measured (2026-10-04): `screen-all-1` was killed at 22:42
  861     # with five of these still `running`, and `reap` reported only "network ... has
  862     # active endpoints" because it could not name the containers holding them.
  863     labels = ["--label", f"harnessgrad.run={run_id}"]
  864     if task_id:
  865         labels += ["--label", f"harnessgrad.task={task_id}"]
  866     cmd = ["create", "--name", name, "--network", network, *labels,
  867            "--pids-limit", str(limits.get("pids", 1024)),
  868            "--memory", str(limits.get("memory", "4g")),
  869            "--cpus", str(limits.get("cpus", "2"))]
  870     for path in tmpfs:
  871         cmd += ["--tmpfs", path]
  872     if startup_entrypoint is not None:
  873         cmd += ["--entrypoint", startup_entrypoint, "--no-healthcheck", "--workdir", "/"]
  874     cmd += list(extra_mounts)
  875     # Numeric duration works with static BusyBox versions that reject "infinity".
  876     # The verifier's own timeout/teardown bounds its lifetime, not this idle timer.
  877     cmd += [image, "sleep", "2147483647" if startup_entrypoint else "infinity"]
  878     proc = docker(*cmd, timeout=180)
  879     if proc.returncode != 0:
  880         raise ContainerUnavailable(
  881             f"could not create a container for {image!r}: {proc.stderr.strip()[:300]}")
  882     return name
  883

  885 def cp_in(name: str, source: Path, dest: str, *, contents: bool = False) -> None:
  886     """Copy a host file or directory into a container.
  887 
  888     `docker cp` rather than a bind mount because the point of container state is that the
  889     task lives *in* the container: what is copied in is part of the image the check will
  890     grade, and what is not copied in is not there at all.
  891 
  892     `contents=True` copies what is *inside* the source rather than the source directory
  893     itself, which is what copying a task's initial state into an existing `workdir`
  894     needs. It is a parameter rather than `source / "."` at the call site because
  895     `pathlib` **normalizes that dot away** — measured: `docker cp <setup> <ctr>:/app`
  896     put the task at `/app/setup/...` instead of `/app/...`, and the harness found
  897     nothing, which looked exactly like a harness that could not do the task.
  898     """
  899     src = f"{source}/." if contents else str(source)
  900     proc = docker("cp", src, f"{name}:{dest}", timeout=300)
  901     if proc.returncode != 0:
  902         raise ContainerUnavailable(
  903             f"could not copy {source} into {name}:{dest}: {proc.stderr.strip()[:300]}")

  906 def cp_out(name: str, source: str, dest: Path) -> bool:
  907     """Copy out of a container. Returns False when the source does not exist.
  908 
  909     False rather than an exception because the common call is "read the harness's trace",
  910     and a harness that wrote no trace is a fact to record, not a failure to raise.

  912     dest.parent.mkdir(parents=True, exist_ok=True)

  931 def exec_in(name: str, argv: list[str], *, cwd: str | None = None,
  932             env: dict[str, str] | None = None, timeout_s: int = 3600,
  933             ) -> subprocess.CompletedProcess:

  937         cmd += ["-w", cwd]

  944 def stop(name: str) -> None:
  945     """Stop the container, which is what guarantees nothing it started survives.
  946 
  947     A whole container stopping is a stronger statement than killing one process tree,
  948     and it is free here: `docker commit` works on a stopped container, so the snapshot
  949     is taken from a filesystem on which nothing can still be writing.
  950     """
  951     try:
  952         proc = docker("stop", "--time", "5", name, timeout=60)
  953     except (subprocess.TimeoutExpired, OSError) as exc:
  954         raise _snapshot_failure("stop", exc=exc) from exc
  955     if proc.returncode != 0:
  956         raise _snapshot_failure("stop", proc)

  959 def commit(name: str, tag: str) -> str:
  960     """Snapshot a container into an image, and return its id.

  429 def _copy_verifier_inputs(name: str, spec: dict, staging: Path, task_mount: str) -> None:
  430     """Step 6: put a check's inputs where the check expects them.
  431 
  432     `dst` is a container path; `path` remains the original relative-to-the-workdir form, so
  433     nothing that worked before changes. Split out of `_verify_container_state` so the
  434     *decision* below can be tested without a docker daemon -- the decision is what was wrong.

  439             # already exists, and whether it exists is not the platform's to assume: the
  440             # container is built from the snapshot, which is the **harness's** filesystem.

  447             # happened not to touch `/tests` verified normally, which is what made this
  448             # look like a flaky task rather than a staging bug.
  449             #

  451             # not it exists (measured both ways on a real container). A file input keeps
  452             # the plain form: `docker cp <file> <ctr>:<dst>` overwrites rather than nests.

  454             container_mod.cp_in(name, source, str(entry["dst"]),

  531 def _verify_inner(workdir: Path, verifier: dict | None, answer: str,
  532             *, sandbox: bool, timeout_s: int,
  533             env: dict | None = None,
  534             service_specs: list[dict] | None = None,
  535             running=None, verdict_dir: Path | None = None) -> dict:
  536     """Decide whether the task was completed. Returns {kind, passed, detail}.
  537 
  538     `answer` is the legacy path and stays byte-for-byte what it was: stripped,
  539     lowercased, compared to `expected`. A dataset that declares only `SCORABLE`
  540     reaches exactly this branch, which is what makes adding verifiers a change that
  541     did not touch any existing dataset's numbers.
  542     """
  543     spec = verifier or {"kind": "answer", "expected": None}
  544     kind = spec.get("kind", "answer")
  545     env_spec = env or {"kind": "files"}
  546     is_exec = env_spec.get("kind") == "exec"
  547     #: A `files` check that declares a `reward_file`. The file channels used to be
  548     #: reachable only from a container snapshot (`_verify_container_state`); this is the
  549     #: host spelling of the same contract, added so a no-Docker dataset can keep its
  550     #: evidence. See the block below the command run for what it does.
  551     host_reward = bool(spec.get("reward_file"))
  552     task_mount = env_spec.get("workdir") or container_mod.DEFAULT_WORKDIR
  553     service_specs = list(service_specs or [])
  554 
  555     if kind == "answer":
  556         expected = spec.get("expected")
  557         passed = bool(expected is not None
  558                       and answer.strip().lower() == str(expected).strip().lower())
  559         return {"kind": "answer", "passed": passed,
  560                 "detail": f"answer {answer.strip()[:80]!r} vs {str(expected)[:80]!r}"}
  561 
  562     if kind != "command":
  563         raise ValueError(f"unknown verifier kind {kind!r}")
  564 
  565     # Re-materialize the check's inputs, overwriting whatever the harness left there.
  566     try:
  567         _materialize(workdir, spec.get("inputs"), PLATFORM_ROOT)
  568     except (ValueError, OSError) as exc:
  569         return {"kind": "command", "passed": False,
  570                 "detail": f"could not stage the verifier's inputs: {exc}"}
  571 
  572     argv = [str(a) for a in spec["argv"]]
  573     cwd = workdir / spec["cwd"] if spec.get("cwd") else workdir
  574 
  575     # A host check that declares a `reward_file` is given its verdict directory back
  576     # writable (`writable_paths` below), even though verification freezes the rest of the
  577     # task tree. Created here rather than by the check because bwrap binds a path that
  578     # must already exist, and because the platform -- not the subject -- decides where a
  579     # verdict may be written.
  580     reward_rel = str(spec["reward_file"]).lstrip("/") if host_reward else ""
  581     exec_verdict_dir = verdict_dir if (host_reward and is_exec) else None
  582     reward_dir = (exec_verdict_dir if is_exec else workdir / Path(reward_rel).parent) if host_reward else None
  583     host_reward_path = (workdir / reward_rel) if (host_reward and not is_exec) else None
  584     host_evidence_path = None
  585     if host_reward and not is_exec and spec.get("evidence_file"):
  586         host_evidence_path = workdir / str(spec["evidence_file"]).lstrip("/")
  587     if host_reward and not is_exec:
  588         if (not _verdict_parent_is_safe(host_reward_path, workdir)
  589                 or (host_evidence_path is not None
  590                     and not _verdict_parent_is_safe(host_evidence_path, workdir))):
  591             return {"kind": kind, "passed": False, "score": 0.0,
  592                     "detail": "verdict path uses a subject-controlled symlink"}
  593         _clear_verdict_file(host_reward_path)
  594         if host_evidence_path is not None:
  595             _clear_verdict_file(host_evidence_path)
  596     if reward_dir is not None:
  597         reward_dir.mkdir(parents=True, exist_ok=True)
  598 
  599     try:
  600         if is_exec:
  601             # A **fresh** container from the same image, sharing only the task bind
  602             # mount -- see `eval/container.run` for why the check does not reuse the
  603             # harness's container. Its `cwd` and its `argv` are container paths, and
  604             # it is handed no host environment at all.
  605             #
  606             # With services, the check **joins the task's internal network and they are
  607             # still up** (INTERFACE.md §2.5.7 step 6): §2.2 already allows a check that
  608             # needs a database in a particular state, and a check that needs to ask the
  609             # mock API what it was sent has the same shape. `running` being None here
  610             # would be a caller bug -- a service task whose check silently could not
  611             # reach its service would fail the task for a reason nothing records.
  612             verifier_mounts = list(services_mod.mounts_for(
  613                     "verifier", service_specs,
  614                     running.recordings if running else {}))
  615             if exec_verdict_dir is not None:
  616                 verifier_mounts += ["--mount",
  617                                     f"type=bind,src={exec_verdict_dir},dst={Path(spec['reward_file']).parent}"]
  618             proc = container_mod.run(
  619                 argv, image=spec.get("image") or env_spec["image"], task_dir=workdir,
  620                 task_mount=task_mount, cwd=container_mod.to_container(
  621                     cwd, workdir, task_mount),
  622                 env=_verifier_env_container(),
  623                 timeout_s=timeout_s,
  624                 network=(running.network if running is not None
  625                          else env_spec.get("network", "none")),
  626                 task_dir_writable=False,
  627                 extra_mounts=tuple(verifier_mounts))
  628         elif sandbox:
  629             # A host check whose verdict is a file still runs sandboxed: `writable_paths`
  630             # re-opens exactly its verdict directory on top of the read-only task tree,
  631             # so the reward-file contract does not require switching the namespace off.
  632             # (An unsandboxed check runs with the harness's workdir as `sys.path[0]`,
  633             # which is a route from the subject into a process that can read the
  634             # platform -- the failure this module exists to prevent.)
  635             work_root = Path(os.environ.get("HARNESSGRAD_WORK_ROOT")
  636                              or workdir.parent.parent)
  637             # The namespace freezes the task tree and gives the check a private tmpfs at
  638             # `/tmp`, so HOME/TMPDIR point **there** and not at the workdir. Measured on
  639             # ALFWorld: a check whose toolchain writes `$HOME/.cache` against a read-only
  640             # workdir dies with `[Errno 30] Read-only file system` before it can grade
  641             # anything, and the failure would have been recorded as a harness failure.
  642             check_env = _verifier_env(workdir)
  643             check_env.update({"HOME": "/tmp", "TMPDIR": "/tmp"})
  644             proc = sandbox_mod.run(argv, platform=PLATFORM_ROOT, work_root=work_root,
  645                                    task_dir=workdir, env=check_env,
  646                                    timeout_s=timeout_s,
  647                                    cwd=cwd, repo=workdir, task_dir_writable=False,
  648                                    writable_paths=(reward_dir,) if reward_dir else ())
  649         else:
  650             proc = subprocess.run(argv, capture_output=True, text=True,
  651                                   timeout=timeout_s, cwd=cwd,
  652                                   env=_verifier_env(workdir))
  653         code, out = proc.returncode, proc.stdout
  654     except FileNotFoundError as exc:
  655         return {"kind": "command", "passed": False,
  656                 "detail": f"the verifier command does not exist: {exc}"}
  657     except subprocess.TimeoutExpired:
  658         detail = f"the verifier timed out after {timeout_s}s"
  659         return {"kind": "command", "passed": False,
  660                 "outcome": "timeout", "unmeasured": detail, "detail": detail}
  661 
  662     # ---- a host (`files`) check's reward and evidence files ------------------
  663     #
  664     # The absolute paths in `reward_file`/`evidence_file` are the container contract.
  665     # A `files` environment has no container, so the path is resolved against the task
  666     # workdir -- which *is* that environment's root -- and the check's own `argv` is
  667     # given the same location in the relative spelling its working directory
  668     # understands. `workdir / "verifier/reward.txt"` is therefore where both sides meet.
  669     #
  670     # Before this, the host branch read only the exit code, so a dataset that could not
  671     # use Docker had no way to keep the per-item evidence the contract asks for: the
  672     # run record would carry a bare `exit 1` and nothing a method could learn from.
  673     # **No existing dataset changes**: no dataset declared `reward_file` on a non-exec
  674     # environment before ALFWorld (measured 2026-10-08).
  675     if host_reward:
  676         got = ((exec_verdict_dir / Path(reward_rel).name)
  677                if exec_verdict_dir is not None else host_reward_path)
  678         evidence_path = spec.get("evidence_file")
  679         evidence_file = ((exec_verdict_dir / Path(str(evidence_path)).name)
  680                          if exec_verdict_dir is not None and evidence_path
  681                          else (None if not evidence_path
  682                                else host_evidence_path))
  683         evidence = (None if evidence_file is None or evidence_file.is_symlink()
  684                     else _read_json(evidence_file))
  685 
  686         if got is None or got.is_symlink() or not got.is_file():
  687             # No reward file: the check did not reach a verdict. If it also left no
  688             # evidence, the platform cannot say what happened; if it left evidence that
  689             # says `measured: false`, the task was not measured rather than failed --
  690             # the same distinction the container path draws, and the reason a bare 0 is
  691             # not an acceptable output.
  692             why = f"the check wrote no {spec['reward_file']} (looked for {got}), exit {code}"
  693             verdict = {"kind": kind, "passed": False, "score": 0.0,
  694                        "detail": " | ".join(p for p in
  695                                             [why, _check_output_detail(out, proc.stderr)]
  696                                             if p)[:VERDICT_DETAIL_LIMIT]}
  697             if evidence is not None:
  698                 verdict["evidence"] = evidence
  699                 if evidence.get("measured") is False:
  700                     verdict["unmeasured"] = str(
  701                         evidence.get("error") or "the check could not measure the task")
  702             return verdict
  703 
  704         raw = got.read_text().strip()
  705         try:
  706             score = float(raw)
  707         except ValueError:
  708             return {"kind": kind, "passed": False, "score": 0.0,
  709                     "detail": f"{spec['reward_file']} held {raw!r}, which is not a number"}
  710 
  711         detail = [f"reward {score:g} from {spec['reward_file']} (exit {code})"]
  712         if evidence is not None and evidence.get("why"):
  713             detail.append(str(evidence["why"]))
  714         detail.append(_check_output_detail(out, proc.stderr))
  715         verdict = {"kind": kind, "passed": bool(score), "score": score,
  716                    "detail": " | ".join(p for p in detail if p)[:VERDICT_DETAIL_LIMIT]}
  717         if evidence is not None:
  718             verdict["evidence"] = evidence
  719         if evidence_path and evidence is None:
  720             verdict["unmeasured"] = (f"the check wrote no {evidence_path}: its tests did "
  721                                      f"not run, so this task was not measured")
  722         return verdict
  723 
  724     pass_when = spec.get("pass_when", "exit0")
  725     passed = code == 0
  726     detail = f"exit {code}"
  727     if pass_when == "exit0_stdout":
  728         want = str(spec.get("stdout", ""))
  729         passed = passed and out.strip() == want.strip()
  730         detail = f"exit {code}, stdout {out.strip()[:80]!r} vs {want[:80]!r}"
  731     if not passed and out.strip():
  732         detail += f" | {out.strip()[-200:]}"
  733     return {"kind": "command", "passed": passed, "detail": detail}

   91

   93     configuration is not separation until something enforces it. Without this the

  107     `container_env`'s allowlist already excluded it -- so the leak would have been on
  108     the path with no services at all, which is exactly where nobody looks for one.

  111     improver registry declares each improver's credential variable (`api_key_env`), and
  112     the `deepseek` entry names `DEEPSEEK_API_KEY` -- which matches none of the prefixes
  113     above, so the harness inherited the *improver's* key. Measured: it took making
  114     `deepseek` the default improver (so the name had to exist in `.env`) for the
  115     isolation probe to see it. Both earlier leaks were a new budget with a new prefix;
  116     this one was the same budget under a name the prefix rule could not know. So the
  117     variables the registry declares are stripped by name, and a new improver is covered
  118     the moment it is added rather than when someone remembers the prefix.
  119     """
  120     env = {k: v for k, v in os.environ.items()
  121            if not k.startswith(("HG_METHOD_", "HG_ENV_", "HARNESSGRAD_SEED"))
  122            and k not in _improver_credentials()}
  123     return env
  124

  126 def _improver_credentials() -> set[str]:
  127     """Every environment variable the improver registry declares as a credential.
  128 
  129     Read lazily and defensively: the harness environment must not fail to build because a
  130     config file is missing, and `harness_env()` is called once per task on paths that have
  131     nothing to do with improvers. The registry is read through `tools/improver.py`'s own
  132     loader so there is one reader of that file, not two.
  133     """
  134     try:
  135         from tools.improver import load_config
  136         config = load_config() or {}
  137     except Exception:                                          # noqa: BLE001
  138         return set()
  139     names = {(entry or {}).get("api_key_env")
  140              for entry in (config.get("improvers") or {}).values()}
  141     return {name for name in names if name}

  151 def run_one(repo: Path, task: dict, timeout_s: int | None = None,
  152             sandbox: bool = True, setup: dict | None = None,
  153             verifier: dict | None = None,
  154             verify_timeout_s: int | None = None,
  155             env: dict | None = None,
  156             run_id: str = "", run_seed: str = "",
  157             recordings_root: Path | None = None,
  158             diagnostic_authorization: Authorization | None = None,
  159             trial_id: str | None = None) -> dict:
  160     if diagnostic_authorization is not None:
  161         diagnostic_authorization.require_scope(run_id, [task["task_id"]])
  162     refusal = _refuse_unobserved_policies(
  163         task, env, verifier, run_id=run_id,
  164         diagnostic_authorization=diagnostic_authorization)
  165     result = refusal if refusal is not None else _run_one(
  166         repo, task, timeout_s, sandbox, setup, verifier, verify_timeout_s,
  167         env, run_id, run_seed, recordings_root, trial_id=trial_id)
  168     if result.get("verdict") is not None:
  169         result["verdict"].setdefault("scorer_isolation_class", "unknown")
  170     if diagnostic_authorization is not None:
  171         result["enforcement"] = diagnostic_authorization.record()
  172     return result

  175 def _run_one(repo: Path, task: dict, timeout_s: int | None = None,
  176             sandbox: bool = True, setup: dict | None = None,
  177             verifier: dict | None = None,
  178             verify_timeout_s: int | None = None,
  179             env: dict | None = None,
  180             run_id: str = "", run_seed: str = "",
  181             recordings_root: Path | None = None, *, trial_id: str | None = None) -> dict:
  182     """Run the harness on one task in a throwaway dir, off the platform's disk.
  183 
  184     Returns {task_id, answer, trace, exit_code}. A non-zero exit is recorded as
  185     a harness failure rather than raised: one broken task must not lose a whole
  186     paid round.
  187 
  188     `sandbox=True` runs the subject inside its own mount namespace, so the platform
  189     is not on its disk at all. Paths are bound at their host locations, so the
  190     repository path, `--workdir`, and anything the harness writes into its trace
  191     keep the same spelling they would have had unsandboxed -- a translated path is
  192     a place where a harness's own record stops matching the filesystem it ran on.
  193 
  194     Native tasks can declare network='none'; live model backends are refused for
  195     those tasks before launch. See `eval/containers/sandbox.py`.
  196     """
  197     # Resolve before the subprocess switches cwd: a relative repo path would
  198     # otherwise be interpreted against the sandbox directory and the harness
  199     # would silently fail to launch, scoring zero for the wrong reason.
  200     repo = Path(repo).resolve()
  201     # The environment is a property of the task (INTERFACE.md §2.5). `files` is the
  202     # absence of a declaration, not a separate code path invented for old datasets.
  203     env_spec = env or {"kind": "files"}
  204     is_exec = env_spec.get("kind") == "exec"
  205     try:
  206         protocol = chess_stepwise.protocol_of(task)
  207         stepwise_session = chess_stepwise.session_for(
  208             task, verifier, sandbox=sandbox, env=env_spec)
  209     except ValueError as exc:
  210         return _invalid(task, exc, stage="stepwise admission", service=None)
  211     # Inside a container the task is mounted at the dataset's `workdir` (default
  212     # `/app`), so the harness is told the path it can actually see. Outside one the
  213     # task *is* the host temporary directory. This is the only path translation in
  214     # the runner, and it exists because the dataset names a location in the
  215     # environment rather than on the host.
  216     task_mount = env_spec.get("workdir") or container_mod.DEFAULT_WORKDIR
  217     # Declared per environment: the real task set spans 600-12000 s, and the module
  218     # default of 300 would report a harness failing a task it was never given time to
  219     # attempt (INTERFACE.md §2.5.9).
  220     timeout_s = phase_timeout(timeout_s, env_spec.get("agent_timeout_s"))
  221     verify_timeout_s = phase_timeout(verify_timeout_s, env_spec.get("verify_timeout_s"),
  222                                     (verifier or {}).get("timeout_s"))
  223     agent_runtime = env_spec.get("agent_runtime")
  224     if agent_runtime and is_exec:
  225         raise ValueError("agent_runtime package projections require a files environment")
  226     if not is_exec and not sandbox and (agent_runtime or env_spec.get("network") == "none"):
  227         raise ValueError("declared native runtime/network isolation requires sandbox=True")
  228     # Read before the dispatch: both state kinds need it, and it used to be computed
  229     # further down in the `mount` path only.
  230     _manifest = json.loads((repo / "harness.json").read_text())
  231     if not is_exec:
  232         admit_native_network(env_spec, _manifest, os.environ.get("HG_AGENT_BACKEND", "mock"))
  233     runtime_paths = tuple(Path(p).expanduser()
  234                           for p in (_manifest.get("runtime_paths") or []))
  235     if is_exec and env_spec.get("state") == "container":
  236         return _run_one_container_state(
  237             repo, task, env_spec=env_spec, setup=setup, verifier=verifier,
  238             timeout_s=timeout_s, verify_timeout_s=verify_timeout_s,
  239             run_id=run_id, run_seed=run_seed, recordings_root=recordings_root,
  240             runtime_paths=runtime_paths)
  241     service_specs = list(env_spec.get("services") or [])
  242     web_spec = env_spec.get("web") or {}
  243     running = None
  244     # The internal network, whoever created it: `services.start` when there are
  245     # services, the model gateway when there are not. Tracked separately from `running`
  246     # so teardown knows which of the two it owns.
  247     network_name: str | None = None
  248     owns_network = False
  249     model_gateway = None
  250     web_gateway = None
  251     held = progress_mod.task_of()
  252     phases = progress_mod.phases_for(held)
  253     runtime_dir = None
  254     stepwise_relay = None
  255     stepwise_dir = None
  256     package_overlays = ()
  257     workdir = Path(tempfile.mkdtemp(prefix="hg-task-"))
  258     try:
  259         if agent_runtime:
  260             runtime_dir = Path(tempfile.mkdtemp(prefix="hg-agent-runtime-"))
  261             package_overlays = (stage_agent_runtime(agent_runtime, runtime_dir),)
  262         # The task's initial state, before the harness sees anything. It used to be an
  263         # empty directory, which is why the old contract could only express "answer a
  264         # question": a dataset had nowhere to put the thing the task is *about*.
  265         with phases("setup"):
  266             if setup:
  267                 _materialize(workdir, setup.get("files"), PLATFORM_ROOT)
  268 
  269             task_file = workdir / "task.json"
  270             actor_task = ({**task, "goal": chess_stepwise.GOAL}
  271                           if stepwise_session is not None else task)
  272             task_file.write_text(json.dumps(runtime_task(actor_task, setup)))
  273 
  274         # Services come up before the harness and are health-checked before it starts
  275         # (INTERFACE.md §2.5.7, steps 1-3). A harness that starts against a service
  276         # that is still initialising does not produce a bad measurement, it produces a
  277         # measurement of a race -- so the failure is refused here rather than scored.
  278         if service_specs:
  279             try:
  280                 running = services_mod.start(
  281                     service_specs,
  282                     # A caller that names no root gets a throwaway one beside the task
  283                     # directory: an in-process caller (a test) wants the lifecycle
  284                     # exercised, not a directory in somebody's run tree.
  285                     recordings_root=(recordings_root
  286                                      or workdir.parent / f".recordings-{run_id or 'run'}"),
  287                     run_id=run_id or "run", task_id=task["task_id"],
  288                     run_seed=run_seed or "0")
  289                 network_name = running.network
  290             except services_mod.ServiceUnavailable as exc:
  291                 return _invalid(task, exc, stage=exc.stage, service=exc.service)
  292 
  293         if web_spec:
  294             try:
  295                 if network_name is None:
  296                     network_name = services_mod.create_network(run_id or "run", task["task_id"])
  297                     owns_network = True
  298                 record_base = recordings_root or workdir.parent
  299                 web_gateway = web_gateway_mod.WebGateway(
  300                     bind_host=services_mod.gateway_ip(network_name),
  301                     record_path=record_base / task["task_id"] / "web" / "requests.jsonl",
  302                     enable_search=bool(web_spec.get("search")),
  303                     enable_fetch=bool(web_spec.get("fetch"))).start()
  304             except Exception as exc:
  305                 return _invalid(task, exc, stage="start", service="web")
  306 
  307         manifest = json.loads((repo / "harness.json").read_text())
  308         entry = manifest["entrypoint"]
  309         # Trees this harness needs in order to run at all — a CLI it drives, a
  310         # runtime it was built against. Declared, not inferred: the platform cannot
  311         # know where somebody's agent CLI is installed.
  312         runtime_paths = tuple(Path(p).expanduser()
  313                               for p in (manifest.get("runtime_paths") or []))
  314         # Inside a container the entrypoint is still `<repo>/<entry>` because the
  315         # harness tree is mounted at its own host path; only the task's paths move,
  316         # because the dataset named a location *in the environment*.
  317         #
  318         # The interpreter is **the platform's own** for `exec`, mounted in read-only,
  319         # rather than whatever `python` the image happens to have -- which may be
  320         # nothing (measured: the real Terminal-Bench image has no `python3` at all).
  321         # `files` keeps the bare name, because there the host's PATH is the right answer
  322         # and changing it would change what every existing dataset measures.
  323         task_arg = (container_mod.to_container(task_file, workdir, task_mount)
  324                     if is_exec else str(task_file))
  325         workdir_arg = task_mount if is_exec else str(workdir)
  326         # The image's interpreter when it has one (§2.5.3), the platform's when it does
  327         # not. `files` keeps the bare name: there the host's PATH is the right answer and
  328         # changing it would change what every existing dataset measures.
  329         image_python = container_mod.image_python(env_spec["image"]) if is_exec else None
  330         python = image_python or (sys.executable if is_exec else "python")
  331         argv = [python, str(repo / entry), "--task", task_arg,
  332                 "--workdir", workdir_arg]
  333 
  334         # The harness's own dependencies, from a directory outside the task's
  335         # environment, resolved for **this** interpreter's Python version. `None` is the
  336         # ordinary case: nobody configured one, and the run then behaves exactly as it
  337         # did before this existed. See `eval/containers/harness_runtime.py` for why this is not a
  338         # derived image and not the platform's interpreter.
  339         overlay = harness_runtime.lookup(
  340             repo, python, image=(env_spec["image"] if is_exec else None),
  341             image_digest=(env_spec.get("image_digest") if is_exec else None))
  342         if overlay:
  343             # Only the paths are collected here; the environment is applied below, where
  344             # `child_env` exists. It used to be written here, one screen above its own
  345             # definition -- a `NameError` on the one branch nobody had configured yet,
  346             # which is how it survived: `overlay` is `None` unless a harness runtime is
  347             # declared, and then the run that declares one is the run that crashes.
  348             runtime_paths = tuple(runtime_paths) + harness_runtime.extra_paths(overlay)
  349 
  350         # The harness's own writable scratch space, inside its otherwise read-only
  351         # tree. Exported so a harness need not know the platform's layout, and
  352         # stated in the environment rather than passed as an argument because the
  353         # entrypoint contract (INTERFACE.md §1.2) is `--task` and `--workdir` only.
  354         state_dir = repo / STATE_DIRNAME
  355         # Created on demand, not just at staging time: a workspace committed before
  356         # `.state/` existed would otherwise make every run refuse to start, and the
  357         # sandbox needs a real directory to mount over. Empty and gitignored, so it
  358         # cannot move a sha.
  359         state_dir.mkdir(exist_ok=True)
  360 
  361         # A writable HOME, and the XDG directories, per task.
  362         #
  363         # This is what makes an agent CLI usable as a harness at all. The sandbox
  364         # starts from an empty root -- only /usr, /bin, /sbin, /lib*, /etc and the
  365         # trees it is told to bind exist inside -- so the inherited `HOME`, pointing at
  366         # a real `/home/<user>` that does not exist there, is a path a CLI cannot even
  367         # `mkdir` under. Every one of them (Claude Code, Codex, OpenCode, ...) writes
  368         # session state, config and caches under HOME, and dies immediately if it
  369         # cannot.
  370         #
  371         # Under `workdir` rather than somewhere of its own, because `workdir` is
  372         # already bound read-write and is already a per-task temporary directory: the
  373         # isolation this needs is the isolation it already has, and inventing a second
  374         # mount for it would be a second thing to get wrong. Named with the platform's
  375         # prefix so it cannot be mistaken for a task's own file.
  376         #
  377         # The same directory is created for a container, where it is reached as
  378         # `<workdir>/.harnessgrad-home` through the task bind mount -- so the two
  379         # environments agree on where a CLI keeps its state without agreeing on how
  380         # that path is spelled on the host.
  381         home = workdir / ".harnessgrad-home"
  382         (home / "tmp").mkdir(parents=True, exist_ok=True)
  383 
  384         if is_exec:
  385             # An allowlist, and container-side HOME/XDG/TMPDIR: forwarding the host
  386             # values would name directories that do not exist inside the image.
  387             child_env = container_mod.container_env(task_mount, deadline_s=timeout_s)
  388         else:
  389             child_env = harness_env()
  390             child_env.update({
  391                 "HOME": str(home),
  392                 "XDG_CONFIG_HOME": str(home / ".config"),
  393                 "XDG_DATA_HOME": str(home / ".local" / "share"),
  394                 "XDG_CACHE_HOME": str(home / ".cache"),
  395                 # A CLI's scratch files should not land in the directory the task is
  396                 # about, where a verifier would then be reading them.
  397                 "TMPDIR": str(home / "tmp"),
  398             })
  399         child_env["HG_AGENT_DEADLINE_S"] = str(timeout_s)
  400         child_env["HARNESSGRAD_STATE"] = str(state_dir)
  401         if overlay:
  402             child_env.update(harness_runtime.env_for(overlay))
  403 
  404         # Publish the agent's model onto the task's network. A containerised harness
  405         # cannot reach a model on the host's loopback -- measured, every obvious route
  406         # fails -- and the wrong fix is `--network bridge`, which would give the task the
  407         # internet and silently change what it is (`eval/service/modelgate.py`).
  408         if is_exec:
  409             base = child_env.get("HG_AGENT_BASE_URL")
  410             if base and os.environ.get("HG_AGENT_BACKEND", "mock") != "mock" \
  411                     and modelgate_mod.is_host_local(base):
  412                 if network_name is None:
  413                     network_name = services_mod.create_network(
  414                         run_id or "run", task["task_id"])
  415                     owns_network = True
  416                 target, path = modelgate_mod.split(base)
  417                 model_gateway = modelgate_mod.ModelGateway(
  418                     bind_host=services_mod.gateway_ip(network_name),
  419                     target=target, path=path).start()
  420                 # The harness is handed the published address, and nothing else about
  421                 # the mechanism: `HG_AGENT_*` is the contract it already reads.
  422                 child_env["HG_AGENT_BASE_URL"] = model_gateway.base_url
  423 
  424         # How the harness finds the services, and the whole of what it is told about
  425         # them. `HG_SVC_*` rather than `HG_ENV_*`: the environment's own model,
  426         # credential and seed go to the service containers only (§2.5.7).
  427         if running is not None:
  428             child_env.update(services_mod.harness_env(service_specs))
  429         if web_gateway is not None:
  430             child_env["HG_SVC_WEB_URL"] = web_gateway.url
  431 
  432         # On a `services` environment the harness joins the task's internal network,
  433         # which is how "reachable by name, not on the internet" is delivered. Passing
  434         # `env_spec["network"]` straight through would ask docker for a network called
  435         # `services`, which does not exist.
  436         # With a gateway published, the harness joins the internal network so it can
  437         # reach it. Otherwise the declared value stands.
  438         network = (network_name if network_name is not None
  439                    else env_spec.get("network", "none"))
  440 
  441         # Everything the harness needs to run at all, plus the platform's interpreter.
  442         # The platform's interpreter is mounted only when it is going to be used, so an
  443         # image that brings its own does not also get the host's Python on its disk.
  444         runtime = ((tuple(runtime_paths)
  445                     + (() if image_python else container_mod.interpreter_paths()))
  446                    if is_exec else ())
  447 
  448         if stepwise_session is not None:
  449             # Project only this read-only socket directory. Neither gold nor session
  450             # state is written beneath the actor's writable task directory.
  451             stepwise_dir = Path(tempfile.mkdtemp(prefix="hg-stepwise-"))
  452             stepwise_relay = StepwiseRelay(
  453                 stepwise_dir, stepwise_session, contract=chess_stepwise.STEPWISE,
  454                 public_fields=chess_stepwise.PUBLIC_FIELDS, timeout_s=timeout_s)
  455             child_env["HG_STEPWISE_SOCKET"] = str(stepwise_relay.path)
  456 
  457         def launch():
  458             if is_exec:
  459                 # `runtime_paths` is deliberately not applied: it exists because a
  460                 # host CLI lives outside `/usr`, and inside a container the runtime is
  461                 # the image's. Mounting a host tree in would defeat the point of
  462                 # having an image at all.
  463                 return container_mod.run(
  464                     argv, image=env_spec["image"], task_dir=workdir,
  465                     task_mount=task_mount, harness_root=repo,
  466                     state_dirs=(state_dir,), cwd=task_mount, env=child_env,
  467                     timeout_s=timeout_s, network=network,
  468                     # Stated rather than omitted: the harness gets **no** view of a
  469                     # service's recording, and the empty list is that decision.
  470                     readonly_paths=runtime,
  471                     extra_mounts=tuple(services_mod.mounts_for(
  472                         "harness", service_specs,
  473                         running.recordings if running else {})))
  474             if not sandbox:
  475                 return subprocess.run(argv, capture_output=True, text=True,
  476                                       timeout=timeout_s, cwd=workdir, env=child_env)
  477             # `repo` is `<work_root>/<run-id>/workspace`, so the work root -- the one
  478             # tree that must stay visible -- is two levels up. Read from the
  479             # platform's own environment rather than guessed from the run id.
  480             work_root = Path(os.environ.get("HARNESSGRAD_WORK_ROOT")
  481                              or repo.parent.parent)
  482             return sandbox_mod.run(argv, platform=PLATFORM_ROOT,
  483                                    work_root=work_root, task_dir=workdir,
  484                                    env=child_env, timeout_s=timeout_s, cwd=workdir,
  485                                    repo=repo, state_dirs=(state_dir,),
  486                                    runtime_paths=runtime_paths,
  487                                    visible_paths=((stepwise_dir,) if stepwise_dir else ()),
  488                                    network=env_spec.get("network"),
  489                                    package_overlays=package_overlays)
  490 
  491         try:
  492             with phases("agent"):
  493                 if stepwise_relay is not None:
  494                     stepwise_relay.start()
  495                 try:
  496                     proc = launch()
  497                 finally:
  498                     if stepwise_relay is not None:
  499                         stepwise_relay.stop()
  500             exit_code, stderr = proc.returncode, proc.stderr[-2000:]
  501         except subprocess.TimeoutExpired:
  502             exit_code, stderr = -1, f"timeout after {timeout_s}s"
  503         except OSError:
  504             if stepwise_relay is None:
  505                 raise
  506             exit_code, stderr = -1, "stepwise harness launch failed"
  507         except sandbox_mod.SandboxUnavailable as exc:
  508             # Not swallowed. A run that asked for a sandbox and silently got none
  509             # produces numbers that mean something other than what they say.
  510             raise SystemExit(f"refusing to run: sandbox requested but unavailable: "
  511                              f"{exc}. Pass --no-sandbox to accept that a harness "
  512                              f"can reach the platform.")
  513 
  514         with phases("collect"):
  515             answer_path = workdir / "answer.txt"
  516             answer = (answer_path.read_text()
  517                       if stepwise_relay is None and answer_path.exists() else "")
  518             trace_path = workdir / "trace.jsonl"
  519             trace = trace_path.read_text() if trace_path.exists() else ""
  520 
  521         # The harness has exited. Grade the controller's completed live exchange,
  522         # or its frozen artifacts on the existing offline path (§2.2 and §2.4).
  523         with phases("verify"):
  524             if stepwise_relay is not None:
  525                 verdict = chess_stepwise.finish(stepwise_relay, exit_code)
  526             else:
  527                 verdict = _verify(workdir, verifier, answer, sandbox=sandbox, task_id=task["task_id"],
  528                                   run_id=run_id, trial_id=trial_id,
  529                                   timeout_s=verify_timeout_s, env=env_spec,
  530                                   service_specs=service_specs, running=running)
  531                 if protocol is not None:
  532                     verdict["protocol"] = protocol
  533 
  534         # The environment's own spend, read from the recordings **before teardown**:
  535         # the containers are about to be removed and their evidence with them.
  536         env_usage = running.usage() if running is not None else {}
  537 
  538         return {"task_id": task["task_id"], "answer": answer,
  539                 "trace": trace, "exit_code": exit_code, "stderr": stderr,
  540                 "verdict": verdict, "env_usage": env_usage,
  541                 "harness_runtime": harness_runtime.record(
  542                     overlay, platform_interpreter=(image_python is None)),
  543                 "runtime_boundary": {
  544                     "agent_timeout_s": timeout_s,
  545                     "verifier_timeout_s": verify_timeout_s,
  546                     "agent_network": env_spec.get("network", "inherited") if not is_exec else network,
  547                     "agent_package_projected": bool(package_overlays),
  548                     "verifier_package": "trusted"},
  549                 # `account()` and not just "published": the connection counts are what
  550                 # turn a harness-side `Connection error` into a named cause. Read while
  551                 # the gateway is still up -- `finally` stops it right after this returns.
  552                 "model_gateway": ({"published": True,
  553                                    "target": f"{model_gateway.target[0]}:{model_gateway.target[1]}",
  554                                    **model_gateway.account()}
  555                                   if model_gateway is not None else None),
  556                 "web_gateway": ({"published": True, "url": web_gateway.url,
  557                                  "search": web_gateway.search, "fetch": web_gateway.fetch}
  558                                 if web_gateway is not None else None),
  559                 "identity": _last_identity(trace)}
  560     finally:
  561         if stepwise_relay is not None:
  562             stepwise_relay.stop()
  563         if stepwise_dir is not None:
  564             shutil.rmtree(stepwise_dir, ignore_errors=True)
  565         # Step 8, and in a `finally` for the reason the table in §2.5.7 gives: a leaked
  566         # network would be inherited by the next task, where it would look like
  567         # isolation that is not there. `stop()` never raises.
  568         if model_gateway is not None:
  569             model_gateway.stop()
  570         if web_gateway is not None:
  571             web_gateway.stop()
  572         if owns_network and network_name is not None:
  573             leaked_net = services_mod.remove_network(network_name)
  574             if leaked_net:
  575                 print(f"warning: could not remove the model gateway network for "
  576                       f"{task['task_id']}: {leaked_net}", file=sys.stderr)
  577         if running is not None:
  578             leaked = running.stop()
  579             if leaked:
  580                 # Not raised. The task's result is the platform's report about the
  581                 # *task*, and a teardown problem after it is a fact about the run.
  582                 print(f"warning: could not fully tear down services for "
  583                       f"{task['task_id']}: {leaked}", file=sys.stderr)
  584         shutil.rmtree(workdir, ignore_errors=True)
  585         if runtime_dir is not None:
  586             shutil.rmtree(runtime_dir, ignore_errors=True)

  625 def _provider_failure(result: dict) -> dict | None:
  626     """harness 是不是**因为连不上模型端点**而死的。`None` 表示不是。
  627 
  628     判据刻意很窄,只在这种情况成立:
  629 
  630       * harness 以**非零退出码**结束 —— 它自己没把这次故障吸收掉;
  631       * stderr 的**尾部**有客户端连接/超时的痕迹 —— 尾部,不是全文:一个连上又掉了、
  632         但自己重试成功的 harness 会在中间留下同样的字眼,而那是它做对了。
  633 
  634     退出码 0 的一律不算:它跑完了,分数就是它该得的分数。
  635     """
  636     if not result.get("exit_code"):            # None(无效)或 0 都不算
  637         return None
  638     tail = (result.get("stderr") or "")[-1500:]
  639     for marker in _PROVIDER_FAILURES:
  640         if marker in tail:
  641             # 摘出那一行,放进记录里 —— 读者要的是"哪一句"。
  642             line = next((l.strip() for l in reversed(tail.splitlines())
  643                          if marker in l), marker)
  644             return {"marker": marker, "line": line[:240],
  645                     "exit_code": result.get("exit_code")}
  646     return None

  649 def _invalid(task: dict, exc, *, stage: str, service: str | None) -> dict:
  650     """A task the platform could not measure, as opposed to one it measured as zero.
  651 
  652     INTERFACE.md §2.5.7: the result is **not** a `per_task` score and **not** a zero,
  653     and it is attributed to the dataset and the service by name. The distinction is the
  654     entire reason this platform exists — a broken mock API and a weak harness must not
  655     be the same number — and it is independently attested: JarvisBench reports that
  656     "harness or provider failures are marked invalid rather than assigned a score of
  657     zero" (`INTERFACE.md` §2.5.7).
  658     """
  659     return {"task_id": task["task_id"], "invalid": {
  660                 "stage": stage, "service": service, "detail": str(exc)},
  661             "answer": "", "trace": "", "exit_code": None, "stderr": "",
  662             "env_usage": {}, "model_gateway": None, "identity": None}

  667 def _spread(xs: list[float]) -> float:
  668     """Population standard deviation, with no dependency and no ambiguity.
  669 
  670     The population form because these are *all* the trials this run took, not a sample
  671     from a larger population of runs -- `statistics.pstdev` is the honest name for it, and
  672     the difference from `stdev` matters most at small n, which is exactly where a reader
  673     compares two harnesses.
  674     """
  675     if len(xs) < 2:
  676         return 0.0
  677     mu = sum(xs) / len(xs)
  678     return (sum((x - mu) ** 2 for x in xs) / len(xs)) ** 0.5

  681 def _harness_failure(result: dict) -> dict | None:
  682     """Why a harness left nothing to read, when it left nothing.
  683 
  684     A task can be **measured** and still say nothing about the harness: the check reads
  685     the artifact, finds it wrong, and records a 0.0 -- and that 0.0 is a real score. But
  686     if the harness *also* wrote no trace, then no method has anything to read while the
  687     curve reports a number, and those two situations are indistinguishable on the curve.
  688 
  689     Measured, and the reason this exists: a harness that needed a package the task image
  690     did not have died on its first model call with `ModuleNotFoundError`, exit code 1.
  691     The trace was 0 bytes, the score was 0.000, and the run reported *nothing* -- the
  692     exit code and stderr were computed here and then dropped, so "the method decided not
  693     to edit" was the only reading available. A harness that cannot start is a fact about
  694     the run and belongs in the record.
  695 
  696     Every harness the platform ships writes `trace.jsonl`, so an empty one is always
  697     abnormal; a harness that simply chose to write nothing is worth knowing about too.
  698     Deliberately **not** folded into `invalid`: the task was measured, so it keeps its
  699     score. This is the attribution, not a reason to drop the measurement (§2.5.7).
  700 
  701     **The second trigger, and the measured case that forced it.** A non-empty trace is
  702     not proof that the harness ran. Measured 2026-10-04 on the current platform:
  703     `--harness cli_agent` with no CLI writes exactly one line,
  704     `{"error": "HG_CLI is unset: no CLI to run"}`, never calls a model, exits 2 -- and the
  705     curve said `score 0.0, harness_failed: null, invalid: null`. The console warned and
  706     the verdict file carried the exit code, so the fact existed in two places and not on
  707     the point, which is where a reader compares runs.
  708 
  709     So: a harness that **exited non-zero without recording a single model call** is a
  710     harness failure too. "A model call" is read from the `usage` block, which the harness
  711     contract requires (`INTERFACE.md` §1.2) -- the platform is entitled to rely on it, and
  712     it is the only schema-independent evidence that a harness talked to a model. A
  713     scripted harness that legitimately makes no calls (`base_harness/oracle`) exits 0 and
  714     is not flagged.
  715     """
  716     trace = (result.get("trace") or "").strip()
  717     reason = None
  718     if not trace:
  719         reason = "the harness left no trace at all"
  720     elif result.get("exit_code") and _last_usage(trace) is None:
  721         # **只说平台知道的事。** 第一版这条写的是"没有记录到任何模型调用",而那是错的:
  722         # `usage` 块是 harness 跑完之后才写一次的,所以**中途崩溃**永远看起来像"一次都没调
  723         # 过模型"。实测(2026-10-04,本地模型,5 道题):三道题各有 15-23 次调用、轨迹 12-14 万
  724         # 字节,控制台却按"left no trace"播报 —— 一句假话,而且是我自己上一版引入的。
  725         reason = (f"the harness exited {result['exit_code']} and its trace has no `usage` "
  726                   f"block, so the run did not reach its end")
  727     if reason is None:
  728         return None
  729     return {"reason": reason, "exit_code": result.get("exit_code"),
  730             "stderr": (result.get("stderr") or "")[-1500:]}

  733 def _reporter():
  734     """The reporter an evaluation should narrate through.
  735 
  736     The context first, then the console's own. Both are needed and neither is
  737     redundant:
  738 
  739     * the **context** is how a task's phases reach the code that runs them
  740       (`eval/containers/container.py` beats while it waits inside a single `docker exec`);
  741     * the **console** is what makes that context exist at all when the driver has not
  742       set one. Measured: the driver's own calls to `evaluate` (`driver.py` round 0, the
  743       mode B loop, the eval side) run on the main thread with no context set, and the
  744       first version of this put the progress events *only* in the context -- so a real
  745       run reported the method call's phases and nothing at all for the four tasks it
  746       spent the hour on. A run that narrates its shortest wait and hides its longest
  747       is worse than one that narrates neither.
  748     """
  749     from eval.observe.console import CONSOLE
  750     return progress_mod.current() or CONSOLE.reporter

  785 def _in_parallel(tasks: list[dict], measure, jobs: int) -> dict[str, dict]:
  786     """Measure every task, up to `jobs` at a time, and collect the outcomes by task id.
  787 
  788     `jobs=1` stays a plain sequential loop, because the platform's default record must not
  789     depend on a thread pool to be the record it has always been.
  790 
  791     Why this axis and not another: on a measured three-task run the agent phase was 87.8% of
  792     the wall clock (2578 s of 2936 s) and the platform sat idle for all of it -- the tasks are
  793     independent by construction (own container, own network, own model gateway, own workspace,
  794     own cache key), so this is the cheap parallelism. Splitting *inside* a task is not
  795     available: the harness is a process the platform does not control.
  796 
  797     The first exception is re-raised after cancelling what has not started. A task already
  798     running is left to finish: `run_one` tears its own containers down in a `finally`, and
  799     abandoning a thread mid-teardown is how a leaked network gets inherited by whatever runs
  800     next (§2.5.7).
  801     """
  802     if jobs <= 1:
  803         # Computed in task order, which is what makes the merge reproducible at one
  804         # `--jobs` setting as well as comparable across two of them.
  805         return {task["task_id"]: measure(index, task)
  806                 for index, task in enumerate(tasks, start=1)}
  807     pool = futures.ThreadPoolExecutor(max_workers=jobs, thread_name_prefix="hg-task")
  808     pending = [pool.submit(measure, index, task)
  809                for index, task in enumerate(tasks, start=1)]
  810     outcomes: dict[str, dict] = {}
  811     try:
  812         for done in futures.as_completed(pending):
  813             outcome = done.result()
  814             outcomes[outcome["task_id"]] = outcome
  815             # **Stop paying for a harness that cannot start.** See `_all_unstarted`: the
  816             # failure is task-independent, so `jobs` agreements are enough, and the rest
  817             # of the round can only return the same `0.000` at full price.
  818             if _all_unstarted(outcomes, jobs):
  819                 for fut in pending:
  820                     fut.cancel()
  821                 break
  822     except BaseException:
  823         pool.shutdown(wait=True, cancel_futures=True)
  824         raise
  825     pool.shutdown(wait=True)
  826     # **A cancelled task is not a task that scored nothing.** The merge indexes `outcomes`
  827     # by every task, so the cancelled ones are marked `not_run` and kept out of `per_task`
  828     # rather than given a zero the platform never measured (§2.5.7).
  829     for task in tasks:
  830         outcomes.setdefault(task["task_id"],
  831                             {"task_id": task["task_id"], "not_run": True})
  832     return outcomes
  833 
  834 
  835 def _all_unstarted(outcomes: dict[str, dict], jobs: int) -> bool:
  836     """Has the harness failed to *start* on every task measured so far, `jobs` of them?
  837 
  838     This is the platform's own question (`_harness_failure`) asked about a round instead of
  839     a task. A harness whose entrypoint cannot be invoked fails identically on every task --
  840     that is the task-independent part of a run -- so `jobs` agreements are enough to stop
  841     the round. Measured (`codex-dgm`, 2026-10-05): the improver's edit dropped the
  842     `--task`/`--workdir` contract, all ten tasks produced `FileNotFoundError: '--task'`
  843     with no trace, and that happened in **two consecutive rounds** -- twenty
  844     task-measurements whose only possible contribution was `0.000` over and over.
  845 
  846     Deliberately not "any task failed to start": a task-specific failure (an image with no
  847     interpreter, say) is a fact about *that task*, and stopping the round on it would trade
  848     one wrong record for another.
  849     """
  850     if len(outcomes) < jobs:
  851         return False
  852     for outcome in outcomes.values():
  853         if outcome.get("not_run") or outcome.get("invalid") or "cached" in outcome:
  854             return False
  855         measured = outcome.get("measured") or []
  856         if not measured or not _harness_failure(measured[-1]):
  857             return False
  858     return True

  861 def _evaluate(repo: Path, tasks: list[dict], scorable: dict[str, str],
  862               cache: dict | None = None, harness_sha: str = "",
  863               sandbox: bool = True, setups: dict | None = None,
  864               verifiers: dict | None = None,
  865               envs: dict | None = None,
  866               round_no: int = 0,
  867               run_id: str = "", run_seed: str = "",
  868               progress_dir: Path | None = None,
  869               recordings_root: Path | None = None,
  870               trials: int = 1, jobs: int = 1,
  871               diagnostic_authorization: Authorization | None = None) -> dict:
  872     """Score a harness on a task set.
  873 
  874     `trials` is how many times each `(harness, task)` pair is measured. One is the
  875     default and one is not enough: measured on this platform, **the same unmodified
  876     harness scored 0.333 in one run and 0.000 in the next**, because the agent model
  877     intermittently emits a tool call in a dialect the harness cannot parse and the harness
  878     then ends the task. Nothing about that is a property of the dataset; it is the floor of
  879     what a single sample can tell you, and a curve that does not say how noisy it is cannot
  880     support a claim that an edit helped. With `trials > 1` a task's score is the mean over
  881     trials and `per_task_std` is their spread (population standard deviation).
  882 
  883     `jobs` is how many tasks are measured **at the same time**, and it multiplies with
  884     `trials` rather than replacing it: trials buy a spread, jobs buy wall clock. Both are
  885     recorded -- `trials` on the point, `jobs` in the run's own parameters -- because a run
  886     that used four containers at once is a different claim about this host than one that
  887     used one.
  888     """
  889     if diagnostic_authorization is not None:
  890         diagnostic_authorization.require_scope(
  891             run_id, [t["task_id"] for t in tasks], trials=trials, jobs=jobs, round_no=round_no)
  892         if cache:
  893             raise ValueError("unobserved diagnostic requires an empty cache; no cached measurements")
  894         # A later ordinary caller must not reuse a diagnostic's provisional score.
  895         cache = {}
  896     if trials < 1:
  897         raise ValueError(f"trials must be at least 1, not {trials}")
  898     if jobs < 1:
  899         raise ValueError(f"jobs must be at least 1, not {jobs}")
  900     evaluation_protocols = {task["task_id"]: protocol for task in tasks
  901                             if (protocol := chess_stepwise.protocol_of(task)) is not None}
  902     if len(set(evaluation_protocols.values())) > 1:
  903         raise ValueError("cannot aggregate interactive and offline ChessBench protocols")
  904 
  905     def measurement_key(task):
  906         key = f"{harness_sha}:{task['task_id']}:{trials}"
  907         # A changed scorer declaration must not reuse another runtime's score.
  908         bundle = ((verifiers or {}).get(task["task_id"]) or {}).get("scorer_bundle")
  909         if bundle is not None:
  910             key += ":scorer:" + hashlib.sha256(json.dumps(
  911                 bundle, sort_keys=True, default=str).encode()).hexdigest()
  912         if evaluation_protocols.get(task["task_id"]) == chess_stepwise.STEPWISE:
  913             key += ":" + chess_stepwise.STEPWISE
  914         return key
  915 
  916     def trial_identity(trial: int) -> str:
  917         # The ordinal repeats in each round and harness revision within an outer run.
  918         return f"round-{round_no}.harness-{harness_sha}.trial-{trial + 1}"
  919 
  920     cache = cache if cache is not None else {}
  921     per_task, traces, verdicts = {}, {}, {}
  922     per_task_scorer_isolation = {}
  923     #: One score per trial, per task, kept so a reader can see the spread rather than
  924     #: only its mean. Absent when `trials == 1`.
  925     per_task_trials: dict[str, list[float]] = {}
  926     per_task_std: dict[str, float] = {}
  927     #: **Every** trial's trace and verdict, so a task that passed twice out of three can
  928     #: be explained (`traces`/`verdicts` keep only the last trial, for compatibility).
  929     #: Absent when `trials == 1`, where they would repeat the two dicts above.
  930     trial_traces: dict[str, list[str]] = {}
  931     trial_verdicts: dict[str, list[dict]] = {}
  932     #: Tasks whose measurement was **reused** rather than taken, by task id. Recorded on
  933     #: the point because a reused measurement is not an independent one: with the cache
  934     #: content-addressed (see `driver.py`), an unchanged tree has exactly one measurement,
  935     #: and a reader who sees the same number twice must be able to tell that from two
  936     #: measurements that happened to agree.
  937     cached_tasks: list[str] = []
  938     #: Tasks the platform could not measure, as opposed to ones it measured as zero.
  939     #: Kept out of `per_task` entirely (INTERFACE.md §2.5.7): a task that never ran is
  940     #: not a task that scored nothing, and putting a `0.0` here is the specific lie
  941     #: this whole mechanism exists to prevent.
  942     invalid: dict[str, dict] = {}
  943     #: Tasks the round **stopped short of**, because the harness had already failed to
  944     #: start on every task it had tried. Kept out of `per_task` for the same reason
  945     #: `invalid` is: those tasks were never measured, and a `0.0` for them would be the
  946     #: platform's invention rather than its verdict.
  947     not_run: list[str] = []
  948     #: Tasks the platform *did* measure but whose harness left no trace, with the exit
  949     #: code and the tail of its stderr. Kept apart from `invalid` on purpose: an invalid
  950     #: task is absent from `per_task` because there is no measurement, while these keep
  951     #: their score. Same split the platform makes everywhere else -- the attribution is
  952     #: not a reason to throw the number away.
  953     harness_failed: dict[str, dict] = {}
  954     #: What each harness **printed**, per task, kept whether it failed or not.
  955     #:
  956     #: `harness_failed` is the narrow version of this: it fires only when the trace is
  957     #: empty, and it answers "did the harness start?". A task that scored 0 with a
  958     #: non-empty trace still has a question attached to it ("it ran, so what did it
  959     #: say?"), and the answer was in `result["stderr"]` and thrown away. Measured: four
  960     #: of five tasks at 0.0 on `loop-terminal_bench-26452c` with no explanation anywhere
  961     #: in the run; the explanation had to be reconstructed by hand from the traces.
  962     harness_output: dict[str, dict] = {}
  963     #: The check's own tally, as a rate per task. **A second, finer signal beside
  964     #: `score`.** A `command` check usually reports "4 passed 1 failed" and the reward file
  965     #: collapses that to 0. Measured on this platform: a task that passed 4 of 5 tests
  966     #: scored 0.0, and a harness edit that took it to 3 of 5 also scored 0.0 -- the curve
  967     #: said "no change" while the artifact got *worse*, and a task set of ten binary scores
  968     #: (resolution 0.1, noise +/-0.1) cannot resolve the effect the platform exists to
  969     #: measure. `score` stays the headline and stays comparable; this is the same
  970     #: measurement at the resolution the check already provides.
  971     #:
  972     #: Only tasks whose check reports tests appear, so `test_rate_n_tasks` travels beside
  973     #: the mean -- a 0.8 over three tasks and a 0.8 over ten are not the same claim.
  974     per_task_test_rate: dict[str, float] = {}
  975     per_task_test_rate_trials: dict[str, list[float]] = {}
  976     tests_passed_total = 0
  977     tests_total_total = 0
  978     #: Which interpreter ran the harness on each task and where its own dependencies
  979     #: came from (`eval/containers/harness_runtime.py`). Distinct records, because the version is a
  980     #: property of the task's image and one run may cross several.
  981     runtime_records: list[dict] = []
  982     declarations: list[dict] = []
  983     tampered: dict[str, list[str]] = {}
  984     env_usage = {"input": 0, "output": 0, "calls": 0, "services_reported": 0}
  985     #: Whether the platform had to publish the agent's model onto the task's network.
  986     #: A fact about the measurement: a containerised harness that could not reach a
  987     #: model would score zero for a reason that is not about the harness.
  988     model_gateway: dict | None = None
  989 
  990     # The harness is arbitrary code the platform did not write, and it runs with
  991     # the platform on disk. One pair of hashes around the whole task set, not one
  992     # pair per task: a 400-task benchmark would otherwise hash the platform 400
  993     # times, and the cost of the check would scale with the cost of the run it is
  994     # protecting. Granularity is not lost -- the report names files, and which
  995     # task touched them is not something a failing run needs.
  996     before = snapshot(PLATFORM_ROOT)
  997 
  998     # ---- one task, `trials` times, as a function ----------------------------------
  999     #

  111 def service_env(services: list[dict], *, task_id: str, run_seed: str) -> dict[str, str]:
  112     """What the *services* get, and the harness never does.
  113 
  114     A third model budget, after the harness's and the method's, separated the same way
  115     the other two are: `harness_env()` strips `HG_METHOD_*`, and this is handed only to
  116     the service containers. An environment whose simulator can spend the agent's
  117     credential is an environment that can impersonate the agent.
  118     """
  119     # A service is scoped to one task.  Exposing this identifier lets an image bind its
  120     # API to the current task without trusting the request body to choose a dataset row.
  121     env: dict[str, str] = {"HG_ENV_TASK_ID": task_id}
  122     if any(svc.get("needs_seed") for svc in services):
  123         env["HG_ENV_SEED"] = task_seed(run_seed, task_id)
  124     if any(svc.get("needs_model") for svc in services):
  125         # The third model budget, from the declared configuration rather than from
  126         # `os.environ` at the point of use. The variable names stay the same because a
  127         # service image reads them by name; only the platform's read moved.
  128         environment = current_config().environment
  129         for key, value in (("HG_ENV_MODEL", environment.model),
  130                            ("HG_ENV_API_KEY", environment.api_key),
  131                            ("HG_ENV_BASE_URL", environment.base_url)):

  158 class RunningServices:
  159     """The containers started for one task, and the network they are on.
  160 
  161     Held as one object so that teardown cannot be forgotten piecemeal: `stop` is the
  162     only way to end a task's services and it removes the network too. A leaked network
  163     is the failure mode here — containers carry `--rm`, networks do not.
  164     """
  165 
  166     def __init__(self, *, network: str, containers: dict[str, str],
  167                  recordings: dict[str, Path], services: list[dict]):
  168         self.network = network
  169         self.containers = containers
  170         self.recordings = recordings
  171         self.services = services
  172 
  173     def urls(self) -> dict[str, str]:
  174         return {svc["name"]: f"http://{svc['name']}:{svc['port']}"
  175                 for svc in self.services}
  176 
  177     def usage(self) -> dict:
  178         """The environment's own model spend, read from the recordings.
  179 
  180         Summed from `harnessgrad.json` in each recording rather than asked for over the
  181         network: the service has stopped or is about to, and a number the platform asks
  182         for is a number it can be lied to about. `eval/run/runner.py` reads a harness's
  183         usage from its trace for the same reason.
  184         """
  185         total: dict[str, int] = {}
  186         reported = 0
  187         for name, path in self.recordings.items():
  188             record = path / USAGE_FILE
  189             if not record.is_file():
  190                 continue
  191             try:
  192                 payload = json.loads(record.read_text())
  193             except (OSError, json.JSONDecodeError):
  194                 continue
  195             usage = payload.get("usage")
  196             if not isinstance(usage, dict):
  197                 continue
  198             reported += 1
  199             for key in ("input_tokens", "output_tokens", "calls"):
  200                 value = usage.get(key)
  201                 if isinstance(value, int):
  202                     total[key] = total.get(key, 0) + value
  203         total["services_reported"] = reported
  204         return total
  205 
  206     def stop(self) -> list[str]:
  207         """Tear down containers and the network. Never raises.
  208 
  209         Never raises because it runs in a `finally`, and a teardown that can throw
  210         replaces the real failure with a misleading one. It returns what it could not
  211         remove so a caller that cares (a test) can say so.
  212         """
  213         problems: list[str] = []
  214         for name, cid in self.containers.items():
  215             # --rm destroys the service's only startup/action diagnostics. Keep them
  216             # beside its recording before removal, including on failed health checks.
  217             record = self.recordings.get(name)
  218             if record is not None:
  219                 try:
  220                     logs = container_mod.docker("logs", "--tail", "200", cid, timeout=10)
  221                     (record / "service.log").write_text(
  222                         logs.stdout + logs.stderr, encoding="utf-8")
  223                 except (subprocess.TimeoutExpired, OSError):
  224                     pass  # Evidence collection must never prevent teardown.
  225             try:
  226                 proc = container_mod.docker("rm", "--force", cid, timeout=60)
  227             except (subprocess.TimeoutExpired, OSError) as exc:
  228                 problems.append(f"service {name!r}: {exc}")
  229                 continue
  230             if proc.returncode != 0 and "No such container" not in proc.stderr:
  231                 problems.append(f"service {name!r}: {proc.stderr.strip()[:200]}")
  232         problems.extend(remove_network(self.network))
  233         return problems

  255 def remove_network(network: str) -> list[str]:
  256     """Remove a network. Never raises, like `RunningServices.stop`."""
  257     try:
  258         proc = container_mod.docker("network", "rm", network, timeout=60)
  259     except (subprocess.TimeoutExpired, OSError) as exc:
  260         return [f"network {network!r}: {exc}"]
  261     # Docker 29 uses "network NAME not found"; older daemons use "No such
  262     # network". Reaping followed by a finally-block stop must remain idempotent.
  263     missing = ("No such network" in proc.stderr or
  264                f"Error response from daemon: network {network} not found"
  265                in proc.stderr.splitlines())
  266     if proc.returncode != 0 and not missing:
  267         return [f"network {network!r}: {proc.stderr.strip()[:200]}"]
  268     return []

  271 def reap(run_id: str) -> list[str]:
  272     """Remove containers and networks a **previous crash** left under this run id.
  273 
  274     Measured (2026-10-04): a run killed mid-task leaves its per-task networks and any
  275     running service container behind -- `local-floor-5/6` and `local-cap12` each left
  276     one network per task, and a probe run left a service container up for 15 hours.
  277     `docker run --rm` does not help: it removes a container when it *exits*, and a
  278     killed driver never stops it. Ctrl-C is handled (`except BaseException` around the
  279     service start path), but SIGTERM and SIGKILL are not, and neither is a test that
  280     is interrupted.
  281 
  282     The cost is not disk. It is that the **next** run with the same id dies at
  283     `docker network create ... already exists`, so a measurement is lost to a failure
  284     that belongs to the previous crash and is reported as if it were about this run.
  285 
  286     Called at the run's first write, beside the same decision for the run directory
  287     (`_write_run_meta(..., fresh=True)`): a reused id means "start clean". Only this
  288     run id's objects are touched, so a different run executing concurrently is not
  289     disturbed. Never raises; returns what it could not remove.
  290     """
  291     problems: list[str] = []
  292     selector = f"label=harnessgrad.run={run_id}"
  293     #: And by **name**, because the label is not always there. The harness container was
  294     #: created with only `--name` until 2026-10-04 (`eval/containers/container.py:create` now labels
  295     #: it too), so the containers a hard-killed run actually leaves behind are exactly the
  296     #: ones a label search cannot see -- measured on `screen-all-1`: five containers still
  297     #: `running`, and this function reported only "network ... has active endpoints".
  298     #: `hg-<run_id>-` is the platform's own naming (`_task_name`), so matching it cannot
  299     #: reach another run's objects.
  300     prefix = f"hg-{run_id}-"
  301     for kind, list_args, remove_args, by_name in (
  302             ("container", ["container", "ls", "--all", "--quiet"], ["rm", "--force"],
  303              True),
  304             ("network", ["network", "ls", "--quiet"], ["network", "rm"], True)):
  305         found: list[str] = []
  306         for flt in ([selector] + ([f"name={prefix}"] if by_name else [])):
  307             try:
  308                 listed = container_mod.docker(*list_args, "--filter", flt, timeout=60)
  309             except (subprocess.TimeoutExpired, OSError) as exc:
  310                 problems.append(f"{kind} list ({flt}): {exc}")
  311                 continue
  312             if listed.returncode != 0:
  313                 problems.append(f"{kind} list ({flt}): {listed.stderr.strip()[:200]}")
  314                 continue
  315             found += [i for i in listed.stdout.split() if i not in found]
  316         for ident in found:
  317             try:
  318                 removed = container_mod.docker(*remove_args, ident, timeout=120)
  319             except (subprocess.TimeoutExpired, OSError) as exc:
  320                 problems.append(f"{kind} {ident[:12]}: {exc}")
  321                 continue
  322             if removed.returncode != 0:
  323                 problems.append(f"{kind} {ident[:12]}: {removed.stderr.strip()[:200]}")
  324     return problems

  327 def gateway_ip(network: str) -> str:

  329

  344 def start(services: list[dict], *, recordings_root: Path, run_id: str,
  345           task_id: str, run_seed: str) -> RunningServices:
  346     """Create the network, start every service, and wait for each to be healthy.
  347 
  348     Raises `ServiceUnavailable` — having already torn down whatever it started, so a
  349     failed task does not leave a network behind for the next one to inherit.
  350     """
  351     # The same safety net `eval/container.run` carries, for the same reason: the driver
  352     # pins every service image before the run starts (§2.5.4), and a service started
  353     # from a tag would pull whatever the registry served that day.
  354     for svc in services:
  355         if not container_mod.looks_pinned(svc["image"]):
  356             raise ServiceUnavailable(
  357                 f"service {svc['name']!r} names {svc['image']!r}, which is not pinned "
  358                 f"to a content address. Resolve it with `container.resolve_digest` "
  359                 f"first (INTERFACE.md §2.5.4).", service=svc["name"], stage="start")
  360 
  361     net = create_network(run_id, task_id)
  362 
  363     containers: dict[str, str] = {}
  364     recordings: dict[str, Path] = {}
  365     try:
  366         # Configuration can fail after network creation; keep it inside the same
  367         # cleanup boundary as container startup.
  368         env = service_env(services, task_id=task_id, run_seed=run_seed)
  369         for svc in services:
  370             name = svc["name"]
  371             record_host: Path | None = None
  372             if svc.get("record"):
  373                 # Under the **run directory**, with the round diffs and the curve: a
  374                 # recording is evidence the run produced, and the place evidence goes
  375                 # is the record. An earlier version put it under the work root, where
  376                 # nothing ever removed it -- and a per-task directory that no code owns
  377                 # is a directory that grows for the life of the machine.
  378                 record_host = recordings_root / task_id / name
  379                 record_host.mkdir(parents=True, exist_ok=True)
  380                 recordings[name] = record_host
  381 
  382             cmd = ["run", "--detach", "--rm",
  383                    "--name", label(f"hg-{run_id}-{task_id}-{name}"),
  384                    # The container's *name* is platform-mangled -- it has to be unique
  385                    # on the daemon, and several runs may be live at once -- so the name
  386                    # the harness was told must be attached as an **alias**, or it
  387                    # resolves to nothing. Measured: without this the harness got
  388                    # `Name or service not known` for a service that was running and
  389                    # healthy, so every task that touched it failed for a reason nothing
  390                    # recorded, and the tasks that only touched the filesystem passed.
  391                    "--network", net, "--network-alias", name,
  392                    "--user", f"{os.getuid()}:{os.getgid()}",
  393                    "--label", f"harnessgrad.run={run_id}",
  394                    "--label", f"harnessgrad.task={task_id}",
  395                    "--label", f"harnessgrad.service={name}"]
  396             # Service resources are opt-in; task ENV caps do not cover sidecars.
  397             # Reuse the container envelope (including its fixed PID cap) only when
  398             # the service declares resources, preserving existing uncapped services.
  399             if "cpus" in svc or "memory_mb" in svc:
  400                 limits = container_mod.limits_from(svc)
  401                 cmd += ["--cpus", str(limits["cpus"]),
  402                         "--memory", limits["memory"],
  403                         "--pids-limit", str(limits["pids"])]
  404             if record_host is not None:
  405                 cmd += ["--mount",
  406                         f"type=bind,src={record_host},dst={RECORDINGS_ROOT}/{name}"]
  407             for key, value in sorted(env.items()):
  408                 cmd += ["-e", f"{key}={value}"]
  409             # No docker socket, and no host tree: a service is infrastructure the
  410             # harness must not be able to read or kill its way out of (§2.5.8).
  411             if "entrypoint" in svc:
  412                 cmd += ["--entrypoint", svc["entrypoint"]]
  413             cmd += [svc["image"], *svc.get("command", [])]
  414             started = container_mod.docker(*cmd, timeout=180)
  415             if started.returncode != 0:
  416                 raise ServiceUnavailable(
  417                     f"service {name!r} did not start: {started.stderr.strip()[:300]}",
  418                     service=name, stage="start")
  419             containers[name] = started.stdout.strip()
  420 
  421         running = RunningServices(network=net, containers=containers,
  422                                   recordings=recordings, services=services)
  423         for svc in services:
  424             _await_health(running, svc)
  425         return running
  426     except BaseException:
  427         # Including `KeyboardInterrupt` and `ServiceUnavailable` itself: the one thing
  428         # that must not happen is a network surviving a failed start and being reused
  429         # by the next task, where it would look like isolation that is not there.
  430         RunningServices(network=net, containers=containers,
  431                         recordings=recordings, services=services).stop()
  432         raise

  435 def _await_health(running: RunningServices, svc: dict) -> None:
  436     """Run the service's own health check until it passes or its budget runs out.
  437 
  438     The check runs **inside the service container**, so it sees the service as the
  439     service sees itself — a host-side port probe would answer a different question and
  440     would be defeated by a container that binds to localhost only.
  441     """
  442     health = svc["health"]
  443     argv = [str(a) for a in health["argv"]]
  444     budget = float(health.get("timeout_s", 60))
  445     deadline = time.monotonic() + budget
  446     last = ""
  447     cid = running.containers[svc["name"]]
  448     while True:
  449         try:
  450             proc = container_mod.docker("exec", cid, *argv, timeout=max(10, int(budget)))
  451         except (subprocess.TimeoutExpired, OSError) as exc:
  452             last = str(exc)
  453         else:
  454             if proc.returncode == 0:
  455                 return
  456             last = f"exit {proc.returncode}: {(proc.stdout + proc.stderr).strip()[:200]}"
  457         if time.monotonic() >= deadline:
  458             raise ServiceUnavailable(
  459                 f"service {svc['name']!r} was not healthy after {budget:g}s "
  460                 f"({last}). This is a dataset-side failure, not a harness result: the "
  461                 f"task is recorded as invalid rather than scored zero.",
  462                 service=svc["name"], stage="health")
  463         time.sleep(HEALTH_POLL_S)
  464

  466 # --------------------------------------------------------------- mount rules ---
  467 
  468 #: Who may see a service's recording. **One place, because the rule is the design.**
  469 #: The harness is absent from this mapping, and that absence is the point: a recording
  470 #: is the only input in this platform the harness cannot even read, and a decision with
  471 #: no code is a decision that gets reversed by accident.
  472 _AUDIENCES = ("harness", "verifier")
  473 
  474 
  475 def mounts_for(audience: str, services: list[dict],
  476                recordings: dict[str, Path]) -> list[str]:
  477     """Docker mount arguments for one audience.
  478 
  479     `harness` → nothing at all. Not read-only, not a copy: the harness cannot see a
  480     recording, because there is no static version to withhold and a copy written before
  481     the run would be a copy of nothing.
  482 
  483     `verifier` → every recording, read-only. Read-only because a check that writes into
  484     the evidence it grades has a verdict that depends on its own leftovers, the same
  485     reason the task mount is frozen for the check (§2.5.6).
  486     """
  487     if audience not in _AUDIENCES:
  488         raise ValueError(f"unknown audience {audience!r}; known: {list(_AUDIENCES)}")
  489     if audience == "harness":
  490         return []
  491     mounts: list[str] = []
  492     for svc in services:
  493         path = recordings.get(svc["name"])
  494         if path is not None:
  495             mounts += ["--mount",
  496                        f"type=bind,src={path},dst={RECORDINGS_ROOT}/{svc['name']},readonly"]
  497     return mounts
