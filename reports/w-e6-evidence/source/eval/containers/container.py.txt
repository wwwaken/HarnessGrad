    1 """Run a task's harness and its check inside a container. INTERFACE.md §2.5.
    2 
    3 This is the `exec` environment kind. It is a **second, independent boundary**, not a
    4 replacement for `eval/containers/sandbox.py`:
    5 
    6 * `files` -- the harness runs on the host inside a bwrap namespace. The platform is
    7   hidden by not being mounted (`eval/containers/sandbox.py`), and the runtime is the host's.
    8 * `exec`  -- the harness runs **inside a container**. The platform is hidden the same
    9   way (it is not mounted), but the runtime, the interpreter and the tools are the
   10   image's, and `network: none` becomes available because the container has its own
   11   network namespace rather than borrowing the host's.
   12 
   13 Three properties are load-bearing and each is a decision rather than an accident:
   14 
   15 **The container is mounted from a digest, never a tag.** `:latest` moves, and two runs
   16 under one tag are not two measurements of the same thing -- the same failure this
   17 platform already shipped once for `agent_model`. `resolve_digest` is therefore not a
   18 convenience: a run whose digest cannot be resolved **refuses to start** (§2.5.4).
   19 
   20 **Nothing is built here.** A task image is built by `tools/import_env.py`, once, by a
   21 human, at import time (§2.5.5). `docker run` on a missing image would silently pull
   22 whatever the registry served that day; this module does not pull and does not build.
   23 If the image is not on the local daemon, that is a refusal with instructions.
   24 
   25 **The container is not given a way out of itself.** No docker socket, no host home, no
   26 platform. It gets exactly three mounts (§2.5.3), and the order they are applied in is
   27 load-bearing -- see `_mounts`.
   28 """

   75 # A snapshot is a Docker image layer. Bound both the writable layer and the space Docker
   76 # can consume while committing it; otherwise a large GAIA attachment or JVM cache can turn
   77 # commit into a host OOM/disk failure and the task is incorrectly reported as a harness score.
   78 SNAPSHOT_MAX_BYTES = 16 << 30
   79 SNAPSHOT_RESERVE_BYTES = 1 << 30
   80 SNAPSHOT_INSPECT_TIMEOUT_S = 60
   81 SNAPSHOT_COMMIT_TIMEOUT_S = 1800

   84 def _snapshot_limit(key: str, default: int) -> int:
   85     try:
   86         value = int(os.environ.get(key, default))
   87         if not 0 < value <= 2**63 - 1:
   88             raise ValueError("outside positive signed 64-bit range")
   89         return value
   90     except (TypeError, ValueError) as exc:
   91         raise SnapshotUnavailable(f"snapshot admission failed kind=resource: invalid {key}") from exc

   94 def _snapshot_failure(stage: str, proc: subprocess.CompletedProcess | None = None,
   95                       exc: BaseException | None = None) -> SnapshotUnavailable:
   96     text = " ".join(filter(None, [
   97         str(exc) if exc else "",
   98         (proc.stderr if proc else "") if proc else "",
   99         (proc.stdout if proc else "") if proc else "",
  100     ])).strip()
  101     lowered = text.lower()
  102     if ((proc is not None and proc.returncode == 137)
  103             or "out of memory" in lowered or "cannot allocate memory" in lowered
  104             or "oom" in lowered):
  105         kind = "oom"
  106     elif any(marker in lowered for marker in
  107              ("no space left", "disk quota", "quota exceeded", "insufficient space")):
  108         kind = "disk"
  109     elif any(marker in lowered for marker in
  110              ("resource temporarily unavailable", "too many processes", "pids limit")):
  111         kind = "resource"
  112     else:
  113         kind = "docker"
  114     code = f" exit={proc.returncode}" if proc is not None else ""
  115     return SnapshotUnavailable(f"snapshot {stage} failed kind={kind}{code}: {text[-1200:]}")

  118 def snapshot_admission(name: str) -> dict[str, int | str | bool]:
  119     """Check bounded writable size and Docker-root headroom before ``docker commit``.
  120 
  121     This is byte-capacity preflight, not a filesystem quota or inode/resource gate.
  122     A commit killed by the daemon can leave an incomplete image and used to surface
  123     as an unexplained exit 137. The shared runner admission contract covers the
  124     additional observations a backend must supply for those stronger gates.
  125     """
  126     maximum = _snapshot_limit("HG_SNAPSHOT_MAX_BYTES", SNAPSHOT_MAX_BYTES)
  127     reserve = _snapshot_limit("HG_SNAPSHOT_RESERVE_BYTES", SNAPSHOT_RESERVE_BYTES)
  128     try:
  129         proc = docker("inspect", "--size", "-f", "{{.SizeRw}}", name,
  130                       timeout=SNAPSHOT_INSPECT_TIMEOUT_S)
  131     except (subprocess.TimeoutExpired, OSError) as exc:
  132         raise _snapshot_failure("admission inspect", exc=exc) from exc
  133     if proc.returncode != 0:
  134         raise _snapshot_failure("admission inspect", proc)
  135     try:
  136         writable = int(proc.stdout.strip())
  137     except ValueError:
  138         raise SnapshotUnavailable(
  139             "snapshot admission failed kind=docker: invalid writable size")
  140     if not 0 <= writable <= 2**63 - 1:
  141         raise SnapshotUnavailable("snapshot admission failed kind=docker: invalid writable size")
  142     if writable > maximum:
  143         raise SnapshotUnavailable(
  144             f"snapshot admission rejected kind=disk writable_bytes={writable} "
  145             f"max_bytes={maximum}")
  146 
  147     try:
  148         info = docker("info", "--format", "{{.DockerRootDir}}",
  149                       timeout=SNAPSHOT_INSPECT_TIMEOUT_S)
  150     except (subprocess.TimeoutExpired, OSError) as exc:
  151         raise _snapshot_failure("admission docker-info", exc=exc) from exc
  152     if info.returncode != 0:
  153         raise _snapshot_failure("admission docker-info", info)
  154     root_text = info.stdout.strip()
  155     root = Path(root_text)
  156     if (not root.is_absolute() or len(root_text) > 1024
  157             or any(ord(c) < 32 for c in root_text)):
  158         raise SnapshotUnavailable("snapshot admission failed kind=docker: invalid Docker root")
  159     try:
  160         free = shutil.disk_usage(root).free
  161     except OSError as exc:
  162         raise _snapshot_failure("admission disk-usage", exc=exc) from exc
  163     if type(free) is not int or not 0 <= free <= 2**63 - 1:
  164         raise SnapshotUnavailable("snapshot admission failed kind=disk: invalid free bytes")
  165     required = writable + reserve
  166     if free < required:
  167         raise SnapshotUnavailable(
  168             f"snapshot admission rejected kind=disk docker_root={root} free_bytes={free} "
  169             f"required_bytes={required} writable_bytes={writable}")
  170     return {"writable_bytes": writable, "free_bytes": free, "required_bytes": required,
  171             "docker_root": str(root), "max_bytes": maximum, "reserve_bytes": reserve,
  172             "inspect_timeout_s": SNAPSHOT_INSPECT_TIMEOUT_S,
  173             "commit_timeout_s": SNAPSHOT_COMMIT_TIMEOUT_S,
  174             "preflight_only": True, "quota_enforced": False}

  179 #: How much of a streamed command's output is kept. `docker exec` on a long task can
  180 #: print for hours, and only the tail is ever read (the runner keeps `stderr[-2000:]`).
  181 #: A cap rather than a spool to disk: the point of the cap is that a chatty subject
  182 #: cannot fill the machine's memory with output nobody will read.
  183 _STREAM_KEEP = 4 << 20

  186 def _drain(pipe, sink: list[str], limit: int = _STREAM_KEEP) -> None:
  187     """Read a pipe to EOF, keeping the last `limit` bytes. Runs on its own thread.
  188 
  189     Interleaving stdout and stderr is what `subprocess.run` would not do, and it is
  190     what makes a live log readable: `docker exec` writes its own errors and the
  191     subject's output to two different pipes, and a reader wants them in the order they
  192     were said.
  193     """
  194     try:
  195         while True:
  196             chunk = pipe.readline()
  197             if not chunk:
  198                 break
  199             sink.append(chunk)
  200             if sum(len(c) for c in sink) > limit:
  201                 del sink[:max(1, len(sink) // 2)]
  202     except (ValueError, OSError):
  203         pass
  204     finally:
  205         try:
  206             pipe.close()
  207         except OSError:
  208             pass

  211 def _run_streaming(argv: list[str], timeout: int, _label: str = ""):
  212     """`subprocess.run` shape, but in slices so this thread stays interruptible.
  213 
  214     Only used while a task is in flight (`eval/observe/progress.py`). The beats themselves come
  215     from the *phase*, not from here -- one thread that beats every 10s regardless of
  216     which call is blocking, which is why this loop only has to wait and check a
  217     deadline. The first version beat here as well and produced two heartbeats per
  218     interval, each with a different elapsed time, which is a log that contradicts
  219     itself: measured, `elapsed_s: 10.0` twice in a row.
  220 
  221     The cost -- two reader threads, merged output -- is paid only inside a task, so the
  222     30-odd short docker calls a task makes (`create`, `cp`, `commit`) are untouched.
  223     """
  224     if timeout:
  225         deadline = time.monotonic() + timeout
  226     else:
  227         deadline = None
  228     proc = subprocess.Popen(argv, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
  229                             text=True, bufsize=1)
  230     out_chunks: list[str] = []
  231     err_chunks: list[str] = []
  232     readers = [threading.Thread(target=_drain, args=(proc.stdout, out_chunks),
  233                                 daemon=True),
  234                threading.Thread(target=_drain, args=(proc.stderr, err_chunks),
  235                                 daemon=True)]
  236     for r in readers:
  237         r.start()
  238     #: One second, so that a timeout is enforced to within about a second -- the same
  239     #: promise `subprocess.run`'s own timeout makes -- and short enough that a stopping
  240     #: driver is noticed promptly.
  241     SLICE = 1.0
  242     while True:
  243         try:
  244             proc.wait(timeout=SLICE)
  245             break
  246         except subprocess.TimeoutExpired:
  247             if deadline is not None and time.monotonic() > deadline:
  248                 proc.kill()
  249                 proc.wait()
  250                 for r in readers:
  251                     r.join(timeout=2)
  252                 raise subprocess.TimeoutExpired(argv, timeout)
  253     for r in readers:
  254         r.join(timeout=5)
  255     return subprocess.CompletedProcess(argv, proc.returncode, "".join(out_chunks),
  256                                        "".join(err_chunks))

  259 def docker(*args: str, timeout: int = 120) -> subprocess.CompletedProcess:
  260     """Run the docker client.
  261 
  262     Public because `eval/service/services.py` needs the same client, and two modules each
  263     shelling out to `docker` with their own idea of how to report a failure is two
  264     error surfaces to keep in step.
  265 
  266     Long calls made *inside a task* stream and beat; everything else is one
  267     `subprocess.run`. The branch is on the presence of a task rather than on a flag per
  268     call site, because the calls that need it are the two that block for minutes
  269     (`exec` a harness, `exec` a check) and they are made from three places, none of
  270     which knows whether a watcher exists.
  271     """
  272     from eval.observe import progress as progress_mod
  273 
  274     if progress_mod.task_of() is None:
  275         return subprocess.run(["docker", *args], capture_output=True, text=True,
  276                               timeout=timeout)
  277     return _run_streaming(["docker", *args], timeout, f"docker {args[0]}")
  278

  295 def looks_pinned(image: str) -> bool:
  296     """True for `repo@sha256:<64 hex>` and for a bare `sha256:<64 hex>` image id."""
  297     if "@sha256:" in image:
  298         return _is_hex(image.rsplit("@sha256:", 1)[1])
  299     if image.startswith("sha256:"):
  300         return _is_hex(image[len("sha256:"):])
  301     return False

  305     return len(value) == 64 and all(c in "0123456789abcdef" for c in value.lower())

  308 def resolve_digest(image: str) -> str:
  309     """Pin `image` to a content address, locally, without pulling or building.
  310 
  311     Returns a reference `docker run` accepts -- `repo@sha256:...` when the image
  312     carries a registry digest, else the bare image ID, which is itself a content hash
  313     of the local image and therefore pinnable.
  314 
  315     Refuses rather than pulling. A pull during a run is a network dependency that
  316     makes the run depend on what the registry served that day; importing the image is
  317     a one-time step (`tools/import_env.py`), which is where §2.5.5 puts it.
  318     """
  319     if looks_pinned(image):
  320         return image
  321     if image in _DIGEST_CACHE:
  322         return _DIGEST_CACHE[image]
  323 
  324     try:
  325         proc = _docker("image", "inspect", "--format",
  326                        "{{if .RepoDigests}}{{index .RepoDigests 0}}{{else}}{{.Id}}{{end}}",
  327                        image, timeout=60)
  328     except subprocess.TimeoutExpired as exc:
  329         raise ContainerUnavailable(f"timed out inspecting image {image!r}") from exc
  330     if proc.returncode != 0:
  331         raise ContainerUnavailable(
  332             f"image {image!r} is not on the local daemon, and this platform does not "
  333             f"pull during a run. Import it first:\n"
  334             f"    python tools/import_env.py --image {image}\n"
  335             f"docker said: {proc.stderr.strip()[:300]}")
  336 
  337     pinned = proc.stdout.strip()
  338     if not looks_pinned(pinned):
  339         raise ContainerUnavailable(
  340             f"could not pin image {image!r} to a content address; "
  341             f"`docker image inspect` returned {pinned!r}")
  342     _DIGEST_CACHE[image] = pinned
  343     return pinned
  344 
  345 
  346 #: One `docker image inspect` per distinct reference, not one per task: resolution is
  347 #: a startup step, and a 400-task benchmark should not fork the docker client 400
  348 #: times to learn something that cannot change during a run.
  349 _DIGEST_CACHE: dict[str, str] = {}

  361 def interpreter_paths() -> tuple[Path, ...]:
  362     """The trees the platform's own interpreter needs, so a container can run it.
  363 
  364     A harness is a Python program -- the platform already assumes that when it invokes
  365     `python <entry>` -- and **a task image need not have a Python at all**. Measured: the
  366     real Terminal-Bench image is `ubuntu:24.04` plus a `COPY`, with no `python3`, no `R`
  367     and no `gcc`; the task explicitly expects the agent to `apt-get install` what it
  368     needs. So assuming the image provides the interpreter is assuming something the
  369     benchmark does not promise.
  370 
  371     Both `sys.prefix` and `sys.base_prefix` are needed, and that is measured too: a
  372     virtualenv's `bin/python3` resolves to its base prefix, so mounting the venv alone
  373     gives `not found` inside the container. Together they give a working interpreter
  374     **and** the packages the platform installed for its harnesses.
  375     """
  376     import sys as _sys
  377     out: list[Path] = []
  378     for candidate in (_sys.prefix, _sys.base_prefix):
  379         path = Path(candidate)
  380         if path.is_dir() and path not in out:
  381             out.append(path)
  382     return tuple(out)

  389 def image_python(image: str) -> str | None:
  390     """The interpreter the image itself provides, or None if it has none.
  391 
  392     The platform injects its own interpreter into a container, because a task image need
  393     not have one -- measured: the real Terminal-Bench image is `ubuntu:24.04` plus a
  394     `COPY`, with no `python3` at all. But injecting it unconditionally would contradict
  395     §2.5.3: the reason the harness runs *inside* the environment is so that "`bash`, the
  396     file editor and **the interpreter** are the image's".
  397 
  398     So the image wins when it has an interpreter, and the platform's is the fallback.
  399     Both halves are measured: a `files`-style image with its own Python keeps it (and
  400     keeps its own installed packages), and an image with none still runs the harness.
  401     """
  402     if image in _PYTHON_CACHE:
  403         return _PYTHON_CACHE[image]
  404     found: str | None = None
  405     try:
  406         proc = docker("run", "--rm", "--network", "none", "--entrypoint", "sh", image,
  407                       "-c", "command -v python3 || command -v python || true",
  408                       timeout=180)
  409         out = proc.stdout.strip()
  410         found = out.splitlines()[-1].strip() if out else None
  411     except (subprocess.TimeoutExpired, OSError):
  412         found = None
  413     _PYTHON_CACHE[image] = found or None
  414     return _PYTHON_CACHE[image]

  428 def container_env(task_mount: str, deadline_s: int | None = None) -> dict[str, str]:
  429     """Everything a harness sees in its environment inside the container.
  430 
  431     Built from an allowlist rather than forwarded, and with `PATH` deliberately absent:
  432     the container has its own, the host's names directories that do not exist inside,
  433     and forwarding it would let a harness pick up a host binary if the two ever
  434     overlapped.
  435 
  436     **`HG_AGENT_DEADLINE_S` is the environment's own `agent_timeout_s`.** It is not a
  437     hint: the platform kills the container at that second, and until this existed a
  438     harness that spawns a subprocess had no way to know the number it was being measured
  439     against. Measured (`codex-task-dgm`, 2026-10-05): an adapter that gave its own child a
  440     900 s timeout raced the platform's 900 s kill -- the container died first and the
  441     curve recorded `exit code -1 -- timeout after 900s`, a killed container instead of a
  442     harness that decided to stop. The budget a harness is subject to is a fact about the
  443     measurement, so it is stated where the harness can read it (INTERFACE.md §3.3).
  444     """
  445     env = {k: v for k, v in os.environ.items()
  446            if k.startswith(PASS_THROUGH_PREFIXES) or k in PASS_THROUGH_EXACT}
  447     home = env_home(task_mount)
  448     env.update({
  449         "HOME": home,
  450         "XDG_CONFIG_HOME": f"{home}/.config",
  451         "XDG_DATA_HOME": f"{home}/.local/share",
  452         "XDG_CACHE_HOME": f"{home}/.cache",
  453         # A CLI's scratch files should not land in the directory the task is about,
  454         # where a check would then be reading them.
  455         "TMPDIR": f"{home}/tmp",
  456     })
  457     if deadline_s:
  458         env["HG_AGENT_DEADLINE_S"] = str(int(deadline_s))
  459     return env
  460 
  461 
  462 #: The `container_env` entries whose value is a **directory a harness is invited to write
  463 #: into**. `PATH` and the model variables are deliberately absent: a name belongs here only
  464 #: when something is expected to put a file underneath it.
  465 SCRATCH_ENV_VARS = ("HOME", "XDG_CONFIG_HOME", "XDG_DATA_HOME", "XDG_CACHE_HOME", "TMPDIR")
  466

  468 def writable_env_dirs(task_mount: str) -> tuple[str, ...]:
  469     """Every directory `container_env` names that a harness may write into.
  470 
  471     **A variable that names a directory is not a directory.** `docker create --tmpfs
  472     <home>` brings its own mount point and nothing below it, so `HOME` exists while
  473     `HOME/tmp` -- the value this module hands out as `TMPDIR` -- does not. A harness is
  474     entitled to believe the environment it was given: `mkdtemp(TMPDIR + "/scratch-XXXXXX")`
  475     needs the *parent* to be there, and `ENOENT` when it is not is a failure caused by the
  476     platform, not by the harness.
  477 
  478     Measured 2026-10-06, on the `dsh_agent` harness: DSH's spill store died on exactly that
  479     `mkdtemp` and took its whole plugin tree with it, reporting `dsh: plugin tree failed to
  480     load: ... failed to apply loader entry spill-local` -- a message naming neither `TMPDIR`
  481     nor this module, at the end of a four-layer hunt. Returning the list instead of doing
  482     the `mkdir` here is deliberate: `mkdir_in` is a `docker exec`, so the container has to
  483     be **running**, and the caller that owns the start/stop order is the only one that knows
  484     when that is true.
  485     """
  486     env = container_env(task_mount)
  487     seen: list[str] = []
  488     for var in SCRATCH_ENV_VARS:
  489         value = env.get(var)
  490         if value and value not in seen:
  491             seen.append(value)
  492     return tuple(seen)

  495 def _mounts(task_dir: Path, task_mount: str, harness_root: Path | None,
  496             state_dirs: tuple[Path, ...], task_dir_writable: bool = True,
  497             readonly_paths: tuple[Path, ...] = ()) -> list[str]:
  498     """The mount plan. Parent before child, so a reader can follow it.
  499 
  500     **The order is not what makes the `.state/` hole work here, and this docstring
  501     used to claim it was.** `eval/containers/sandbox.py` needs its holes bound after the
  502     read-only parent, because bwrap's binds are last-one-wins and the reverse order
  503     silently produced a `Read-only file system` on the harness's own scratch space.
  504     Docker is not bwrap: it *sorts* nested mounts by destination depth, so the same
  505     plan in the opposite CLI order gives the same result. Measured, both orders:
  506     `state_writable=yes`, with the `readonly` parent bind given last.
  507 
  508     So the property comes from Docker's nesting-aware sort, not from this list, and
  509     the ordering below is kept only because a deterministic plan is easier to read and
  510     to test than an arbitrary one. It is stated this way rather than copied over from
  511     the other module because "it worked there for this reason" is exactly how a
  512     correct-looking plan acquires a false explanation.
  513 
  514     The task directory is mounted first of all: it is the one tree that must always be
  515     writable, because it is what the harness is being asked to change.
  516     """
  517     plan = ["--mount",
  518             f"type=bind,src={task_dir},dst={task_mount}"
  519             + ("" if task_dir_writable else ",readonly")]
  520     if harness_root is not None:
  521         plan += ["--mount", f"type=bind,src={harness_root},dst={harness_root},readonly"]
  522     for state in state_dirs:
  523         plan += ["--mount", f"type=bind,src={state},dst={state}"]
  524     # Trees the harness needs to run at all: a CLI it drives, the interpreter it was
  525     # written against. Read-only, and **also for `exec`** -- an earlier version applied
  526     # them only to `files` on the theory that "inside a container the runtime is the
  527     # image's", which is false for a task image that ships no runtime.
  528     for extra in readonly_paths:
  529         plan += ["--mount", f"type=bind,src={extra},dst={extra},readonly"]
  530     return plan
  531

  533 def command(argv: list[str], *, image: str, task_dir: Path, task_mount: str,
  534             harness_root: Path | None = None, state_dirs: tuple[Path, ...] = (),
  535             cwd: str | None = None, env: dict[str, str] | None = None,
  536             network: str = "none", read_only: bool = True,
  537             task_dir_writable: bool = True,
  538             extra_mounts: tuple[str, ...] = (),
  539             readonly_paths: tuple[Path, ...] = (),
  540             limits: dict | None = None) -> list[str]:
  541     """The `docker run` argv. Split out so `self_check` and the tests can read it."""
  542     limits = limits or {}
  543     cmd = ["docker", "run", "--rm", "--network", network]
  544 
  545     if read_only:
  546         # The environment is the image (§2.5.3). A rootfs the harness can write to
  547         # would make the check's world depend on what the harness did to it, and two
  548         # runs of one task would stop being comparable. `/tmp` is the exception
  549         # because enough software hardcodes it that refusing would only produce
  550         # confusing failures; it is a tmpfs, so it dies with the container.
  551         cmd += ["--read-only", "--tmpfs", "/tmp:rw,exec,nosuid,size=512m,mode=1777"]
  552 
  553     cmd += ["--pids-limit", str(limits.get("pids", 1024))]
  554     cmd += ["--memory", str(limits.get("memory", "4g"))]
  555     cmd += ["--cpus", str(limits.get("cpus", "2"))]
  556 
  557     cmd += _mounts(task_dir, task_mount, harness_root, tuple(state_dirs),
  558                    task_dir_writable, tuple(readonly_paths))
  559     # Mounts this module does not know the meaning of: a service's recording, whose
  560     # audience (`eval/service/services.py`) is the design decision and not this layer's to
  561     # make. Appended after the task mount so a recording can never shadow it.
  562     cmd += list(extra_mounts)
  563 
  564     # Run as the invoking user rather than root, for two reasons that are both
  565     # correctness rather than hygiene: files the harness creates would otherwise land
  566     # on the host owned by root, which the platform then cannot delete (its cleanup is
  567     # `shutil.rmtree` as this user), and root-in-container is a strictly larger
  568     # privilege set than the task needs.
  569     cmd += ["--user", f"{os.getuid()}:{os.getgid()}"]
  570     for key, value in sorted((env or {}).items()):
  571         cmd += ["-e", f"{key}={value}"]
  572     cmd += ["-w", cwd or task_mount]
  573     cmd += [image, *argv]
  574     return cmd
  575

  577 def run(argv: list[str], *, image: str, task_dir: Path, task_mount: str,
  578         harness_root: Path | None = None, state_dirs: tuple[Path, ...] = (),
  579         cwd: str | None = None, env: dict[str, str] | None = None,
  580         timeout_s: int = 3600, network: str = "none",
  581         read_only: bool = True, task_dir_writable: bool = True,
  582         extra_mounts: tuple[str, ...] = (),
  583         readonly_paths: tuple[Path, ...] = (),
  584         ) -> subprocess.CompletedProcess:
  585     """`subprocess.run` shape, inside a **fresh** container. Same return type.
  586 
  587     Fresh per call, not one container reused for the harness and then the check. The
  588     property §2.5.6 asks for -- that nothing the harness started is still running when
  589     the check runs -- cannot be *guaranteed* by killing the process tree of a
  590     `docker exec`: anything the harness backgrounded is reparented to the container's
  591     PID 1 and survives the exec. Starting the check in a new container from the same
  592     image makes the guarantee structural instead of best-effort, and it costs one
  593     container start.
  594 
  595     What survives between the two is the task directory, because it is on a bind mount
  596     -- which is exactly the set of things a check is supposed to grade.
  597     """
  598     # The safety net, not the mechanism. The driver resolves every dataset image once
  599     # at startup (§2.5.4: no digest, no start); this refuses anyway, because the cost
  600     # of being wrong is a silent pull and the cost of checking is a string compare.
  601     if not looks_pinned(image):
  602         raise ContainerUnavailable(
  603             f"refusing to run {image!r}: it is not pinned to a content address, and "
  604             f"`docker run` on a tag would pull whatever the registry served today. "
  605             f"Resolve it with `resolve_digest` first (INTERFACE.md §2.5.4).")
  606     cmd = command(argv, image=image, task_dir=task_dir, task_mount=task_mount,
  607                   harness_root=harness_root, state_dirs=state_dirs, cwd=cwd,
  608                   env=env, network=network, read_only=read_only,
  609                   task_dir_writable=task_dir_writable, extra_mounts=extra_mounts,
  610                   readonly_paths=readonly_paths)
  611     return subprocess.run(cmd, capture_output=True, text=True, timeout=timeout_s)
  612

  614 def to_container(path: Path | str, task_dir: Path, task_mount: str) -> str:
  615     """Translate a host path under `task_dir` to its container path.
  616 
  617     Only the task directory is translated. The harness tree is mounted at its own host
  618     path precisely so that the entrypoint argv needs no rewriting, and a translation
  619     is the one place a harness's own record can stop matching the filesystem it ran on.
  620     """
  621     path = Path(path)
  622     try:
  623         rel = path.relative_to(task_dir)
  624     except ValueError:
  625         return str(path)
  626     return task_mount if str(rel) == "." else f"{task_mount.rstrip('/')}/{rel}"

  629 # --------------------------------------------------------------- self check ---
  630 
  631 #: The probe the self-check runs inside a throwaway container. `sh` rather than
  632 #: Python: the image is the task's, and a task image need not have a Python we can
  633 #: rely on. Every line is `key=value` so the result can be read without a parser.
  634 #:
  635 #: `rootfs_writable` writes to `/var/tmp` and not `/usr`, and that is not arbitrary:
  636 #: the container runs as a non-root uid, which cannot write `/usr` in *any* image, so
  637 #: a `/usr` probe reports "read-only" whether or not `--read-only` was passed -- a
  638 #: check that cannot fail. `/var/tmp` is 1777 in a Debian-family image, so it is
  639 #: writable by that uid exactly when the rootfs is not read-only. Measured both ways.
  640 _PROBE = r"""
  641 p() { if [ -e "$1" ]; then echo "$2=yes"; else echo "$2=no"; fi; }
  642 p __PLATFORM__ platform_visible
  643 p __SIBLING__ sibling_visible
  644 p /var/run/docker.sock docker_socket
  645 p __HARNESS__/agent.py harness_readable
  646 if ( echo x > __HARNESS__/probe 2>/dev/null ); then echo harness_writable=yes; else echo harness_writable=no; fi
  647 if ( echo x > __HARNESS__/.state/probe 2>/dev/null ); then echo state_writable=yes; else echo state_writable=no; fi
  648 if ( echo x > __TASK__/written 2>/dev/null ); then echo task_writable=yes; else echo task_writable=no; fi
  649 if ( echo x > /var/tmp/probe 2>/dev/null ); then echo rootfs_writable=yes; else echo rootfs_writable=no; fi
  650 if ( echo x > /tmp/probe 2>/dev/null ); then echo tmp_writable=yes; else echo tmp_writable=no; fi
  651 echo uid=$(id -u)
  652 """
  653

  655 def self_check(platform: Path, image: str) -> dict:
  656     """Prove the container boundary holds before a paid run depends on it.
  657 
  658     Returns {"ok": bool, "detail": str}. Called at startup for a dataset that needs
  659     `exec`: a boundary that cannot be built must stop the run at the door, not once
  660     per task, and never by falling back to running on the host.
  661 
  662     Two of the probes exist because the obvious version of this check passes for the
  663     wrong reason:
  664 
  665     * `platform_visible` probes the **real platform path**, not a marker under `/tmp`.
  666       The container gets a tmpfs on `/tmp`, so a marker placed there would be shadowed
  667       and reported invisible whether or not anything was mounted -- a check that
  668       cannot fail. The real path is also the fact we actually care about.
  669     * `sibling_visible` probes a file next to the harness mount. Without it, "the
  670       platform is invisible" is satisfied by a plan that mounts the harness's whole
  671       parent directory, which would be a far larger hole than intended.
  672     """
  673     try:
  674         pinned = resolve_digest(image)
  675     except ContainerUnavailable as exc:
  676         return {"ok": False, "detail": str(exc)}
  677 
  678     # `/var/tmp`, not `/tmp`: everything under `/tmp` is behind the container's tmpfs.
  679     with tempfile.TemporaryDirectory(prefix="hg-container-check-", dir="/var/tmp") as tmp:
  680         tmp_path = Path(tmp)
  681         task_dir = tmp_path / "task"
  682         task_dir.mkdir()
  683 
  684         harness = tmp_path / "harness"
  685         state = harness / STATE_DIRNAME
  686         state.mkdir(parents=True)
  687         (harness / "agent.py").write_text("# a harness\n")
  688         # A file the harness mount must NOT expose: its sibling.
  689         sibling = tmp_path / "sibling.txt"
  690         sibling.write_text("not the harness\n")
  691 
  692         probe = (_PROBE
  693                  .replace("__PLATFORM__", str(platform / "driver.py"))
  694                  .replace("__SIBLING__", str(sibling))
  695                  .replace("__HARNESS__", str(harness))
  696                  .replace("__TASK__", str(task_dir)))
  697         # The task is mounted at its own host path here, unlike a real run where it is
  698         # mounted at the dataset's `workdir`: the probe asserts host paths that the
  699         # module did not choose.
  700         try:
  701             proc = run(["sh", "-c", probe], image=pinned, task_dir=task_dir,
  702                        task_mount=str(task_dir), harness_root=harness,
  703                        state_dirs=(state,), cwd=str(task_dir), timeout_s=180)
  704         except (subprocess.TimeoutExpired, OSError) as exc:
  705             return {"ok": False, "detail": f"the probe container did not run: {exc}"}
  706 
  707         if proc.returncode != 0:
  708             return {"ok": False,
  709                     "detail": (f"the probe container exited {proc.returncode}: "
  710                                f"{(proc.stderr or proc.stdout).strip()[:400]}")}
  711 
  712     facts = {}
  713     for line in proc.stdout.splitlines():
  714         if "=" in line:
  715             key, _, value = line.partition("=")
  716             facts[key.strip()] = value.strip()
  717 
  718     problems = []
  719     if facts.get("platform_visible") != "no":
  720         problems.append("the platform is visible inside the container")
  721     if facts.get("sibling_visible") != "no":
  722         problems.append("a file beside the harness mount is visible, so the plan "
  723                         "mounts more than the harness")
  724     if facts.get("docker_socket") != "no":
  725         problems.append("the docker socket is visible inside the container, so the "
  726                         "harness can start a sibling container with any mount it likes")
  727     if facts.get("harness_readable") != "yes":
  728         problems.append("the harness tree is not mounted, so the harness cannot run")
  729     if facts.get("harness_writable") != "no":
  730         problems.append("the harness tree is writable, so tamper detection is void")
  731     if facts.get("state_writable") != "yes":
  732         problems.append("the harness's .state/ is not writable: the read-only bind of "
  733                         "its parent won over the writable bind of the hole")
  734     if facts.get("task_writable") != "yes":
  735         problems.append("the task directory is not writable")
  736     if facts.get("rootfs_writable") != "no":
  737         problems.append("the container rootfs is writable, so the environment is not "
  738                         "the image and two runs of one task need not agree")
  739     if facts.get("tmp_writable") != "yes":
  740         problems.append("/tmp is not writable, which several agent CLIs require")
  741 
  742     return {"ok": not problems,
  743             "detail": ("; ".join(problems) if problems
  744                        else f"image {digest_of(pinned)}: " + ", ".join(
  745                            f"{k}={v}" for k, v in sorted(facts.items()))),
  746             "facts": facts, "image_digest": digest_of(pinned)}
  747

  751 # INTERFACE.md §2.5.9. Some tasks put their state in the container's own filesystem
  752 # rather than in a bind mount -- measured on the real Terminal-Bench images: `WORKDIR

  779 def _require_static_elf(path: Path) -> None:
  780     """Reject loaders and dynamic dependencies in the platform's idle executable.
  781 
  782     An absolute image path alone is unsafe: its executable, ELF loader and preloads
  783     can all belong to the actor. Validate the copied host binary, not image tools.
  784     """
  785     raw = path.read_bytes()
  786     try:
  787         if raw[:4] != b"\x7fELF" or raw[4] not in (1, 2) or raw[5] not in (1, 2):
  788             raise ValueError("not ELF")
  789         endian = "<" if raw[5] == 1 else ">"
  790         layout = "16sHHIIIIIHHHHHH" if raw[4] == 1 else "16sHHIQQQIHHHHHH"
  791         header = struct.unpack_from(endian + layout, raw)
  792         offset, stride, count = header[5], header[9], header[10]
  793         if (header[1] != 2 or not 0 < count <= 4096
  794                 or stride < (32 if raw[4] == 1 else 56)
  795                 or offset + stride * count > len(raw)):
  796             raise ValueError("invalid executable headers")
  797         for i in range(count):
  798             kind, = struct.unpack_from(endian + "I", raw, offset + i * stride)
  799             if kind in (2, 3):  # PT_DYNAMIC / PT_INTERP
  800                 raise ValueError("dynamic executable")
  801     except (IndexError, struct.error, ValueError) as exc:
  802         raise SnapshotUnavailable("snapshot startup requires a static ELF BusyBox") from exc

  805 @contextmanager
  806 def snapshot_startup():
  807     """A trusted idle PID 1; no actor entrypoint, PATH, loader or healthcheck.
  808 
  809     Requires a static BusyBox on the trusted host; absence fails closed. The private
  810     copy remains mounted read-only until verifier removal. Scorer execution itself
  811     still uses the snapshot runtime and is outside this startup guarantee.
  812     """
  813     source = shutil.which("busybox")
  814     if not source:
  815         raise SnapshotUnavailable("snapshot startup requires host static BusyBox")
  816     with tempfile.TemporaryDirectory(prefix="hg-snapshot-idle-") as tmp:
  817         binary = Path(tmp) / "busybox"
  818         try:
  819             shutil.copyfile(source, binary)
  820             _require_static_elf(binary)
  821             binary.chmod(0o755)
  822         except OSError as exc:
  823             raise SnapshotUnavailable(f"snapshot startup unavailable: {exc}") from exc
  824         target = "/" + Path(tmp).name
  825         yield target + "/busybox", ("--mount", f"type=bind,src={tmp},dst={target},readonly")

  828 def create(name: str, *, image: str, limits: dict | None = None,
  829            network: str = "none", extra_mounts: tuple[str, ...] = (),
  830            tmpfs: tuple[str, ...] = (), run_id: str = "", task_id: str = "",
  831            startup_entrypoint: str | None = None) -> str:
  832     """Create a stopped container that stays behind, so it can be committed.
  833 
  834     `sleep infinity` as the command because the image is the task's, and a task image
  835     need not have a long-running entrypoint -- measured: the Terminal-Bench images have
  836     no `CMD` at all, only `FROM`, `WORKDIR` and a `COPY`. The platform brings its own
  837     process to keep the container alive while it copies the task in.
  838 
  839     **Root, unlike everything else here.** These images and their checks need it: the
  840     real Terminal-Bench `test.sh` opens with `apt-get update && apt-get install -y curl`.
  841     The two host-ownership problems that pushed every other container to the invoking uid
  842     are avoided instead of ignored -- the harness tree is mounted read-only, and its
  843     scratch space is a tmpfs that never reaches the host.
  844     """
  845     limits = limits or {}
  846     # **Labelled, because a name is not findable.** A hard kill (SIGKILL, or the harness
  847     # process tree being reaped) skips every `finally`, so these containers survive it --
  848     # and `services.reap` searches by `label=harnessgrad.run=<id>`, which is how the
  849     # *service* containers and the task networks were already findable. The harness
  850     # container had only a name, so the one leak that actually happened was the one a
  851     # reaper could not see. Measured (2026-10-04): `screen-all-1` was killed at 22:42
  852     # with five of these still `running`, and `reap` reported only "network ... has
  853     # active endpoints" because it could not name the containers holding them.
  854     labels = ["--label", f"harnessgrad.run={run_id}"]
  855     if task_id:
  856         labels += ["--label", f"harnessgrad.task={task_id}"]
  857     cmd = ["create", "--name", name, "--network", network, *labels,
  858            "--pids-limit", str(limits.get("pids", 1024)),
  859            "--memory", str(limits.get("memory", "4g")),
  860            "--cpus", str(limits.get("cpus", "2"))]
  861     for path in tmpfs:
  862         cmd += ["--tmpfs", path]
  863     if startup_entrypoint is not None:
  864         cmd += ["--entrypoint", startup_entrypoint, "--no-healthcheck", "--workdir", "/"]
  865     cmd += list(extra_mounts)
  866     # Numeric duration works with static BusyBox versions that reject "infinity".
  867     # The verifier's own timeout/teardown bounds its lifetime, not this idle timer.
  868     cmd += [image, "sleep", "2147483647" if startup_entrypoint else "infinity"]
  869     proc = docker(*cmd, timeout=180)
  870     if proc.returncode != 0:
  871         raise ContainerUnavailable(
  872             f"could not create a container for {image!r}: {proc.stderr.strip()[:300]}")
  873     return name

  876 def cp_in(name: str, source: Path, dest: str, *, contents: bool = False) -> None:
  877     """Copy a host file or directory into a container.
  878 
  879     `docker cp` rather than a bind mount because the point of container state is that the
  880     task lives *in* the container: what is copied in is part of the image the check will
  881     grade, and what is not copied in is not there at all.
  882 
  883     `contents=True` copies what is *inside* the source rather than the source directory
  884     itself, which is what copying a task's initial state into an existing `workdir`
  885     needs. It is a parameter rather than `source / "."` at the call site because
  886     `pathlib` **normalizes that dot away** — measured: `docker cp <setup> <ctr>:/app`
  887     put the task at `/app/setup/...` instead of `/app/...`, and the harness found
  888     nothing, which looked exactly like a harness that could not do the task.
  889     """
  890     src = f"{source}/." if contents else str(source)
  891     proc = docker("cp", src, f"{name}:{dest}", timeout=300)
  892     if proc.returncode != 0:
  893         raise ContainerUnavailable(
  894             f"could not copy {source} into {name}:{dest}: {proc.stderr.strip()[:300]}")

  897 def cp_out(name: str, source: str, dest: Path) -> bool:
  898     """Copy out of a container. Returns False when the source does not exist.

  935 def stop(name: str) -> None:
  936     """Stop the container, which is what guarantees nothing it started survives.
  937 
  938     A whole container stopping is a stronger statement than killing one process tree,
  939     and it is free here: `docker commit` works on a stopped container, so the snapshot

  950 def commit(name: str, tag: str) -> str:
  951     """Snapshot a container into an image, and return its id.
  952 
  953     An **image** rather than a tarball because an image is immutable: the check runs on a
  954     writable copy of it, so a check that writes into what it is grading cannot alter the
  955     thing it graded (INTERFACE.md §2.5.9).
  956     """
  957     snapshot_admission(name)
  958     try:
  959         proc = docker("commit", name, tag, timeout=SNAPSHOT_COMMIT_TIMEOUT_S)
  960     except (subprocess.TimeoutExpired, OSError) as exc:
  961         raise _snapshot_failure("commit", exc=exc) from exc
  962     if proc.returncode != 0:
  963         raise _snapshot_failure("commit", proc)
  964     image_id = proc.stdout.strip()
  965     if not looks_pinned(image_id):
  966         raise SnapshotUnavailable(
  967             f"snapshot commit failed kind=docker: `docker commit` reported {image_id!r}, "
  968             "not a digest")
  969     return image_id

  972 def remove_container(name: str) -> None:
  973     """Never raises: it runs in a `finally`."""
  974     try:
  975         docker("rm", "--force", name, timeout=120)
  976     except (subprocess.TimeoutExpired, OSError):
  977         pass

  980 def remove_image(image: str) -> None:
  981     """Never raises. A leaked snapshot is a disk leak, and there is one per task."""
  982     try:
  983         docker("image", "rm", "--force", image, timeout=300)
  984     except (subprocess.TimeoutExpired, OSError):
  985         pass

  988 def network_connect(name: str, network: str) -> None:
  989     """Attach a container to a second network.
  990 
  991     Needed because a task may want both things at once and docker gives one `--network`
  992     at create time: the **internal** network, on whose gateway the platform publishes the
  993     agent's model, and the default bridge, which is the internet the task declared. Every
  994     real Terminal-Bench task declares `allow_internet = true`, so this is the normal case
  995     for them rather than an exotic one.
  996     """
  997     proc = docker("network", "connect", network, name, timeout=120)
  998     if proc.returncode != 0:
  999         raise ContainerUnavailable(
 1000             f"could not attach {name} to {network!r}: {proc.stderr.strip()[:300]}")
