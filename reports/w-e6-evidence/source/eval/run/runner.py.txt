   39 def _refuse_unobserved_policies(task: dict, env: dict | None,
   40                                verifier: dict | None, *, run_id: str = "",
   41                                diagnostic_authorization: Authorization | None = None) -> dict | None:
   42     """Refuse policy-bearing tasks until a trusted backend observer is wired in.
   43 
   44     The offline runner_admission checks cannot discover enforcement. Launch flags,
   45     adapter observations and cached verdicts are not a substitute for that observer.
   46     An operator may authorize one explicitly unmeasured diagnostic attempt. That
   47     permission is never treated as backend observation or production admission.
   48     """
   49     env_spec = env or {}
   50     fields = {"env." + key for key in (
   51         "cpus", "memory_mb", "agent_timeout_s", "verify_timeout_s", "network",
   52         "resources", "filesystems", "resource_policy", "network_policy", "artifact_policy",
   53     ) if key in env_spec}
   54     # Even direct callers that omit normalized fields request these default bounds
   55     # by choosing exec. The registry normally fills them before reaching the runner.
   56     if env_spec.get("kind") == "exec":
   57         fields.add("env.exec_defaults")
   58     if env_spec.get("state") == "container":
   59         fields.add("env.state=container")
   60     # `verifier.timeout_s` is deliberately NOT a trigger. The platform enforces it
   61     # itself -- `eval/verification/verifier.py` passes it to `phase_timeout` and to the verifier
   62     # subprocess' `timeout=`, and records expiry as its own outcome ("the verifier
   63     # timed out after Ns"), which is an unmeasured result rather than a score. There
   64     # is no external backend capability to observe, so demanding an observation for it
   65     # refused every dataset that declares one (the shared QA scaffold sets 3600s for
   66     # all of its adapters) while adding no guarantee. Everything whose enforcement
   67     # really does live outside this process -- exec defaults, container state, network,
   68     # cpus/memory, filesystems, resource and artifact policy -- still refuses.
   69     if "artifact_policy" in (verifier or {}):
   70         fields.add("verifier.artifact_policy")
   71     if not fields:
   72         return None
   73     if diagnostic_authorization is not None:
   74         # Still refuse every ordinary run, pilot, unbound task/run and cached
   75         # measurement. Only the bound one-task diagnostic may attempt execution;
   76         # its caller records non-observation and forbids comparable publication.
   77         diagnostic_authorization.require_scope(run_id, [task["task_id"]])
   78         return None
   79     result = _invalid(task, "backend_observation_unavailable", stage="admission", service=None)
   80     result["invalid"]["receipt"] = {
   81         "schema": "harnessgrad.runner-policy-refusal.v1", "stage": "admission",
   82         "status": "blocked", "reason": "backend_observation_unavailable",
   83         "measured": False, "production_authorized": False, "observation": "unknown",
   84         "policy_fields": sorted(fields),
   85     }
   86     return result

   89 def harness_env() -> dict:
   90     """The environment a harness runs in: the agent's model, and not the method's.
   91 
   92     The two are configured separately on purpose (`.env.example`), and separate
   93     configuration is not separation until something enforces it. Without this the
   94     harness inherits `HG_METHOD_API_KEY` and can spend the method's quota, or be
   95     pointed at the method's endpoint -- so a harness that misbehaves under
   96     measurement would look like a harness problem when it is a plumbing one.
   97 
   98     The reverse direction is deliberately left open: a method is allowed to see how
   99     the harness will be run, because it is being asked to reason about the harness.
  100     One-way isolation is the useful half, and it is the half that keeps the two
  101     budgets from being confused for each other.
  102 
  103     `HG_ENV_*` is stripped here for the same reason, and it closed a real leak: this
  104     function forwards everything that is not the *method's*, so the moment a third
  105     budget existed for the environment, a `files` task's harness inherited the
  106     environment's credential. Services only exist on `exec` -- where
  107     `container_env`'s allowlist already excluded it -- so the leak would have been on
  108     the path with no services at all, which is exactly where nobody looks for one.
  109 
  110     **A fourth name, and the reason the rule is no longer only a prefix list.** The
  111     improver registry declares each improver's credential variable (`api_key_env`), and
  112     the `deepseek` entry names `DEEPSEEK_API_KEY` -- which matches none of the prefixes
  113     above, so the harness inherited the *improver's* key. Measured: it took making
  114     `deepseek` the default improver (so the name had to exist in `.env`) for the
  115     isolation probe to see it. Both earlier leaks were a new budget with a new prefix;
  116     this one was the same budget under a name the prefix rule could not know. So the
  117     variables the registry declares are stripped by name, and a new improver is covered
  118     the moment it is added rather than when someone remembers the prefix.
  119     """
  120     env = {k: v for k, v in os.environ.items()
  121            if not k.startswith(("HG_METHOD_", "HG_ENV_", "HARNESSGRAD_SEED"))
  122            and k not in _improver_credentials()}
  123     return env
  124

  126 def _improver_credentials() -> set[str]:
  127     """Every environment variable the improver registry declares as a credential.
  128 
  129     Read lazily and defensively: the harness environment must not fail to build because a
  130     config file is missing, and `harness_env()` is called once per task on paths that have
  131     nothing to do with improvers. The registry is read through `tools/improver.py`'s own
  132     loader so there is one reader of that file, not two.
  133     """
  134     try:
  135         from tools.improver import load_config
  136         config = load_config() or {}
  137     except Exception:                                          # noqa: BLE001
  138         return set()
  139     names = {(entry or {}).get("api_key_env")
  140              for entry in (config.get("improvers") or {}).values()}
  141     return {name for name in names if name}

  151 def run_one(repo: Path, task: dict, timeout_s: int | None = None,
  152             sandbox: bool = True, setup: dict | None = None,
  153             verifier: dict | None = None,
  154             verify_timeout_s: int | None = None,
  155             env: dict | None = None,
  156             run_id: str = "", run_seed: str = "",
  157             recordings_root: Path | None = None,
  158             diagnostic_authorization: Authorization | None = None,
  159             trial_id: str | None = None) -> dict:
  160     if diagnostic_authorization is not None:
  161         diagnostic_authorization.require_scope(run_id, [task["task_id"]])
  162     refusal = _refuse_unobserved_policies(
  163         task, env, verifier, run_id=run_id,
  164         diagnostic_authorization=diagnostic_authorization)
  165     result = refusal if refusal is not None else _run_one(
  166         repo, task, timeout_s, sandbox, setup, verifier, verify_timeout_s,
  167         env, run_id, run_seed, recordings_root, trial_id=trial_id)
  168     if result.get("verdict") is not None:
  169         result["verdict"].setdefault("scorer_isolation_class", "unknown")
  170     if diagnostic_authorization is not None:
  171         result["enforcement"] = diagnostic_authorization.record()
  172     return result

  175 def _run_one(repo: Path, task: dict, timeout_s: int | None = None,
  176             sandbox: bool = True, setup: dict | None = None,
  177             verifier: dict | None = None,
  178             verify_timeout_s: int | None = None,
  179             env: dict | None = None,
  180             run_id: str = "", run_seed: str = "",
  181             recordings_root: Path | None = None, *, trial_id: str | None = None) -> dict:
  182     """Run the harness on one task in a throwaway dir, off the platform's disk.
  183 
  184     Returns {task_id, answer, trace, exit_code}. A non-zero exit is recorded as
  185     a harness failure rather than raised: one broken task must not lose a whole
  186     paid round.
  187 
  188     `sandbox=True` runs the subject inside its own mount namespace, so the platform
  189     is not on its disk at all. Paths are bound at their host locations, so the
  190     repository path, `--workdir`, and anything the harness writes into its trace
  191     keep the same spelling they would have had unsandboxed -- a translated path is
  192     a place where a harness's own record stops matching the filesystem it ran on.
  193 
  194     Native tasks can declare network='none'; live model backends are refused for
  195     those tasks before launch. See `eval/containers/sandbox.py`.
  196     """
  197     # Resolve before the subprocess switches cwd: a relative repo path would
  198     # otherwise be interpreted against the sandbox directory and the harness
  199     # would silently fail to launch, scoring zero for the wrong reason.
  200     repo = Path(repo).resolve()
  201     # The environment is a property of the task (INTERFACE.md §2.5). `files` is the
  202     # absence of a declaration, not a separate code path invented for old datasets.
  203     env_spec = env or {"kind": "files"}
  204     is_exec = env_spec.get("kind") == "exec"
  205     try:
  206         protocol = chess_stepwise.protocol_of(task)
  207         stepwise_session = chess_stepwise.session_for(
  208             task, verifier, sandbox=sandbox, env=env_spec)
  209     except ValueError as exc:
  210         return _invalid(task, exc, stage="stepwise admission", service=None)
  211     # Inside a container the task is mounted at the dataset's `workdir` (default
  212     # `/app`), so the harness is told the path it can actually see. Outside one the
  213     # task *is* the host temporary directory. This is the only path translation in
  214     # the runner, and it exists because the dataset names a location in the
  215     # environment rather than on the host.
  216     task_mount = env_spec.get("workdir") or container_mod.DEFAULT_WORKDIR
  217     # Declared per environment: the real task set spans 600-12000 s, and the module
  218     # default of 300 would report a harness failing a task it was never given time to
  219     # attempt (INTERFACE.md §2.5.9).
  220     timeout_s = phase_timeout(timeout_s, env_spec.get("agent_timeout_s"))
  221     verify_timeout_s = phase_timeout(verify_timeout_s, env_spec.get("verify_timeout_s"),
  222                                     (verifier or {}).get("timeout_s"))
  223     agent_runtime = env_spec.get("agent_runtime")
  224     if agent_runtime and is_exec:
  225         raise ValueError("agent_runtime package projections require a files environment")
  226     if not is_exec and not sandbox and (agent_runtime or env_spec.get("network") == "none"):
  227         raise ValueError("declared native runtime/network isolation requires sandbox=True")
  228     # Read before the dispatch: both state kinds need it, and it used to be computed
  229     # further down in the `mount` path only.
  230     _manifest = json.loads((repo / "harness.json").read_text())
  231     if not is_exec:
  232         admit_native_network(env_spec, _manifest, os.environ.get("HG_AGENT_BACKEND", "mock"))
  233     runtime_paths = tuple(Path(p).expanduser()
  234                           for p in (_manifest.get("runtime_paths") or []))
  235     if is_exec and env_spec.get("state") == "container":
  236         return _run_one_container_state(
  237             repo, task, env_spec=env_spec, setup=setup, verifier=verifier,
  238             timeout_s=timeout_s, verify_timeout_s=verify_timeout_s,
  239             run_id=run_id, run_seed=run_seed, recordings_root=recordings_root,
  240             runtime_paths=runtime_paths)
  241     service_specs = list(env_spec.get("services") or [])
  242     web_spec = env_spec.get("web") or {}
  243     running = None
  244     # The internal network, whoever created it: `services.start` when there are
  245     # services, the model gateway when there are not. Tracked separately from `running`
  246     # so teardown knows which of the two it owns.
  247     network_name: str | None = None
  248     owns_network = False
  249     model_gateway = None
  250     web_gateway = None
  251     held = progress_mod.task_of()
  252     phases = progress_mod.phases_for(held)
  253     runtime_dir = None
  254     stepwise_relay = None
  255     stepwise_dir = None
  256     package_overlays = ()
  257     workdir = Path(tempfile.mkdtemp(prefix="hg-task-"))
  258     try:
  259         if agent_runtime:
  260             runtime_dir = Path(tempfile.mkdtemp(prefix="hg-agent-runtime-"))
  261             package_overlays = (stage_agent_runtime(agent_runtime, runtime_dir),)
  262         # The task's initial state, before the harness sees anything. It used to be an
  263         # empty directory, which is why the old contract could only express "answer a
  264         # question": a dataset had nowhere to put the thing the task is *about*.
  265         with phases("setup"):
  266             if setup:
  267                 _materialize(workdir, setup.get("files"), PLATFORM_ROOT)
  268 
  269             task_file = workdir / "task.json"
  270             actor_task = ({**task, "goal": chess_stepwise.GOAL}
  271                           if stepwise_session is not None else task)
  272             task_file.write_text(json.dumps(runtime_task(actor_task, setup)))
  273 
  274         # Services come up before the harness and are health-checked before it starts
  275         # (INTERFACE.md §2.5.7, steps 1-3). A harness that starts against a service
  276         # that is still initialising does not produce a bad measurement, it produces a
  277         # measurement of a race -- so the failure is refused here rather than scored.
  278         if service_specs:
  279             try:
  280                 running = services_mod.start(
  281                     service_specs,
  282                     # A caller that names no root gets a throwaway one beside the task
  283                     # directory: an in-process caller (a test) wants the lifecycle
  284                     # exercised, not a directory in somebody's run tree.
  285                     recordings_root=(recordings_root
  286                                      or workdir.parent / f".recordings-{run_id or 'run'}"),
  287                     run_id=run_id or "run", task_id=task["task_id"],
  288                     run_seed=run_seed or "0")
  289                 network_name = running.network
  290             except services_mod.ServiceUnavailable as exc:
  291                 return _invalid(task, exc, stage=exc.stage, service=exc.service)
  292 
  293         if web_spec:
  294             try:
  295                 if network_name is None:
  296                     network_name = services_mod.create_network(run_id or "run", task["task_id"])
  297                     owns_network = True
  298                 record_base = recordings_root or workdir.parent
  299                 web_gateway = web_gateway_mod.WebGateway(
  300                     bind_host=services_mod.gateway_ip(network_name),
  301                     record_path=record_base / task["task_id"] / "web" / "requests.jsonl",
  302                     enable_search=bool(web_spec.get("search")),
  303                     enable_fetch=bool(web_spec.get("fetch"))).start()
  304             except Exception as exc:
  305                 return _invalid(task, exc, stage="start", service="web")
  306 
  307         manifest = json.loads((repo / "harness.json").read_text())
  308         entry = manifest["entrypoint"]
  309         # Trees this harness needs in order to run at all — a CLI it drives, a
  310         # runtime it was built against. Declared, not inferred: the platform cannot
  311         # know where somebody's agent CLI is installed.
  312         runtime_paths = tuple(Path(p).expanduser()
  313                               for p in (manifest.get("runtime_paths") or []))
  314         # Inside a container the entrypoint is still `<repo>/<entry>` because the
  315         # harness tree is mounted at its own host path; only the task's paths move,
  316         # because the dataset named a location *in the environment*.
  317         #
  318         # The interpreter is **the platform's own** for `exec`, mounted in read-only,
  319         # rather than whatever `python` the image happens to have -- which may be
  320         # nothing (measured: the real Terminal-Bench image has no `python3` at all).
  321         # `files` keeps the bare name, because there the host's PATH is the right answer
  322         # and changing it would change what every existing dataset measures.
  323         task_arg = (container_mod.to_container(task_file, workdir, task_mount)
  324                     if is_exec else str(task_file))
  325         workdir_arg = task_mount if is_exec else str(workdir)
  326         # The image's interpreter when it has one (§2.5.3), the platform's when it does
  327         # not. `files` keeps the bare name: there the host's PATH is the right answer and
  328         # changing it would change what every existing dataset measures.
  329         image_python = container_mod.image_python(env_spec["image"]) if is_exec else None
  330         python = image_python or (sys.executable if is_exec else "python")
  331         argv = [python, str(repo / entry), "--task", task_arg,
  332                 "--workdir", workdir_arg]
  333 
  334         # The harness's own dependencies, from a directory outside the task's
  335         # environment, resolved for **this** interpreter's Python version. `None` is the
  336         # ordinary case: nobody configured one, and the run then behaves exactly as it
  337         # did before this existed. See `eval/containers/harness_runtime.py` for why this is not a
  338         # derived image and not the platform's interpreter.
  339         overlay = harness_runtime.lookup(
  340             repo, python, image=(env_spec["image"] if is_exec else None),
  341             image_digest=(env_spec.get("image_digest") if is_exec else None))
  342         if overlay:
  343             # Only the paths are collected here; the environment is applied below, where
  344             # `child_env` exists. It used to be written here, one screen above its own
  345             # definition -- a `NameError` on the one branch nobody had configured yet,
  346             # which is how it survived: `overlay` is `None` unless a harness runtime is
  347             # declared, and then the run that declares one is the run that crashes.
  348             runtime_paths = tuple(runtime_paths) + harness_runtime.extra_paths(overlay)
  349 
  350         # The harness's own writable scratch space, inside its otherwise read-only
  351         # tree. Exported so a harness need not know the platform's layout, and
  352         # stated in the environment rather than passed as an argument because the
  353         # entrypoint contract (INTERFACE.md §1.2) is `--task` and `--workdir` only.
  354         state_dir = repo / STATE_DIRNAME
  355         # Created on demand, not just at staging time: a workspace committed before
  356         # `.state/` existed would otherwise make every run refuse to start, and the
  357         # sandbox needs a real directory to mount over. Empty and gitignored, so it
  358         # cannot move a sha.
  359         state_dir.mkdir(exist_ok=True)
  360 
  361         # A writable HOME, and the XDG directories, per task.
  362         #
  363         # This is what makes an agent CLI usable as a harness at all. The sandbox
  364         # starts from an empty root -- only /usr, /bin, /sbin, /lib*, /etc and the
  365         # trees it is told to bind exist inside -- so the inherited `HOME`, pointing at
  366         # a real `/home/<user>` that does not exist there, is a path a CLI cannot even
  367         # `mkdir` under. Every one of them (Claude Code, Codex, OpenCode, ...) writes
  368         # session state, config and caches under HOME, and dies immediately if it
  369         # cannot.
  370         #
  371         # Under `workdir` rather than somewhere of its own, because `workdir` is
  372         # already bound read-write and is already a per-task temporary directory: the
  373         # isolation this needs is the isolation it already has, and inventing a second
  374         # mount for it would be a second thing to get wrong. Named with the platform's
  375         # prefix so it cannot be mistaken for a task's own file.
  376         #
  377         # The same directory is created for a container, where it is reached as
  378         # `<workdir>/.harnessgrad-home` through the task bind mount -- so the two
  379         # environments agree on where a CLI keeps its state without agreeing on how
  380         # that path is spelled on the host.
  381         home = workdir / ".harnessgrad-home"
  382         (home / "tmp").mkdir(parents=True, exist_ok=True)
  383 
  384         if is_exec:
  385             # An allowlist, and container-side HOME/XDG/TMPDIR: forwarding the host
  386             # values would name directories that do not exist inside the image.
  387             child_env = container_mod.container_env(task_mount, deadline_s=timeout_s)
  388         else:
  389             child_env = harness_env()
  390             child_env.update({
  391                 "HOME": str(home),
  392                 "XDG_CONFIG_HOME": str(home / ".config"),
  393                 "XDG_DATA_HOME": str(home / ".local" / "share"),
  394                 "XDG_CACHE_HOME": str(home / ".cache"),
  395                 # A CLI's scratch files should not land in the directory the task is
  396                 # about, where a verifier would then be reading them.
  397                 "TMPDIR": str(home / "tmp"),
  398             })
  399         child_env["HG_AGENT_DEADLINE_S"] = str(timeout_s)
  400         child_env["HARNESSGRAD_STATE"] = str(state_dir)
  401         if overlay:
  402             child_env.update(harness_runtime.env_for(overlay))
  403 
  404         # Publish the agent's model onto the task's network. A containerised harness
  405         # cannot reach a model on the host's loopback -- measured, every obvious route
  406         # fails -- and the wrong fix is `--network bridge`, which would give the task the
  407         # internet and silently change what it is (`eval/service/modelgate.py`).
  408         if is_exec:
  409             base = child_env.get("HG_AGENT_BASE_URL")
  410             if base and os.environ.get("HG_AGENT_BACKEND", "mock") != "mock" \
  411                     and modelgate_mod.is_host_local(base):
  412                 if network_name is None:
  413                     network_name = services_mod.create_network(
  414                         run_id or "run", task["task_id"])
  415                     owns_network = True
  416                 target, path = modelgate_mod.split(base)
  417                 model_gateway = modelgate_mod.ModelGateway(
  418                     bind_host=services_mod.gateway_ip(network_name),
  419                     target=target, path=path).start()
  420                 # The harness is handed the published address, and nothing else about
  421                 # the mechanism: `HG_AGENT_*` is the contract it already reads.
  422                 child_env["HG_AGENT_BASE_URL"] = model_gateway.base_url
  423 
  424         # How the harness finds the services, and the whole of what it is told about
  425         # them. `HG_SVC_*` rather than `HG_ENV_*`: the environment's own model,
  426         # credential and seed go to the service containers only (§2.5.7).
  427         if running is not None:
  428             child_env.update(services_mod.harness_env(service_specs))
  429         if web_gateway is not None:
  430             child_env["HG_SVC_WEB_URL"] = web_gateway.url
  431 
  432         # On a `services` environment the harness joins the task's internal network,
  433         # which is how "reachable by name, not on the internet" is delivered. Passing
  434         # `env_spec["network"]` straight through would ask docker for a network called
  435         # `services`, which does not exist.
  436         # With a gateway published, the harness joins the internal network so it can
  437         # reach it. Otherwise the declared value stands.
  438         network = (network_name if network_name is not None
  439                    else env_spec.get("network", "none"))
  440 
  441         # Everything the harness needs to run at all, plus the platform's interpreter.
  442         # The platform's interpreter is mounted only when it is going to be used, so an
  443         # image that brings its own does not also get the host's Python on its disk.
  444         runtime = ((tuple(runtime_paths)
  445                     + (() if image_python else container_mod.interpreter_paths()))
  446                    if is_exec else ())
  447 
  448         if stepwise_session is not None:
  449             # Project only this read-only socket directory. Neither gold nor session
  450             # state is written beneath the actor's writable task directory.
  451             stepwise_dir = Path(tempfile.mkdtemp(prefix="hg-stepwise-"))
  452             stepwise_relay = StepwiseRelay(
  453                 stepwise_dir, stepwise_session, contract=chess_stepwise.STEPWISE,
  454                 public_fields=chess_stepwise.PUBLIC_FIELDS, timeout_s=timeout_s)
  455             child_env["HG_STEPWISE_SOCKET"] = str(stepwise_relay.path)
  456 
  457         def launch():
  458             if is_exec:
  459                 # `runtime_paths` is deliberately not applied: it exists because a
  460                 # host CLI lives outside `/usr`, and inside a container the runtime is
  461                 # the image's. Mounting a host tree in would defeat the point of
  462                 # having an image at all.
  463                 return container_mod.run(
  464                     argv, image=env_spec["image"], task_dir=workdir,
  465                     task_mount=task_mount, harness_root=repo,
  466                     state_dirs=(state_dir,), cwd=task_mount, env=child_env,
  467                     timeout_s=timeout_s, network=network,
  468                     # Stated rather than omitted: the harness gets **no** view of a
  469                     # service's recording, and the empty list is that decision.
  470                     readonly_paths=runtime,
  471                     extra_mounts=tuple(services_mod.mounts_for(
  472                         "harness", service_specs,
  473                         running.recordings if running else {})))
  474             if not sandbox:
  475                 return subprocess.run(argv, capture_output=True, text=True,
  476                                       timeout=timeout_s, cwd=workdir, env=child_env)
  477             # `repo` is `<work_root>/<run-id>/workspace`, so the work root -- the one
  478             # tree that must stay visible -- is two levels up. Read from the
  479             # platform's own environment rather than guessed from the run id.
  480             work_root = Path(os.environ.get("HARNESSGRAD_WORK_ROOT")
  481                              or repo.parent.parent)
  482             return sandbox_mod.run(argv, platform=PLATFORM_ROOT,
  483                                    work_root=work_root, task_dir=workdir,
  484                                    env=child_env, timeout_s=timeout_s, cwd=workdir,
  485                                    repo=repo, state_dirs=(state_dir,),
  486                                    runtime_paths=runtime_paths,
  487                                    visible_paths=((stepwise_dir,) if stepwise_dir else ()),
  488                                    network=env_spec.get("network"),
  489                                    package_overlays=package_overlays)
  490 
  491         try:
  492             with phases("agent"):
  493                 if stepwise_relay is not None:
  494                     stepwise_relay.start()
  495                 try:
  496                     proc = launch()
  497                 finally:
  498                     if stepwise_relay is not None:
  499                         stepwise_relay.stop()
  500             exit_code, stderr = proc.returncode, proc.stderr[-2000:]
  501         except subprocess.TimeoutExpired:
  502             exit_code, stderr = -1, f"timeout after {timeout_s}s"
  503         except OSError:
  504             if stepwise_relay is None:
  505                 raise
  506             exit_code, stderr = -1, "stepwise harness launch failed"
  507         except sandbox_mod.SandboxUnavailable as exc:
  508             # Not swallowed. A run that asked for a sandbox and silently got none
  509             # produces numbers that mean something other than what they say.
  510             raise SystemExit(f"refusing to run: sandbox requested but unavailable: "
  511                              f"{exc}. Pass --no-sandbox to accept that a harness "
  512                              f"can reach the platform.")
  513 
  514         with phases("collect"):
  515             answer_path = workdir / "answer.txt"
  516             answer = (answer_path.read_text()
  517                       if stepwise_relay is None and answer_path.exists() else "")
  518             trace_path = workdir / "trace.jsonl"
  519             trace = trace_path.read_text() if trace_path.exists() else ""
  520 
  521         # The harness has exited. Grade the controller's completed live exchange,
  522         # or its frozen artifacts on the existing offline path (§2.2 and §2.4).
  523         with phases("verify"):
  524             if stepwise_relay is not None:
  525                 verdict = chess_stepwise.finish(stepwise_relay, exit_code)
  526             else:
  527                 verdict = _verify(workdir, verifier, answer, sandbox=sandbox, task_id=task["task_id"],
  528                                   run_id=run_id, trial_id=trial_id,
  529                                   timeout_s=verify_timeout_s, env=env_spec,
  530                                   service_specs=service_specs, running=running)
  531                 if protocol is not None:
  532                     verdict["protocol"] = protocol
  533 
  534         # The environment's own spend, read from the recordings **before teardown**:
  535         # the containers are about to be removed and their evidence with them.
  536         env_usage = running.usage() if running is not None else {}
  537 
  538         return {"task_id": task["task_id"], "answer": answer,
  539                 "trace": trace, "exit_code": exit_code, "stderr": stderr,
  540                 "verdict": verdict, "env_usage": env_usage,
  541                 "harness_runtime": harness_runtime.record(
  542                     overlay, platform_interpreter=(image_python is None)),
  543                 "runtime_boundary": {
  544                     "agent_timeout_s": timeout_s,
  545                     "verifier_timeout_s": verify_timeout_s,
  546                     "agent_network": env_spec.get("network", "inherited") if not is_exec else network,
  547                     "agent_package_projected": bool(package_overlays),
  548                     "verifier_package": "trusted"},
  549                 # `account()` and not just "published": the connection counts are what
  550                 # turn a harness-side `Connection error` into a named cause. Read while
  551                 # the gateway is still up -- `finally` stops it right after this returns.
  552                 "model_gateway": ({"published": True,
  553                                    "target": f"{model_gateway.target[0]}:{model_gateway.target[1]}",
  554                                    **model_gateway.account()}
  555                                   if model_gateway is not None else None),
  556                 "web_gateway": ({"published": True, "url": web_gateway.url,
  557                                  "search": web_gateway.search, "fetch": web_gateway.fetch}
  558                                 if web_gateway is not None else None),
  559                 "identity": _last_identity(trace)}
  560     finally:
  561         if stepwise_relay is not None:
  562             stepwise_relay.stop()
  563         if stepwise_dir is not None:
  564             shutil.rmtree(stepwise_dir, ignore_errors=True)
  565         # Step 8, and in a `finally` for the reason the table in §2.5.7 gives: a leaked
  566         # network would be inherited by the next task, where it would look like
  567         # isolation that is not there. `stop()` never raises.
  568         if model_gateway is not None:
  569             model_gateway.stop()
  570         if web_gateway is not None:
  571             web_gateway.stop()
  572         if owns_network and network_name is not None:
  573             leaked_net = services_mod.remove_network(network_name)
  574             if leaked_net:
  575                 print(f"warning: could not remove the model gateway network for "
  576                       f"{task['task_id']}: {leaked_net}", file=sys.stderr)
  577         if running is not None:
  578             leaked = running.stop()
  579             if leaked:
  580                 # Not raised. The task's result is the platform's report about the
  581                 # *task*, and a teardown problem after it is a fact about the run.
  582                 print(f"warning: could not fully tear down services for "
  583                       f"{task['task_id']}: {leaked}", file=sys.stderr)
  584         shutil.rmtree(workdir, ignore_errors=True)
  585         if runtime_dir is not None:
  586             shutil.rmtree(runtime_dir, ignore_errors=True)

  625 def _provider_failure(result: dict) -> dict | None:
  626     """harness 是不是**因为连不上模型端点**而死的。`None` 表示不是。
  627 
  628     判据刻意很窄,只在这种情况成立:
  629 
  630       * harness 以**非零退出码**结束 —— 它自己没把这次故障吸收掉;
  631       * stderr 的**尾部**有客户端连接/超时的痕迹 —— 尾部,不是全文:一个连上又掉了、
  632         但自己重试成功的 harness 会在中间留下同样的字眼,而那是它做对了。
  633 
  634     退出码 0 的一律不算:它跑完了,分数就是它该得的分数。
  635     """
  636     if not result.get("exit_code"):            # None(无效)或 0 都不算
  637         return None
  638     tail = (result.get("stderr") or "")[-1500:]
  639     for marker in _PROVIDER_FAILURES:
  640         if marker in tail:
  641             # 摘出那一行,放进记录里 —— 读者要的是"哪一句"。
  642             line = next((l.strip() for l in reversed(tail.splitlines())
  643                          if marker in l), marker)
  644             return {"marker": marker, "line": line[:240],
  645                     "exit_code": result.get("exit_code")}
  646     return None

  649 def _invalid(task: dict, exc, *, stage: str, service: str | None) -> dict:
  650     """A task the platform could not measure, as opposed to one it measured as zero.
  651 
  652     INTERFACE.md §2.5.7: the result is **not** a `per_task` score and **not** a zero,
  653     and it is attributed to the dataset and the service by name. The distinction is the
  654     entire reason this platform exists — a broken mock API and a weak harness must not
  655     be the same number — and it is independently attested: JarvisBench reports that
  656     "harness or provider failures are marked invalid rather than assigned a score of
  657     zero" (`INTERFACE.md` §2.5.7).
  658     """
  659     return {"task_id": task["task_id"], "invalid": {
  660                 "stage": stage, "service": service, "detail": str(exc)},
  661             "answer": "", "trace": "", "exit_code": None, "stderr": "",
  662             "env_usage": {}, "model_gateway": None, "identity": None}

  667 def _spread(xs: list[float]) -> float:
  668     """Population standard deviation, with no dependency and no ambiguity.
  669 
  670     The population form because these are *all* the trials this run took, not a sample
  671     from a larger population of runs -- `statistics.pstdev` is the honest name for it, and
  672     the difference from `stdev` matters most at small n, which is exactly where a reader
  673     compares two harnesses.
  674     """
  675     if len(xs) < 2:
  676         return 0.0
  677     mu = sum(xs) / len(xs)
  678     return (sum((x - mu) ** 2 for x in xs) / len(xs)) ** 0.5

  681 def _harness_failure(result: dict) -> dict | None:
  682     """Why a harness left nothing to read, when it left nothing.
  683 
  684     A task can be **measured** and still say nothing about the harness: the check reads
  685     the artifact, finds it wrong, and records a 0.0 -- and that 0.0 is a real score. But
  686     if the harness *also* wrote no trace, then no method has anything to read while the
  687     curve reports a number, and those two situations are indistinguishable on the curve.
  688 
  689     Measured, and the reason this exists: a harness that needed a package the task image
  690     did not have died on its first model call with `ModuleNotFoundError`, exit code 1.
  691     The trace was 0 bytes, the score was 0.000, and the run reported *nothing* -- the
  692     exit code and stderr were computed here and then dropped, so "the method decided not
  693     to edit" was the only reading available. A harness that cannot start is a fact about
  694     the run and belongs in the record.
  695 
  696     Every harness the platform ships writes `trace.jsonl`, so an empty one is always
  697     abnormal; a harness that simply chose to write nothing is worth knowing about too.
  698     Deliberately **not** folded into `invalid`: the task was measured, so it keeps its
  699     score. This is the attribution, not a reason to drop the measurement (§2.5.7).
  700 
  701     **The second trigger, and the measured case that forced it.** A non-empty trace is
  702     not proof that the harness ran. Measured 2026-10-04 on the current platform:
  703     `--harness cli_agent` with no CLI writes exactly one line,
  704     `{"error": "HG_CLI is unset: no CLI to run"}`, never calls a model, exits 2 -- and the
  705     curve said `score 0.0, harness_failed: null, invalid: null`. The console warned and
  706     the verdict file carried the exit code, so the fact existed in two places and not on
  707     the point, which is where a reader compares runs.
  708 
  709     So: a harness that **exited non-zero without recording a single model call** is a
  710     harness failure too. "A model call" is read from the `usage` block, which the harness
  711     contract requires (`INTERFACE.md` §1.2) -- the platform is entitled to rely on it, and
  712     it is the only schema-independent evidence that a harness talked to a model. A
  713     scripted harness that legitimately makes no calls (`base_harness/oracle`) exits 0 and
  714     is not flagged.
  715     """
  716     trace = (result.get("trace") or "").strip()
  717     reason = None
  718     if not trace:
  719         reason = "the harness left no trace at all"
  720     elif result.get("exit_code") and _last_usage(trace) is None:
  721         # **只说平台知道的事。** 第一版这条写的是"没有记录到任何模型调用",而那是错的:
  722         # `usage` 块是 harness 跑完之后才写一次的,所以**中途崩溃**永远看起来像"一次都没调
  723         # 过模型"。实测(2026-10-04,本地模型,5 道题):三道题各有 15-23 次调用、轨迹 12-14 万
  724         # 字节,控制台却按"left no trace"播报 —— 一句假话,而且是我自己上一版引入的。
  725         reason = (f"the harness exited {result['exit_code']} and its trace has no `usage` "
  726                   f"block, so the run did not reach its end")
  727     if reason is None:
  728         return None
  729     return {"reason": reason, "exit_code": result.get("exit_code"),
  730             "stderr": (result.get("stderr") or "")[-1500:]}

  733 def _reporter():
  734     """The reporter an evaluation should narrate through.
  735 
  736     The context first, then the console's own. Both are needed and neither is
  737     redundant:
  738 
  739     * the **context** is how a task's phases reach the code that runs them
  740       (`eval/containers/container.py` beats while it waits inside a single `docker exec`);
  741     * the **console** is what makes that context exist at all when the driver has not
  742       set one. Measured: the driver's own calls to `evaluate` (`driver.py` round 0, the
  743       mode B loop, the eval side) run on the main thread with no context set, and the
  744       first version of this put the progress events *only* in the context -- so a real
  745       run reported the method call's phases and nothing at all for the four tasks it
  746       spent the hour on. A run that narrates its shortest wait and hides its longest
  747       is worse than one that narrates neither.
  748     """
  749     from eval.observe.console import CONSOLE
  750     return progress_mod.current() or CONSOLE.reporter

  753 def evaluate(repo: Path, tasks: list[dict], scorable: dict[str, str],
  754              cache: dict | None = None, harness_sha: str = "",
  755              sandbox: bool = True, setups: dict | None = None,
  756              verifiers: dict | None = None,
  757              envs: dict | None = None,
  758              run_id: str = "", run_seed: str = "",

  781             result["enforcement"] = diagnostic_authorization.record()
  782         return result

  785 def _in_parallel(tasks: list[dict], measure, jobs: int) -> dict[str, dict]:
  786     """Measure every task, up to `jobs` at a time, and collect the outcomes by task id.
  787 
  788     `jobs=1` stays a plain sequential loop, because the platform's default record must not
  789     depend on a thread pool to be the record it has always been.
  790 
  791     Why this axis and not another: on a measured three-task run the agent phase was 87.8% of
  792     the wall clock (2578 s of 2936 s) and the platform sat idle for all of it -- the tasks are
  793     independent by construction (own container, own network, own model gateway, own workspace,
  794     own cache key), so this is the cheap parallelism. Splitting *inside* a task is not
  795     available: the harness is a process the platform does not control.
  796 
  797     The first exception is re-raised after cancelling what has not started. A task already
  798     running is left to finish: `run_one` tears its own containers down in a `finally`, and
  799     abandoning a thread mid-teardown is how a leaked network gets inherited by whatever runs
  800     next (§2.5.7).
  801     """
  802     if jobs <= 1:
  803         # Computed in task order, which is what makes the merge reproducible at one
  804         # `--jobs` setting as well as comparable across two of them.
  805         return {task["task_id"]: measure(index, task)
  806                 for index, task in enumerate(tasks, start=1)}
  807     pool = futures.ThreadPoolExecutor(max_workers=jobs, thread_name_prefix="hg-task")
  808     pending = [pool.submit(measure, index, task)
  809                for index, task in enumerate(tasks, start=1)]
  810     outcomes: dict[str, dict] = {}
  811     try:
  812         for done in futures.as_completed(pending):
  813             outcome = done.result()
  814             outcomes[outcome["task_id"]] = outcome
  815             # **Stop paying for a harness that cannot start.** See `_all_unstarted`: the
  816             # failure is task-independent, so `jobs` agreements are enough, and the rest
  817             # of the round can only return the same `0.000` at full price.
  818             if _all_unstarted(outcomes, jobs):
  819                 for fut in pending:
  820                     fut.cancel()
  821                 break
  822     except BaseException:
  823         pool.shutdown(wait=True, cancel_futures=True)
  824         raise
  825     pool.shutdown(wait=True)
  826     # **A cancelled task is not a task that scored nothing.** The merge indexes `outcomes`
  827     # by every task, so the cancelled ones are marked `not_run` and kept out of `per_task`
  828     # rather than given a zero the platform never measured (§2.5.7).
  829     for task in tasks:
  830         outcomes.setdefault(task["task_id"],
  831                             {"task_id": task["task_id"], "not_run": True})
  832     return outcomes
  833 
  834 
  835 def _all_unstarted(outcomes: dict[str, dict], jobs: int) -> bool:
  836     """Has the harness failed to *start* on every task measured so far, `jobs` of them?
  837 
  838     This is the platform's own question (`_harness_failure`) asked about a round instead of
  839     a task. A harness whose entrypoint cannot be invoked fails identically on every task --
  840     that is the task-independent part of a run -- so `jobs` agreements are enough to stop
  841     the round. Measured (`codex-dgm`, 2026-10-05): the improver's edit dropped the
  842     `--task`/`--workdir` contract, all ten tasks produced `FileNotFoundError: '--task'`
  843     with no trace, and that happened in **two consecutive rounds** -- twenty
  844     task-measurements whose only possible contribution was `0.000` over and over.
  845 
  846     Deliberately not "any task failed to start": a task-specific failure (an image with no
  847     interpreter, say) is a fact about *that task*, and stopping the round on it would trade
  848     one wrong record for another.
  849     """
  850     if len(outcomes) < jobs:
  851         return False
  852     for outcome in outcomes.values():
  853         if outcome.get("not_run") or outcome.get("invalid") or "cached" in outcome:
  854             return False
  855         measured = outcome.get("measured") or []
  856         if not measured or not _harness_failure(measured[-1]):
  857             return False
  858     return True

  861 def _evaluate(repo: Path, tasks: list[dict], scorable: dict[str, str],
  862               cache: dict | None = None, harness_sha: str = "",
  863               sandbox: bool = True, setups: dict | None = None,
  864               verifiers: dict | None = None,
  865               envs: dict | None = None,
  866               round_no: int = 0,
  867               run_id: str = "", run_seed: str = "",
  868               progress_dir: Path | None = None,
  869               recordings_root: Path | None = None,
  870               trials: int = 1, jobs: int = 1,
  871               diagnostic_authorization: Authorization | None = None) -> dict:
  872     """Score a harness on a task set.
  873 
  874     `trials` is how many times each `(harness, task)` pair is measured. One is the
  875     default and one is not enough: measured on this platform, **the same unmodified
  876     harness scored 0.333 in one run and 0.000 in the next**, because the agent model
  877     intermittently emits a tool call in a dialect the harness cannot parse and the harness
  878     then ends the task. Nothing about that is a property of the dataset; it is the floor of
  879     what a single sample can tell you, and a curve that does not say how noisy it is cannot
  880     support a claim that an edit helped. With `trials > 1` a task's score is the mean over
  881     trials and `per_task_std` is their spread (population standard deviation).
  882 
  883     `jobs` is how many tasks are measured **at the same time**, and it multiplies with
  884     `trials` rather than replacing it: trials buy a spread, jobs buy wall clock. Both are
  885     recorded -- `trials` on the point, `jobs` in the run's own parameters -- because a run
  886     that used four containers at once is a different claim about this host than one that
  887     used one.
  888     """
  889     if diagnostic_authorization is not None:
  890         diagnostic_authorization.require_scope(
  891             run_id, [t["task_id"] for t in tasks], trials=trials, jobs=jobs, round_no=round_no)
  892         if cache:
  893             raise ValueError("unobserved diagnostic requires an empty cache; no cached measurements")
  894         # A later ordinary caller must not reuse a diagnostic's provisional score.
  895         cache = {}
  896     if trials < 1:
  897         raise ValueError(f"trials must be at least 1, not {trials}")
  898     if jobs < 1:
  899         raise ValueError(f"jobs must be at least 1, not {jobs}")
  900     evaluation_protocols = {task["task_id"]: protocol for task in tasks
  901                             if (protocol := chess_stepwise.protocol_of(task)) is not None}
  902     if len(set(evaluation_protocols.values())) > 1:
  903         raise ValueError("cannot aggregate interactive and offline ChessBench protocols")
  904 
  905     def measurement_key(task):
  906         key = f"{harness_sha}:{task['task_id']}:{trials}"
  907         # A changed scorer declaration must not reuse another runtime's score.
  908         bundle = ((verifiers or {}).get(task["task_id"]) or {}).get("scorer_bundle")
  909         if bundle is not None:
  910             key += ":scorer:" + hashlib.sha256(json.dumps(
  911                 bundle, sort_keys=True, default=str).encode()).hexdigest()
  912         if evaluation_protocols.get(task["task_id"]) == chess_stepwise.STEPWISE:
