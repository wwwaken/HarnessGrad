    1 """Services: containers that run alongside a task. INTERFACE.md §2.5.7.
    2 
    3 A service is a mock API, a database, a browser, a simulated user — something that has
    4 to be running *while* the harness runs and reachable only from inside the task's
    5 environment.
    6 
    7 This module owns the lifecycle and nothing else: `eval/containers/container.py` is the docker
    8 primitive layer, and what is here is the order and the rules, which are the parts that
    9 are hard to get right.
   10 
   11 Three of those rules are the whole design:
   12 
   13 **Per-task network, `--internal`.** Per task rather than per run, because task *N*'s
   14 service must not be reachable from task *N+1*: a shared network would make every
   15 service in a run reachable from every task in it, which is a cross-task leak of exactly
   16 the kind this platform otherwise spends its effort preventing. `--internal` because it
   17 is what makes "reachable, and not on the internet" true rather than hoped for —
   18 measured on this host: a peer on such a network resolves and answers, `1.1.1.1` does
   19 not, and DNS fails.
   20 
   21 **The recording is mounted into the service and the check, and never into the
   22 harness.** Every input before this one was static, so re-materializing it defeated
   23 editing (§2.2). A recording is produced while the harness runs, so there is nothing to
   24 re-materialize and the only available rule is a mount rule. `recordings_for` is
   25 therefore the single place that decides who sees it, and the harness is not in the
   26 list.
   27 
   28 **A service that is not healthy refuses the task.** Not a zero: a zero is
   29 indistinguishable from a weak harness, and the difference is the entire reason this
   30 platform exists. `ServiceUnavailable` carries enough to attribute the failure to the
   31 dataset and the service by name.
   32 """
   33 from __future__ import annotations
   34 
   35 import json
   36 import os
   37 import re
   38 import subprocess
   39 import time
   40 from pathlib import Path
   41 
   42 import eval.containers.container as container_mod
   43 from harnessgrad.config import current as current_config
   44 
   45 #: Where a service's recording appears, inside the service container and inside the
   46 #: check. **Fixed by the platform and never named by the dataset** — a dataset-chosen
   47 #: path is a path chosen by something a method may influence (§2.5.8).
   48 RECORDINGS_ROOT = "/recordings"
   49 
   50 #: The one filename inside a recording that the platform itself reads. Everything else
   51 #: in the directory belongs to the dataset's check.
   52 USAGE_FILE = "harnessgrad.json"
   53 
   54 #: How often to re-run a health check while waiting for it to pass.
   55 HEALTH_POLL_S = 0.5
   56

   58 class ServiceUnavailable(RuntimeError):
   59     """A service could not be started, or never became healthy.
   60 
   61     Distinct from a harness failure on purpose: this is a **dataset-side** failure, and
   62     INTERFACE.md §2.5.7 requires it to be attributed rather than scored.
   63     """
   64 
   65     def __init__(self, message: str, *, service: str | None = None,
   66                  stage: str = "start"):
   67         super().__init__(message)
   68         self.service = service
   69         #: `start`, `health` or `died` -- which part of the lifecycle failed. Recorded
   70         #: because "the mock API never answered its own health check" and "the mock API
   71         #: crashed halfway through the task" are different dataset bugs.
   72         self.stage = stage
   73

   77 def label(name: str) -> str:
   78     """A docker-legal name fragment, and the harness-facing variable name.
   79 
   80     Not sanitized silently in the dataset layer (`registry.SERVICE_NAME_RE` already
   81     restricts it to a DNS label), so this only has to handle the run id, which the
   82     platform generates and which may contain characters docker rejects.
   83     """
   84     return re.sub(r"[^A-Za-z0-9_.-]", "-", name)
   85 
   86 
   87 #: Kept so the module's own call sites read unchanged.
   88 _label = label

   91 def url_var(service_name: str) -> str:
   92     """The variable the harness reads to find a service.
   93 
   94     `HG_SVC_*`, not `HG_ENV_*`: the two prefixes have two different audiences, and a
   95     single prefix would make "may the harness see this" a question a reader answers by
   96     knowing which keys happen to exist — which is how a credential leaks (§2.5.7).
   97     """
   98     return f"HG_SVC_{service_name.upper().replace('-', '_')}_URL"

  101 def harness_env(services: list[dict]) -> dict[str, str]:
  102     """What the harness is told about the services, and all it is told.
  103 
  104     The URL rather than the address: an address is a fact about one docker network,
  105     and the name is the thing the dataset declared and can reason about.
  106     """
  107     return {url_var(svc["name"]): f"http://{svc['name']}:{svc['port']}"
  108             for svc in services}

  111 def service_env(services: list[dict], *, task_id: str, run_seed: str) -> dict[str, str]:
  112     """What the *services* get, and the harness never does.
  113 
  114     A third model budget, after the harness's and the method's, separated the same way
  115     the other two are: `harness_env()` strips `HG_METHOD_*`, and this is handed only to
  116     the service containers. An environment whose simulator can spend the agent's
  117     credential is an environment that can impersonate the agent.
  118     """
  119     # A service is scoped to one task.  Exposing this identifier lets an image bind its
  120     # API to the current task without trusting the request body to choose a dataset row.
  121     env: dict[str, str] = {"HG_ENV_TASK_ID": task_id}
  122     if any(svc.get("needs_seed") for svc in services):
  123         env["HG_ENV_SEED"] = task_seed(run_seed, task_id)
  124     if any(svc.get("needs_model") for svc in services):
  125         # The third model budget, from the declared configuration rather than from
  126         # `os.environ` at the point of use. The variable names stay the same because a
  127         # service image reads them by name; only the platform's read moved.
  128         environment = current_config().environment
  129         for key, value in (("HG_ENV_MODEL", environment.model),
  130                            ("HG_ENV_API_KEY", environment.api_key),
  131                            ("HG_ENV_BASE_URL", environment.base_url)):
  132             if value:
  133                 env[key] = value
  134     return env

  137 def task_seed(run_seed: str, task_id: str) -> str:
  138     """A per-task seed derived from the run's, so it is reproducible from the record.
  139 
  140     Derived rather than drawn: a seed chosen at run time and written down is
  141     reproducible only if the record survives, and a seed derived from `(run_seed,
  142     task_id)` is reproducible from two facts the curve already carries (§2.5.4).
  143     """
  144     import hashlib
  145     return hashlib.sha256(f"{run_seed}:{task_id}".encode()).hexdigest()[:16]

  152 def requires_seed(services: list[dict]) -> bool:
  153     return any(svc.get("needs_seed") for svc in services)

  158 class RunningServices:
  159     """The containers started for one task, and the network they are on.
  160 
  161     Held as one object so that teardown cannot be forgotten piecemeal: `stop` is the
  162     only way to end a task's services and it removes the network too. A leaked network
  163     is the failure mode here — containers carry `--rm`, networks do not.
  164     """
  165 
  166     def __init__(self, *, network: str, containers: dict[str, str],
  167                  recordings: dict[str, Path], services: list[dict]):
  168         self.network = network
  169         self.containers = containers
  170         self.recordings = recordings
  171         self.services = services
  172 
  173     def urls(self) -> dict[str, str]:
  174         return {svc["name"]: f"http://{svc['name']}:{svc['port']}"
  175                 for svc in self.services}
  176 
  177     def usage(self) -> dict:
  178         """The environment's own model spend, read from the recordings.
  179 
  180         Summed from `harnessgrad.json` in each recording rather than asked for over the
  181         network: the service has stopped or is about to, and a number the platform asks
  182         for is a number it can be lied to about. `eval/run/runner.py` reads a harness's
  183         usage from its trace for the same reason.
  184         """
  185         total: dict[str, int] = {}
  186         reported = 0
  187         for name, path in self.recordings.items():
  188             record = path / USAGE_FILE
  189             if not record.is_file():
  190                 continue
  191             try:
  192                 payload = json.loads(record.read_text())
  193             except (OSError, json.JSONDecodeError):
  194                 continue
  195             usage = payload.get("usage")
  196             if not isinstance(usage, dict):
  197                 continue
  198             reported += 1
  199             for key in ("input_tokens", "output_tokens", "calls"):
  200                 value = usage.get(key)
  201                 if isinstance(value, int):
  202                     total[key] = total.get(key, 0) + value
  203         total["services_reported"] = reported
  204         return total
  205 
  206     def stop(self) -> list[str]:
  207         """Tear down containers and the network. Never raises.
  208 
  209         Never raises because it runs in a `finally`, and a teardown that can throw
  210         replaces the real failure with a misleading one. It returns what it could not
  211         remove so a caller that cares (a test) can say so.
  212         """
  213         problems: list[str] = []
  214         for name, cid in self.containers.items():
  215             # --rm destroys the service's only startup/action diagnostics. Keep them
  216             # beside its recording before removal, including on failed health checks.
  217             record = self.recordings.get(name)
  218             if record is not None:
  219                 try:
  220                     logs = container_mod.docker("logs", "--tail", "200", cid, timeout=10)
  221                     (record / "service.log").write_text(
  222                         logs.stdout + logs.stderr, encoding="utf-8")
  223                 except (subprocess.TimeoutExpired, OSError):
  224                     pass  # Evidence collection must never prevent teardown.
  225             try:
  226                 proc = container_mod.docker("rm", "--force", cid, timeout=60)
  227             except (subprocess.TimeoutExpired, OSError) as exc:
  228                 problems.append(f"service {name!r}: {exc}")
  229                 continue
  230             if proc.returncode != 0 and "No such container" not in proc.stderr:
  231                 problems.append(f"service {name!r}: {proc.stderr.strip()[:200]}")
  232         problems.extend(remove_network(self.network))
  233         return problems
  234

  236 def create_network(run_id: str, task_id: str) -> str:
  237     """Create the task's `--internal` network. Returns its name.
  238 
  239     Public because a task needs one even when it declares no services: a containerised
  240     harness cannot reach a model on the host's loopback, so the platform publishes the
  241     endpoint on this network (`eval/service/modelgate.py`). Without that the harness has no
  242     model, and a harness with no model is not being measured.
  243     """
  244     net = label(f"hg-{run_id}-{task_id}")
  245     proc = container_mod.docker("network", "create", "--internal", net,
  246                                 "--label", f"harnessgrad.run={run_id}",
  247                                 "--label", f"harnessgrad.task={task_id}", timeout=120)
  248     if proc.returncode != 0:
  249         raise ServiceUnavailable(
  250             f"could not create the internal network for task {task_id!r}: "
  251             f"{proc.stderr.strip()[:300]}")
  252     return net
  253 
  254 
  255 def remove_network(network: str) -> list[str]:
  256     """Remove a network. Never raises, like `RunningServices.stop`."""
  257     try:
  258         proc = container_mod.docker("network", "rm", network, timeout=60)
  259     except (subprocess.TimeoutExpired, OSError) as exc:
  260         return [f"network {network!r}: {exc}"]
  261     # Docker 29 uses "network NAME not found"; older daemons use "No such
  262     # network". Reaping followed by a finally-block stop must remain idempotent.
  263     missing = ("No such network" in proc.stderr or
  264                f"Error response from daemon: network {network} not found"
  265                in proc.stderr.splitlines())
  266     if proc.returncode != 0 and not missing:
  267         return [f"network {network!r}: {proc.stderr.strip()[:200]}"]
  268     return []

  271 def reap(run_id: str) -> list[str]:
  272     """Remove containers and networks a **previous crash** left under this run id.
  273 
  274     Measured (2026-10-04): a run killed mid-task leaves its per-task networks and any
  275     running service container behind -- `local-floor-5/6` and `local-cap12` each left
  276     one network per task, and a probe run left a service container up for 15 hours.
  277     `docker run --rm` does not help: it removes a container when it *exits*, and a
  278     killed driver never stops it. Ctrl-C is handled (`except BaseException` around the
  279     service start path), but SIGTERM and SIGKILL are not, and neither is a test that
  280     is interrupted.
  281 
  282     The cost is not disk. It is that the **next** run with the same id dies at
  283     `docker network create ... already exists`, so a measurement is lost to a failure
  284     that belongs to the previous crash and is reported as if it were about this run.
  285 
  286     Called at the run's first write, beside the same decision for the run directory
  287     (`_write_run_meta(..., fresh=True)`): a reused id means "start clean". Only this
  288     run id's objects are touched, so a different run executing concurrently is not
  289     disturbed. Never raises; returns what it could not remove.
  290     """
  291     problems: list[str] = []
  292     selector = f"label=harnessgrad.run={run_id}"
  293     #: And by **name**, because the label is not always there. The harness container was
  294     #: created with only `--name` until 2026-10-04 (`eval/containers/container.py:create` now labels
  295     #: it too), so the containers a hard-killed run actually leaves behind are exactly the
  296     #: ones a label search cannot see -- measured on `screen-all-1`: five containers still
  297     #: `running`, and this function reported only "network ... has active endpoints".
  298     #: `hg-<run_id>-` is the platform's own naming (`_task_name`), so matching it cannot
  299     #: reach another run's objects.
  300     prefix = f"hg-{run_id}-"
  301     for kind, list_args, remove_args, by_name in (
  302             ("container", ["container", "ls", "--all", "--quiet"], ["rm", "--force"],
  303              True),
  304             ("network", ["network", "ls", "--quiet"], ["network", "rm"], True)):
  305         found: list[str] = []
  306         for flt in ([selector] + ([f"name={prefix}"] if by_name else [])):
  307             try:
  308                 listed = container_mod.docker(*list_args, "--filter", flt, timeout=60)
  309             except (subprocess.TimeoutExpired, OSError) as exc:
  310                 problems.append(f"{kind} list ({flt}): {exc}")
  311                 continue
  312             if listed.returncode != 0:
  313                 problems.append(f"{kind} list ({flt}): {listed.stderr.strip()[:200]}")
  314                 continue
  315             found += [i for i in listed.stdout.split() if i not in found]
  316         for ident in found:
  317             try:
  318                 removed = container_mod.docker(*remove_args, ident, timeout=120)
  319             except (subprocess.TimeoutExpired, OSError) as exc:
  320                 problems.append(f"{kind} {ident[:12]}: {exc}")
  321                 continue
  322             if removed.returncode != 0:
  323                 problems.append(f"{kind} {ident[:12]}: {removed.stderr.strip()[:200]}")
  324     return problems

  327 def gateway_ip(network: str) -> str:
  328     """The host-side address of a task network, which is where a gateway can bind.
  329 
  330     The host is the gateway of a docker network, so a process bound here is reachable
  331     from inside the network and unreachable from outside it -- measured, and it is the
  332     property the model gateway depends on.
  333     """
  334     proc = container_mod.docker("network", "inspect", network, "-f",
  335                                 "{{(index .IPAM.Config 0).Gateway}}", timeout=60)
  336     ip = proc.stdout.strip()
  337     if proc.returncode != 0 or not ip:
  338         raise ServiceUnavailable(
  339             f"could not read the gateway address of network {network!r}: "
  340             f"{proc.stderr.strip()[:200]}")
  341     return ip

  344 def start(services: list[dict], *, recordings_root: Path, run_id: str,
  345           task_id: str, run_seed: str) -> RunningServices:
  346     """Create the network, start every service, and wait for each to be healthy.
  347 
  348     Raises `ServiceUnavailable` — having already torn down whatever it started, so a
  349     failed task does not leave a network behind for the next one to inherit.
  350     """
  351     # The same safety net `eval/container.run` carries, for the same reason: the driver
  352     # pins every service image before the run starts (§2.5.4), and a service started
  353     # from a tag would pull whatever the registry served that day.
  354     for svc in services:
  355         if not container_mod.looks_pinned(svc["image"]):
  356             raise ServiceUnavailable(
  357                 f"service {svc['name']!r} names {svc['image']!r}, which is not pinned "
  358                 f"to a content address. Resolve it with `container.resolve_digest` "
  359                 f"first (INTERFACE.md §2.5.4).", service=svc["name"], stage="start")
  360 
  361     net = create_network(run_id, task_id)
  362 
  363     containers: dict[str, str] = {}
  364     recordings: dict[str, Path] = {}
  365     try:
  366         # Configuration can fail after network creation; keep it inside the same
  367         # cleanup boundary as container startup.
  368         env = service_env(services, task_id=task_id, run_seed=run_seed)
  369         for svc in services:
  370             name = svc["name"]
  371             record_host: Path | None = None
  372             if svc.get("record"):
  373                 # Under the **run directory**, with the round diffs and the curve: a
  374                 # recording is evidence the run produced, and the place evidence goes
  375                 # is the record. An earlier version put it under the work root, where
  376                 # nothing ever removed it -- and a per-task directory that no code owns
  377                 # is a directory that grows for the life of the machine.
  378                 record_host = recordings_root / task_id / name
  379                 record_host.mkdir(parents=True, exist_ok=True)
  380                 recordings[name] = record_host
  381 
  382             cmd = ["run", "--detach", "--rm",
  383                    "--name", label(f"hg-{run_id}-{task_id}-{name}"),
  384                    # The container's *name* is platform-mangled -- it has to be unique
  385                    # on the daemon, and several runs may be live at once -- so the name
  386                    # the harness was told must be attached as an **alias**, or it
  387                    # resolves to nothing. Measured: without this the harness got
  388                    # `Name or service not known` for a service that was running and
  389                    # healthy, so every task that touched it failed for a reason nothing
  390                    # recorded, and the tasks that only touched the filesystem passed.
  391                    "--network", net, "--network-alias", name,
  392                    "--user", f"{os.getuid()}:{os.getgid()}",
  393                    "--label", f"harnessgrad.run={run_id}",
  394                    "--label", f"harnessgrad.task={task_id}",
  395                    "--label", f"harnessgrad.service={name}"]
  396             # Service resources are opt-in; task ENV caps do not cover sidecars.
  397             # Reuse the container envelope (including its fixed PID cap) only when
  398             # the service declares resources, preserving existing uncapped services.
  399             if "cpus" in svc or "memory_mb" in svc:
  400                 limits = container_mod.limits_from(svc)
  401                 cmd += ["--cpus", str(limits["cpus"]),
  402                         "--memory", limits["memory"],
  403                         "--pids-limit", str(limits["pids"])]
  404             if record_host is not None:
  405                 cmd += ["--mount",
  406                         f"type=bind,src={record_host},dst={RECORDINGS_ROOT}/{name}"]
  407             for key, value in sorted(env.items()):
  408                 cmd += ["-e", f"{key}={value}"]
  409             # No docker socket, and no host tree: a service is infrastructure the
  410             # harness must not be able to read or kill its way out of (§2.5.8).
  411             if "entrypoint" in svc:
  412                 cmd += ["--entrypoint", svc["entrypoint"]]
  413             cmd += [svc["image"], *svc.get("command", [])]
  414             started = container_mod.docker(*cmd, timeout=180)
  415             if started.returncode != 0:
  416                 raise ServiceUnavailable(
  417                     f"service {name!r} did not start: {started.stderr.strip()[:300]}",
  418                     service=name, stage="start")
  419             containers[name] = started.stdout.strip()
  420 
  421         running = RunningServices(network=net, containers=containers,
  422                                   recordings=recordings, services=services)
  423         for svc in services:
  424             _await_health(running, svc)
  425         return running
  426     except BaseException:
  427         # Including `KeyboardInterrupt` and `ServiceUnavailable` itself: the one thing
  428         # that must not happen is a network surviving a failed start and being reused
  429         # by the next task, where it would look like isolation that is not there.
  430         RunningServices(network=net, containers=containers,
  431                         recordings=recordings, services=services).stop()
  432         raise

  435 def _await_health(running: RunningServices, svc: dict) -> None:
  436     """Run the service's own health check until it passes or its budget runs out.
  437 
  438     The check runs **inside the service container**, so it sees the service as the
  439     service sees itself — a host-side port probe would answer a different question and
  440     would be defeated by a container that binds to localhost only.
  441     """
  442     health = svc["health"]
  443     argv = [str(a) for a in health["argv"]]
  444     budget = float(health.get("timeout_s", 60))
  445     deadline = time.monotonic() + budget
  446     last = ""
  447     cid = running.containers[svc["name"]]
  448     while True:
  449         try:
  450             proc = container_mod.docker("exec", cid, *argv, timeout=max(10, int(budget)))
  451         except (subprocess.TimeoutExpired, OSError) as exc:
  452             last = str(exc)
  453         else:
  454             if proc.returncode == 0:
  455                 return
  456             last = f"exit {proc.returncode}: {(proc.stdout + proc.stderr).strip()[:200]}"
  457         if time.monotonic() >= deadline:
  458             raise ServiceUnavailable(
  459                 f"service {svc['name']!r} was not healthy after {budget:g}s "
  460                 f"({last}). This is a dataset-side failure, not a harness result: the "
  461                 f"task is recorded as invalid rather than scored zero.",
  462                 service=svc["name"], stage="health")
  463         time.sleep(HEALTH_POLL_S)

  466 # --------------------------------------------------------------- mount rules ---
  467 
  468 #: Who may see a service's recording. **One place, because the rule is the design.**
  469 #: The harness is absent from this mapping, and that absence is the point: a recording
  470 #: is the only input in this platform the harness cannot even read, and a decision with
  471 #: no code is a decision that gets reversed by accident.
  472 _AUDIENCES = ("harness", "verifier")
  473 
  474 
  475 def mounts_for(audience: str, services: list[dict],
  476                recordings: dict[str, Path]) -> list[str]:
  477     """Docker mount arguments for one audience.
  478 
  479     `harness` → nothing at all. Not read-only, not a copy: the harness cannot see a
  480     recording, because there is no static version to withhold and a copy written before
  481     the run would be a copy of nothing.
  482 
  483     `verifier` → every recording, read-only. Read-only because a check that writes into
  484     the evidence it grades has a verdict that depends on its own leftovers, the same
  485     reason the task mount is frozen for the check (§2.5.6).
  486     """
  487     if audience not in _AUDIENCES:
  488         raise ValueError(f"unknown audience {audience!r}; known: {list(_AUDIENCES)}")
  489     if audience == "harness":
  490         return []
  491     mounts: list[str] = []
  492     for svc in services:
  493         path = recordings.get(svc["name"])
  494         if path is not None:
  495             mounts += ["--mount",
  496                        f"type=bind,src={path},dst={RECORDINGS_ROOT}/{svc['name']},readonly"]
  497     return mounts
