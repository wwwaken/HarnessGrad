    1 r"""StrategyQA as a dataset: yes/no questions whose official metric is exact string match.
    2 
    3     python3 tools/datasets.py --fetch --dataset strategyqa
    4     python3 data/scaffolds/qa/control.py --dataset strategyqa --oracle
    5     driver.py --dataset strategyqa ...
    6 
    7 Where the tasks come from
    8 -------------------------
    9 `data/bigbench/strategy_qa.csv` in `github.com/billxbf/ReWOO` at commit `9cd0283...`,
   10 2,290 rows of `input,target`. The input carries its own answer contract verbatim
   11 (`Answer with 'Yes' or 'No': ...`), so the goal does not add one -- two answer contracts
   12 in one prompt is how a model ends up following the one the scorer is not reading.
   13 
   14 **Why not BIG-bench main.** Measured: the task directory is *gone* from
   15 `google/BIG-bench` main (`strategy_qa` was removed). ReWOO's vendored copy is the
   16 reachable one, and it is the copy whose numbers appear in that paper, so using it keeps
   17 the two comparable. the same tarball (pinned by commit sha) serves all four datasets that
   18 take a table from it.
   19 
   20 The metric, verbatim
   21 --------------------
   22 The dataset's own home no longer ships a scorer, so the reference is ReWOO's
   23 `utils/Evaluator.py` -- the code that produced the published numbers for this table:
   24 
   25     em = normalize_answer(pred) == normalize_answer(label)
   26 
   27 `normalize_answer` is lower-case, strip punctuation, remove articles, white-space fix.
   28 The comparison is therefore **lenient about form and strict about content**: `yes.`,
   29 `Yes` and `YES` all score, while `Yes, because ...` does not -- because the whole string
   30 is compared, not the first token. That is ReWOO's rule and it is transcribed in the test
   31 rather than paraphrased.
   32 
   33 A note on the other two numbers in that Evaluator
   34 -------------------------------------------------
   35 It also reports `f1` and an LLM `acc`. `f1` on a one-token gold is a reward for writing
   36 extra words, and `acc` needs an unpinned judge. This adapter computes neither; `em` is the
   37 reproducible, published one.
   38 """

   63 # ------------------------------------------------------------------ the scorer ---

   65 
   66 def normalize_answer(s: str) -> str:
   67     """ReWOO `utils/Evaluator.py::normalize_answer`, verbatim."""
   68 
   69     def remove_articles(text: str) -> str:
   70         return re.sub(r"\b(a|an|the)\b", " ", text)
   71 
   72     def white_space_fix(text: str) -> str:
   73         return " ".join(text.split())
   74 
   75     def remove_punc(text: str) -> str:
   76         return "".join(ch for ch in text if ch not in set(string.punctuation))
   77 
   78     def lower(text: str) -> str:
   79         return text.lower()
   80 
   81     return white_space_fix(remove_articles(remove_punc(lower(str(s)))))
   82 
   83 
   84 def exact_match(prediction: str, ground_truth: str) -> bool:
   85     """ReWOO's `em`: normalised string equality, over the **whole** answer."""
   86     return normalize_answer(prediction) == normalize_answer(ground_truth)
   87 
   88 
   89 def score(prediction: str, gold: dict) -> dict:
   90     want = str(gold.get("answer", ""))
   91     hit = exact_match(prediction or "", want)
   92     return {
   93         "reward": 1.0 if hit else 0.0,
   94         "correct": hit,
   95         "official": {"metric": "exact_match after ReWOO's normalize_answer",
   96                      "label": want,
   97                      "normalized_prediction": normalize_answer(prediction or "")},
   98         "why": (f"{'matches' if hit else 'no match for'} "
   99                 f"{normalize_answer(prediction or '')!r} vs {normalize_answer(want)!r}; "
  100                 f"the whole answer is compared, not its first token"),
  101     }

  107 def _rows() -> list[dict]:
  108     path = DATASET_ROOT / CSV_PATH
  109     if not path.is_file():
  110         raise FileNotFoundError(f"no strategy_qa.csv at {path}")
  111     with path.open(encoding="utf-8", newline="") as handle:
  112         rows = list(csv.DictReader(handle))
  113     if len(rows) != N_TOTAL:
  114         raise ValueError(f"strategyqa: {len(rows)} rows, expected {N_TOTAL}")
  115     return rows

  118 def _gold(row: dict) -> dict:

  121         raise ValueError(f"strategyqa: target {label!r} is not one of {LABELS}")
  122     return {"answer": label}
