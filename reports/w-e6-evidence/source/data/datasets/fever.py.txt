    1 r"""FEVER adapter using the official strict evidence metric.
    2 
    3 The adapter reads the pinned `copenlu/fever_gold_evidence` validation split and keeps
    4 that provenance explicit; it does not claim test gold access. `prediction.json` carries
    5 the answer label and optional `(page, line)` pairs. The scorer checks the first five pairs,
    6 and preserves the upstream evidence exemption for NOT ENOUGH INFO. A legacy answer.txt
    7 prediction still scores, with absent required evidence measured as incorrect.
    8 """

   12 from data.scaffolds.qa.evidence_scorers import MissingEvidence, fever_is_strictly_correct

   17 REVISION = "a6b8d891d393e97a4efac791afffb2d7de5e57c6"
   18 VALID = "valid.jsonl"
   19 TEST = "test.jsonl"
   20 TRAIN = "train.jsonl"
   21 LABELS = ("SUPPORTS", "REFUTES", "NOT ENOUGH INFO")

   27 def score_prediction(prediction: dict, gold: dict) -> dict:
   28     answer = prediction["answer"]
   29     predicted_evidence = prediction.get("predicted_evidence")
   30     try:
   31         correct = fever_is_strictly_correct(
   32             gold["answer"], answer, gold.get("evidence"), predicted_evidence)
   33     except MissingEvidence:
   34         correct = False
   35     return {"reward": float(correct), "correct": correct,
   36             "official": {"metric": "FEVER strict_score", "max_evidence": 5},
   37             "why": "strict label and evidence match" if correct else
   38                   "label or required evidence did not match"}
   39 
   40 
   41 def score(prediction: str, gold: dict) -> dict:
   42     # Legacy answer.txt remains scoreable; omitted evidence is an incorrect prediction.
   43     return score_prediction({"answer": prediction}, gold)
   44 
   45 
   46 def _rows() -> list[dict]:
   47     path = DATASET_ROOT / VALID
   48     if not path.is_file():
   49         raise FileNotFoundError(f"no {VALID} at {path}")
   50     return base.jsonl_rows(path)
