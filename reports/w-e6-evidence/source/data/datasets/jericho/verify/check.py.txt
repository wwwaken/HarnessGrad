    1 #!/usr/bin/env python3
    2 r"""The Jericho check: replay the agent's commands in a fresh game and read the game's score.
    3 
    4 It reads one thing from the agent: the ordered commands in `jer_actions.jsonl`. It then
    5 rebuilds the game from a trusted copy of `jer_game.json` **and a trusted copy of the ROM**
    6 (both re-materialized by the platform immediately before this runs), replays the commands, and
    7 reports the game's own score over the game's own maximum.
    8 
    9 The rule, precisely
   10 -------------------
   11     reward = info['score'] / env.get_max_score()
   12 
   13 `info['score']` is the points the game awards, which is what Jericho's own documentation prints
   14 ("Total Score", "Scored X out of Y"), and `get_max_score()` is a number the game states about
   15 itself. The raw pair, the per-step trace, and the score the reference walkthrough would reach
   16 are all in the evidence, so "0.42" can be read back into "147 of 350".
   17 
   18 **No `done` gate.** A game ends when the game says so; `jer done` exists so a harness can stop
   19 and is recorded, not required -- the same reading as ALFWorld's and ScienceWorld's. §2.
   20 
   21 What makes this check trustworthy
   22 ---------------------------------
   23 * The game file, the ROM and this script are re-materialized by the platform before the check
   24   runs, so editing any of them in the workdir changes nothing.
   25 * The seed is the game's official one, from the same file the agent was given, so the check
   26   replays the episode the agent actually played rather than a different arrangement.
   27 * A check that cannot measure -- no game file, no ROM, an interpreter that will not start --
   28   writes **no reward file**, writes evidence saying `measured: false`, and exits 3.
   29 
   30 Outputs
   31 -------
   32     reward file    0.0-1.0   (`/verifier/reward.txt` by default)
   33     evidence file  JSON      why: every step, its score, the maximum, the fraction
   34     stdout         why, in one line, so the platform's verdict detail carries it too
   35 """
   36 from __future__ import annotations
   37 
   38 import argparse
   39 import json
   40 import sys
   41 import traceback
   42 from pathlib import Path
   43 
   44 sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
   45 import jer_env  # noqa: E402
   46 
   47 DEFAULT_REWARD_FILE = "/verifier/reward.txt"
   48 DEFAULT_EVIDENCE_FILE = "/verifier/evidence.json"
   49 
   50 
   51 def parse_args(argv):
   52     parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
   53     parser.add_argument("--workdir", default=".")
   54     parser.add_argument("--game", default=None, help="the trusted game file")
   55     parser.add_argument("--rom", default=None, help="the ROM name, for evidence")
   56     parser.add_argument("--log", default=None)
   57     parser.add_argument("--reward-file", default=DEFAULT_REWARD_FILE)
   58     parser.add_argument("--evidence-file", default=DEFAULT_EVIDENCE_FILE)
   59     args = parser.parse_args(argv)
   60     workdir = Path(args.workdir)
   61     args.game = Path(args.game) if args.game else workdir / jer_env.GAME_FILENAME
   62     args.log = Path(args.log) if args.log else workdir / jer_env.LOG_FILENAME
   63     args.reward_file = Path(args.reward_file)
   64     args.evidence_file = Path(args.evidence_file)
   65     args.workdir = workdir
   66     return args
   67 
   68 
   69 def write_reward(path: Path, score: float) -> None:
   70     path.parent.mkdir(parents=True, exist_ok=True)
   71     path.write_text(f"{score:g}\n")
   72 
   73 
   74 def write_evidence(path: Path, evidence: dict) -> None:
   75     """Evidence is for a reader, never for the score: failing to write it is not fatal."""
   76     try:
   77         path.parent.mkdir(parents=True, exist_ok=True)
   78         path.write_text(json.dumps(evidence, indent=1, sort_keys=True,
   79                                    ensure_ascii=False) + "\n")
   80     except OSError as exc:
   81         print(f"could not write evidence: {exc}", file=sys.stderr)
   82 
   83 
   84 def main(argv) -> int:
   85     args = parse_args(argv)
   86     evidence: dict = {
   87         "game_file": str(args.game),
   88         "rom": args.rom,
   89         "action_log": str(args.log),
   90     }
   91 
   92     if not args.game.is_file():
   93         evidence.update(measured=False, error=f"no game file at {args.game}")
   94         write_evidence(args.evidence_file, evidence)
   95         print(f"check cannot measure: {evidence['error']}", file=sys.stderr)
   96         return 3
   97 
   98     records = jer_env.action_log_records(args.log)
   99     actions = jer_env.actions_in(records)
  100     declared_done = any(r.get("kind") == "done" for r in records)
  101     blank_or_broken = sum(
  102         1 for line in (args.log.read_text(errors="replace").splitlines()
  103                        if args.log.is_file() else []) if line.strip()) - len(records)
  104     evidence.update(actions_in_log=len(actions), log_records=len(records),
  105                     unparseable_lines=max(0, blank_or_broken),
  106                     agent_declared_done=declared_done)
  107 
  108     try:
  109         env = jer_env.load_game(args.workdir)
  110     except Exception as exc:  # noqa: BLE001 - a check that cannot run is not a zero
  111         evidence.update(measured=False, error=f"could not start the game: {exc}",
  112                         traceback=traceback.format_exc()[-2000:])
  113         write_evidence(args.evidence_file, evidence)
  114         print(f"check cannot measure: {evidence['error']}", file=sys.stderr)
  115         return 3
  116 
  117     try:
  118         maximum = int(env.get_max_score())
  119         spec = json.loads(args.game.read_text())
  120         steps = jer_env.replay(env, actions)
  121     except Exception as exc:  # noqa: BLE001
  122         evidence.update(measured=False, error=f"the game died mid-episode: {exc}",
  123                         traceback=traceback.format_exc()[-2000:])
  124         write_evidence(args.evidence_file, evidence)
  125         print(f"check cannot measure: {evidence['error']}", file=sys.stderr)
  126         return 3
  127 
  128     score = steps[-1]["score"] if steps else 0
  129     moves = steps[-1]["moves"] if steps else 0
  130     fraction = jer_env.score_fraction(score, maximum)
  131     evidence.update(
  132         measured=True,
  133         reward=fraction,
  134         score=score,
  135         max_score=maximum,
  136         moves=moves,
  137         seed=int(spec.get("seed") or 0),
  138         actions_replayed=len(steps),
  139         actions_not_replayed=len(actions) - len(steps),
  140         game_over=bool(steps and steps[-1]["done"]),
  141         solved=bool(maximum > 0 and score >= maximum),
  142         steps=steps,
  143     )
  144     if score >= maximum and maximum > 0:
  145         why = f"the game is finished: {score}/{maximum} points in {moves} moves"
  146     elif not actions:
  147         why = "the log holds no actions, so no points were scored"
  148     elif not steps:
  149         why = (f"none of the {len(actions)} logged action(s) replayed; the game is still at "
  150                f"{score}/{maximum}")
  151     else:
  152         why = (f"{score}/{maximum} points ({fraction:.3f}) after {moves} moves; the game is "
  153                f"not finished")
  154     evidence["why"] = why
  155     write_evidence(args.evidence_file, evidence)
  156 
  157     write_reward(args.reward_file, fraction)
  158     print(f"reward {fraction:g}: {why}")
  159     return 0 if fraction >= 1.0 else 1
  160 
  161 
  162 if __name__ == "__main__":
  163     try:
  164         raise SystemExit(main(sys.argv[1:]))
  165     except SystemExit:
  166         raise
  167     except Exception:  # noqa: BLE001 - a crash must not look like a measured zero
  168         traceback.print_exc()
  169         raise SystemExit(4)
