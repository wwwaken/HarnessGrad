  173 def test_fold() -> list[dict]:
  174     """The official `test` fold, after the budget filter. Read once per process.
  175 
  176     The file is a transcription of the repository's `goldpaths-all.zip` (see the module
  177     docstring for why it is in git rather than fetched), and it carries **every** test path
  178     with its length, including the 22 that are too long. The filter lives here so the
  179     exclusion is recomputed from the data on every load, and the raise below is the guard
  180     that keeps it honest: if the budget is ever lowered without updating the filter, the
  181     load fails loudly instead of silently scoring a correct oracle 0.
  182     """
  183     global _FOLD_CACHE
  184     if _FOLD_CACHE is not None:
  185         return _FOLD_CACHE
  186     if not FOLD_FILE.is_file():
  187         raise FileNotFoundError(f"no test-fold file at {FOLD_FILE}")
  188     payload = json.loads(FOLD_FILE.read_text())
  189     if payload.get("eval_fold") != EVAL_FOLD:
  190         raise ValueError(f"{FOLD_FILE} describes fold {payload.get('eval_fold')!r}, not "
  191                          f"{EVAL_FOLD!r}")
  192     records = []
  193     too_long = []
  194     for entry in payload["tasks"]:
  195         record = {"task_id": f"{entry['task_name']}/{entry['variation']}",
  196                   "task_name": str(entry["task_name"]),
  197                   "variation": int(entry["variation"]),
  198                   "gold_moves": int(entry["gold_moves"])}
  199         (too_long if record["gold_moves"] > STEP_BUDGET else records).append(record)
  200     if not records:
  201         raise ValueError(f"{FOLD_FILE} leaves no tasks inside the {STEP_BUDGET}-move "
  202                          f"budget")
  203     if len(records) + len(too_long) == 0:
  204         raise ValueError(f"{FOLD_FILE} has no tasks")
  205     _FOLD_CACHE = records
  206     return records

  421 def load() -> tuple[list[dict], dict[str, str]]:
  422     """`(tasks, scorable)`. Every task is graded by its `VERIFY` entry, so the `scorable`
  423     map is present only because the interface asks for it."""

  425         records = _ensure()

  429             f"with `python3 tools/datasets.py --fetch --dataset scienceworld` and check "
  430             f"that `pip show scienceworld` reports 1.2.3.") from exc

  432         raise ValueError("scienceworld: the enumeration returned no tasks")

  437 def _is_eval(task_id: str) -> bool:
  438     """A deterministic third, by hash of the task id.
  439 
  440     Hashed rather than sliced off the sorted list so that adding or removing a task does
  441     not reshuffle which side every other task is on -- the same rule `data/datasets/alfworld.py` and
  442     `data/datasets/terminal_bench.py` use. The official `train`/`dev`/`test` folds exist in the
  443     repository's `goldpaths-all.zip` but are not in the package; see the module docstring.
  444     """
  445     import hashlib
  446     return hashlib.sha256(task_id.encode()).digest()[0] % 3 == 0

  455 def _split() -> dict:
  456     """The imposed train/eval split, computed without materialising the per-task dicts.
  457 
  458     Deriving it from the records would mean building 21,621 goal/verifier structures just
  459     to answer "which side is this task on", and `tools/dataset_lint.py` and the driver both
  460     read `SPLIT` before anything else. The identity is the same generator the records use,
  461     so the two cannot disagree.

  463     train, eval_ = [], []

  465         (eval_ if _is_eval(record["task_id"]) else train).append(record["task_id"])
  466     return {"train": train, "eval": eval_}

  473     would make importing this adapter cost a JVM start plus 21,621 dict constructions -- the
  474     lesson the QA batch recorded in `docs/dataset-onboarding-skill.md` §15.7. The platform
  475     asks for them through module `__getattr__` below.
  476     """
  477 
  478     def __init__(self) -> None:
  479         self._cache: dict = {}
  480 
  481     def get(self, name: str):
  482         if name not in self._cache:
  483             records = _ensure()
  484             ids = [r["task_id"] for r in records]
  485             self._cache = {
  486                 "ENV": {tid: {"kind": "files"} for tid in ids},
  487                 "SETUP": {r["task_id"]: _setup(r) for r in records},
  488                 "VERIFY": {r["task_id"]: _verifier(r) for r in records},
  489             }
  490         return self._cache[name]
  491 
  492 
  493 _LAZY = _Lazy()
  494 
  495 _split_mode = (os.environ.get("HG_SCIENCEWORLD_SPLIT") or "hash").lower()
  496 
  497 
  498 def __getattr__(name: str):
  499     if name in ("ENV", "SETUP", "VERIFY"):
  500         return _LAZY.get(name)
  501     if name == "SPLIT":
  502         if _split_mode == "none":
  503             # Every task as eval, accepting the driver's warning -- ALFWorld's escape hatch,
  504             # for the same reason: the benchmark declares no division and the one here is
  505             # imposed, so wanting the whole set scored must be sayable without editing code.
  506             ids = [r["task_id"] for r in _ensure()]
  507             return {"train": [], "eval": ids}
  508         return _split()
  509     raise AttributeError(f"module {__name__!r} has no attribute {name!r}")
