    1 #!/usr/bin/env python3
    2 """The ALFWorld check: replay the agent's actions in a **fresh** environment and read
    3 the environment's own `won` flag.
    4 
    5 This is the dataset's whole grade. It does not read a summary the agent wrote, it does
    6 not read the agent's observation, and it does not read any `won` the agent's CLI
    7 recorded -- all of those are the agent's own bytes and the agent can make them say
    8 anything. It reads one thing from the agent: the ordered list of actions in
    9 `alf_actions.jsonl`. It then rebuilds the game from a trusted copy of the game file,
   10 applies those actions one at a time to TextWorld, and asks TextWorld whether the PDDL
   11 goal is satisfied. Success is `infos["won"]`, exactly as
   12 `alfworld/agents/eval/evaluate_dagger.py` reads it.
   13 
   14 The rule, precisely
   15 -------------------
   16     reward = 1  iff  the replayed episode reports `won` at some step
   17     reward = 0  otherwise
   18 
   19 "At some step" is upstream's rule, not a convenience: ALFWorld's evaluator takes the
   20 maximum of the per-step `won` scores over the episode. An episode that satisfies the
   21 goal and then keeps acting is still a success, and the replay stops at the first `done`
   22 (TextWorld's own end-of-episode signal) exactly as the evaluator's loop does.
   23 
   24 **`alf done` is recorded but is not a gate.** tau-bench's check *must* require `tau
   25 done`, because tau-bench itself only computes a reward for a finished episode. ALFWorld
   26 has no such rule: an episode ends when TextWorld says it ends (the goal is met or the
   27 step budget is spent), and nothing in the benchmark asks the agent to declare anything.
   28 Adding a `done` gate here would be inventing a requirement the benchmark does not have
   29 -- which is a different failure from dropping one, and just as misleading. The record
   30 still carries whether `done` was declared, because it is useful evidence.
   31 
   32 What makes this check trustworthy
   33 ---------------------------------
   34 * The game file is re-materialized by the platform at verification time
   35   (`VERIFY.inputs`), so editing `<workdir>/game.tw-pddl` does not change the grade.
   36 * The same is true of this file and of `gym_env.py`: `SETUP` ships them for the agent's
   37   own CLI, and verification overwrites them. The check imports `gym_env` from the
   38   workdir *after* that overwrite.
   39 * An episode the check cannot measure -- a game that will not load, a game file that is
   40   missing -- writes **no reward file** and exits 3. The platform then reports the task
   41   as unmeasured rather than as a zero the agent earned. A bare 0 is indistinguishable
   42   from a wrong answer, and that distinction is the whole reason this file writes
   43   evidence.
   44 
   45 Outputs
   46 -------
   47     reward file    1 or 0   (`/verifier/reward.txt` by default)
   48     evidence file  JSON     why, per action: what happened, which step won, what the
   49                             official per-step reduction would have produced
   50     stdout         why, in one line, so the platform's verdict detail carries it too
   51 """
   52 from __future__ import annotations
   53 
   54 import argparse
   55 import json
   56 import os
   57 import sys
   58 import traceback
   59 from pathlib import Path
   60 
   61 sys.path.insert(0, str(Path(__file__).resolve().parent))
   62 import gym_env  # noqa: E402
   63 
   64 #: The container contract's locations. On a host (`files`) environment these are
   65 #: overridden by the flags below, because `/verifier` does not exist there.
   66 DEFAULT_REWARD_FILE = "/verifier/reward.txt"
   67 DEFAULT_EVIDENCE_FILE = "/verifier/evidence.json"
   68 
   69 
   70 def parse_args(argv):
   71     parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
   72     parser.add_argument("--workdir", default=os.environ.get("ALFWORLD_WORKDIR", "."))
   73     parser.add_argument("--game", default=None, help="the trusted game file")
   74     parser.add_argument("--log", default=None, help="the agent's action log")
   75     parser.add_argument("--reward-file",
   76                         default=os.environ.get("ALFWORLD_REWARD_FILE",
   77                                                DEFAULT_REWARD_FILE))
   78     parser.add_argument("--evidence-file",
   79                         default=os.environ.get("ALFWORLD_EVIDENCE_FILE",
   80                                                DEFAULT_EVIDENCE_FILE))
   81     parser.add_argument("--budget", type=int, default=gym_env.STEP_BUDGET)
   82     args = parser.parse_args(argv)
   83     workdir = Path(args.workdir)
   84     args.game = Path(args.game) if args.game else workdir / gym_env.GAME_FILENAME
   85     args.log = Path(args.log) if args.log else workdir / gym_env.LOG_FILENAME
   86     args.reward_file = Path(args.reward_file)
   87     args.evidence_file = Path(args.evidence_file)
   88     return args
   89 
   90 
   91 def write_reward(path: Path, score: int) -> None:
   92     path.parent.mkdir(parents=True, exist_ok=True)
   93     path.write_text(f"{score}\n")
   94 
   95 
   96 def write_evidence(path: Path, evidence: dict) -> None:
   97     """Evidence is for a reader, never for the score: failing to write it is not fatal."""
   98     try:
   99         path.parent.mkdir(parents=True, exist_ok=True)
  100         path.write_text(json.dumps(evidence, indent=1, sort_keys=True,
  101                                    ensure_ascii=False) + "\n")
  102     except OSError as exc:
  103         print(f"could not write evidence: {exc}", file=sys.stderr)
  104 
  105 
  106 def official_reduction(won_per_step: list[bool]) -> float:
  107     """The per-episode score ALFWorld's own evaluator computes from a `won` sequence.
  108 
  109     Copied from `alfworld/agents/eval/evaluate_dagger.py`, which builds
  110 
  111         step_rewards = [curr - prev for curr, prev in zip(scores, prev_rewards)]
  112         ...
  113         game_points = np.max(np.array(sequence_game_points), 0).tolist()
  114 
  115     with `scores = [float(item) for item in infos["won"]]`. Because `won` is monotone
  116     (TextWorld does not un-win a game) and the sequence starts at 0.0, the maximum of
  117     the increments equals "did any step win", which is what this file returns. The
  118     reduction is computed here anyway so the evidence carries both numbers, and
  119     `tests/quality/test_alfworld_official_scorer.py` pins the equality against a
  120     verbatim copy of upstream's code -- the copy, not an import of their package.
  121     """
  122     prev = 0.0
  123     increments: list[float] = []
  124     for won in won_per_step:
  125         curr = 1.0 if won else 0.0
  126         increments.append(curr - prev)
  127         prev = curr
  128     return max(increments) if increments else 0.0
  129 
  130 
  131 def main(argv) -> int:
  132     args = parse_args(argv)
  133     evidence: dict = {
  134         "game_file": str(args.game),
  135         "action_log": str(args.log),
  136         "reward_file": str(args.reward_file),
  137     }
  138 
  139     if not args.game.is_file():
  140         # The check's own input is missing: not a measurement.
  141         evidence["error"] = f"no game file at {args.game}"
  142         evidence["measured"] = False
  143         write_evidence(args.evidence_file, evidence)
  144         print(f"check cannot run: {evidence['error']}", file=sys.stderr)
  145         return 3
  146 
  147     records = gym_env.action_log_records(args.log)
  148     actions = [str(r["action"]) for r in records
  149                if r.get("kind") == "action" and isinstance(r.get("action"), str)]
  150     declared_done = any(r.get("kind") == "done" for r in records)
  151     unparseable = len([line for line in
  152                        (args.log.read_text(errors="replace").splitlines()
  153                         if args.log.is_file() else []) if line.strip()]) - len(records)
  154     evidence.update(actions_in_log=len(actions), log_records=len(records),
  155                     unparseable_lines=unparseable, agent_declared_done=declared_done,
  156                     budget=args.budget)
  157 
  158     try:
  159         state = gym_env.replay(str(args.game), actions, args.budget)
  160     except Exception as exc:  # noqa: BLE001 - a check that cannot run is not a zero
  161         evidence["error"] = f"could not build or replay the environment: {exc}"
  162         evidence["traceback"] = traceback.format_exc()[-2000:]
  163         evidence["measured"] = False
  164         write_evidence(args.evidence_file, evidence)
  165         print(f"check cannot run: {evidence['error']}", file=sys.stderr)
  166         return 3
  167 
  168     won_per_step = [bool(r["won"]) for r in state["records"]]
  169     won_step = next((i for i, w in enumerate(won_per_step) if w), None)
  170     replayed = state["records"]
  171     budget_exhausted = (len(actions) > len(replayed)
  172                         and not (replayed and replayed[-1]["done"]))
  173     # A command TextWorld did not understand comes back as "Nothing happens." It still
  174     # cost a step, and recording it is how a reader tells "the agent tried and was
  175     # refused" apart from "the agent did nothing".
  176     no_ops = [r["action"] for r in replayed
  177               if "Nothing happens" in str(r.get("observation", ""))]
  178     reward = 1 if state["won"] else 0
  179     official = official_reduction(won_per_step)
  180 
  181     evidence.update(
  182         measured=True,
  183         reward=reward,
  184         won=bool(state["won"]),
  185         won_at_step=won_step,
  186         actions_replayed=len(replayed),
  187         actions_not_replayed=len(actions) - len(replayed),
  188         budget_exhausted=bool(budget_exhausted),
  189         episode_done=bool(replayed and replayed[-1]["done"]),
  190         no_op_actions=no_ops,
  191         won_per_step=won_per_step,
  192         official_game_points=official,
  193         # The agent's own narrative is not trusted, so it is not included. What is
  194         # included is the *action* at each step and TextWorld's answer, which is the
  195         # material a reader needs to see why the reward is what it is.
  196         steps=[{"step": i, "action": r["action"], "won": bool(r["won"]),
  197                 "done": bool(r["done"]),
  198                 "observation": str(r.get("observation", ""))[:200]}
  199                for i, r in enumerate(replayed)],
  200     )
  201 
  202     if won_step is None:
  203         if budget_exhausted:
  204             why = (f"the goal was not satisfied in the {args.budget}-step budget; the "
  205                    f"log holds {len(actions)} actions and only {len(replayed)} replayed")
  206         elif not actions:
  207             why = "the log holds no actions, so the goal cannot have been satisfied"
  208         else:
  209             why = (f"the goal was not satisfied after {len(replayed)} action(s); the "
  210                    f"episode ended without a win")
  211     else:
  212         why = f"the goal was satisfied at step {won_step + 1} of {len(replayed)}"
  213     evidence["why"] = why
  214     write_evidence(args.evidence_file, evidence)
  215 
  216     if reward == 1:
  217         write_reward(args.reward_file, 1)
  218         print(f"reward 1: {why} (official game points {official:g}; "
  219               f"agent declared done={declared_done})")
  220         return 0
  221 
  222     write_reward(args.reward_file, 0)
  223     print(f"reward 0: {why} (official game points {official:g}; "
  224           f"no-op actions {len(no_ops)})")
  225     return 1
  226 
  227 
  228 if __name__ == "__main__":
  229     try:
  230         raise SystemExit(main(sys.argv[1:]))
  231     except SystemExit:
  232         raise
  233     except Exception:  # noqa: BLE001 - a crash must not look like a measured zero
  234         traceback.print_exc()
  235         raise SystemExit(4)
