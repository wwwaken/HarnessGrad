    1 r"""Sports Understanding as a dataset: is this sports sentence plausible?
    2 
    3     python3 tools/datasets.py --fetch --dataset sportsunderstanding
    4     python3 data/scaffolds/qa/control.py --dataset sportsunderstanding --oracle
    5     driver.py --dataset sportsunderstanding ...
    6 
    7 Where the tasks come from
    8 -------------------------
    9 `data/bigbench/sports_understanding.csv` in `github.com/billxbf/ReWOO` at commit
   10 `9cd0283...`, 1,000 rows of `input,target`. The input is BIG-bench's own sentence and
   11 already names the two labels (`Determine whether the following statement ... are
   12 plausible or implausible: ...`), so the goal adds no second answer contract.
   13 
   14 The metric, and what "official" means here
   15 ------------------------------------------
   16 BIG-bench's `sports_understanding/task.json` scores this task with
   17 **`multiple_choice_grade`** against `target_scores` (`{'plausible': 1, 'implausible': 0}`
   18 or the reverse), not with a free-text metric. ReWOO's copy flattened that into a
   19 `target` column holding the winning label, which is what this table carries.
   20 
   21 `multiple_choice_grade` for a two-way generative task is "does the response identify the
   22 correct choice". The rule transcribed in the test is the narrowest one that does that
   23 without guessing: the answer must contain the correct label, case-insensitively.
   24 
   25 **The one honest weakness, stated here and not discovered later.** A response containing
   26 **both** labels (`it could be plausible or implausible`) satisfies a substring rule for
   27 either gold and is therefore counted correct for both. That is the same failure
   28 `data/datasets/fever.py` describes for a `contains` metric, and it is inherent to substring
   29 matching, not an implementation choice. It is reported rather than hidden: this adapter
   30 treats a two-label answer as correct for neither, because an answer that names both
   31 choices did not choose -- and that **differs from a plain substring rule**, so the test
   32 pins both the rule and the divergence.
   33 
   34 There is no third option in the table: every target is one of the two labels.
   35 """

   54 CSV_PATH = ("ReWOO-9cd0283043ff4be0c9d614fda2789d143ca6ffd1/data/bigbench/"

   56 N_TOTAL = 1000
   57 LABELS = ("plausible", "implausible")

   60 # ------------------------------------------------------------------ the scorer ---
   61 
   62 
   63 def substring_choice(prediction: str, label: str) -> bool:
   64     """Is the label named as a **word** in the answer?
   65 
   66     Not a bare `label in answer`, and the difference is measured: `"implausible"` contains
   67     `"plausible"` as a substring, so the naive rule scores the answer `implausible` as
   68     correct for a gold of `plausible` -- a silent 100% on one of the two classes. Word
   69     boundaries fix it without changing what the rule means.
   70     """
   71     pattern = r"(?<![A-Za-z])" + re.escape(str(label).lower()) + r"(?![A-Za-z])"
   72     return re.search(pattern, str(prediction).lower()) is not None
   73 
   74 
   75 def names_both(prediction: str) -> bool:
   76     """Does the answer name both choices? Then it did not choose."""
   77     return all(substring_choice(prediction, label) for label in LABELS)
   78 
   79 
   80 def score(prediction: str, gold: dict) -> dict:
   81     want = str(gold.get("answer", "")).lower()
   82     both = names_both(prediction or "")
   83     hit = (not both) and substring_choice(prediction or "", want)
   84     return {
   85         "reward": 1.0 if hit else 0.0,
   86         "correct": hit,
   87         "official": {"metric": "multiple_choice_grade (the correct label named)",
   88                      "label": want,
   89                      "substring_rule": substring_choice(prediction or "", want),
   90                      "named_both_labels": both},
   91         "why": (
   92             f"{'names' if hit else 'does not name'} {want!r}"
   93             + ("; the answer names both labels, so it did not choose" if both else "")
   94         ),
   95     }
   96

  101 def _rows() -> list[dict]:
  102     path = DATASET_ROOT / CSV_PATH

  106         rows = list(csv.DictReader(handle))
  107     if len(rows) != N_TOTAL:
  108         raise ValueError(f"sportsunderstanding: {len(rows)} rows, expected {N_TOTAL}")
  109     return rows

  112 def _gold(row: dict) -> dict:
  113     label = str(row["target"]).strip().lower()
  114     if label not in LABELS:
  115         raise ValueError(f"sportsunderstanding: target {label!r} is not one of {LABELS}")
  116     return {"answer": label}
