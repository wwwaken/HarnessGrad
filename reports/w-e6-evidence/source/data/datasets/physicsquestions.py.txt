    1 r"""Physics Questions as a dataset: numeric answers graded by BIG-bench's string metric.
    2 
    3     python3 tools/datasets.py --fetch --dataset physicsquestions
    4     python3 data/scaffolds/qa/control.py --dataset physicsquestions --oracle
    5     driver.py --dataset physicsquestions ...
    6 
    7 Where the tasks come from
    8 -------------------------
    9 `data/bigbench/physics_question.csv` in `github.com/billxbf/ReWOO` at commit
   10 `9cd0283...`, 53 rows of `input,target`, where the target is BIG-bench's **list** of
   11 acceptable spellings:
   12 
   13     "During a recent morning run, Ken averaged a speed of 5.9 m/s for 14.9 minutes. ..."
   14     -> Can be one of ['7187 m', '7 km']
   15 
   16 **53 rows is a fact about every number here.** The 95% confidence interval on an accuracy
   17 at n=53 is roughly ±13 points -- wider than any difference this dataset could be used to
   18 argue for. It is useful for mechanism attribution (does the harness use a calculator?),
   19 not for ranking.
   20 
   21 The metric, verbatim
   22 --------------------
   23 BIG-bench's `physics_questions/task.json` declares `metrics: ["exact_str_match", "bleu"]`
   24 and its examples carry a **list** target. `exact_str_match` is BIG-bench's standard
   25 metric: the prediction, stripped of surrounding whitespace, must equal one of the targets
   26 exactly, after the reference strips the trailing period. So `7 km` and `7187 m` both score
   27 for the row above, and they are **not equal numbers** (7187 m is 7.187 km) -- they are two
   28 roundings of the same quantity, which is why the target is a list rather than a value.
   29 
   30 **Two things this adapter deliberately does not do**, because both were tempting and both
   31 change the number:
   32 
   33 * it does not compare numerically with a relative tolerance. That is a *different, more
   34   generous* metric: it would score `7.2 km` correct for a target of `7 km`, where
   35   `exact_str_match` does not. A tolerance is a defensible scorer for a *different*
   36   question ("is the physics right?"); it is not this benchmark's number, and reporting it
   37   under this benchmark's name would be the §1 error.
   38 * it does not accept the whole `Can be one of [...]` string as an answer. The list is
   39   BIG-bench's encoding of "any of these", not the answer text.
   40 
   41 `bleu` is the task's second metric and is not reported: a BLEU score over a three-token
   42 numeric answer measures tokenisation, and its corpus implementation depends on a library
   43 this platform does not pin.
   44 """

   64             "physics_question.csv")

   67 #: BIG-bench's target encoding: `Can be one of ['7187 m', '7 km']`. The list is parsed
   68 #: rather than compared as a string.
   69 _TARGET = re.compile(r"^Can be one of \[(?P<items>.*)\]\s*$", re.I)
   70 _QUOTED = re.compile(r"'([^']*)'|\"([^\"]*)\"")
   71 
   72 
   73 def accepted_answers(target: str) -> list[str]:
   74     """The acceptable spellings in a BIG-bench target.
   75 
   76     **Raises when the encoding is not recognised** instead of falling back to the raw
   77     string: comparing a prediction with `Can be one of [...]` scores every answer 0 for a
   78     reason nothing records, and a silent wrong number is worse than an error. Measured:
   79     every one of the 53 rows uses the encoding.
   80     """
   81     match = _TARGET.match(str(target).strip())
   82     if not match:
   83         raise ValueError(f"physicsquestions: target {target!r} is not `Can be one of [...]`")
   84     items = [a or b for a, b in _QUOTED.findall(match.group("items"))]
   85     if not items:
   86         raise ValueError(f"physicsquestions: no quoted answers in {target!r}")
   87     return items
   88

   90 # ------------------------------------------------------------------ the scorer ---
   91 
   92 
   93 def exact_str_match(prediction: str, ground_truth: str) -> bool:
   94     """BIG-bench's `exact_str_match`, as its examples define it.
   95 
   96     The prediction is stripped; the reference has a trailing period stripped (the task's
   97     own examples end in one and the answers do not). Nothing else is normalised: case and
   98     units are part of the answer, which is why the dataset lists several spellings.
   99     """
  100     return str(prediction).strip() == str(ground_truth).strip().rstrip(".")
  101 
  102 
  103 def score(prediction: str, gold: dict) -> dict:
  104     accepted = list(gold.get("accepted") or [])
  105     hit = any(exact_str_match(prediction or "", want) for want in accepted)
  106     return {
  107         "reward": 1.0 if hit else 0.0,
  108         "correct": hit,
  109         "official": {"metric": "exact_str_match over BIG-bench's accepted spellings",
  110                      "accepted": accepted,
  111                      "prediction": str(prediction).strip()},
  112         "why": (f"{str(prediction).strip()!r} "
  113                 f"{'matches' if hit else 'matches none of'} {accepted}"),
  114     }
  115

  120 def _rows() -> list[dict]:
  121     path = DATASET_ROOT / CSV_PATH
  122     if not path.is_file():
  123         raise FileNotFoundError(f"no physics_question.csv at {path}")
  124     with path.open(encoding="utf-8", newline="") as handle:
  125         rows = [row for row in csv.DictReader(handle) if str(row.get("input") or "").strip()]
  126     # BIG-bench's own task.json has one row with an empty input and an empty target
  127     # (measured: 54 examples, the first blank); ReWOO's CSV drops it. 53 is the count of
  128     # real questions, and the adapter asserts it so a re-extraction that changes the count
  129     # is noticed rather than scored.
  130     if len(rows) != N_TOTAL:
  131         raise ValueError(f"physicsquestions: {len(rows)} rows, expected {N_TOTAL}")
  132     return rows
  133
