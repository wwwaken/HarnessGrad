    1 r"""HotpotQA fullwiki validation adapter using the official joint metric.
    2 
    3 `prediction.json` carries the answer and optional `(title, sentence_index)` pairs.
    4 Answer EM/F1 and supporting-fact scores feed the official joint EM/F1; support facts use
    5 exact set semantics. Missing evidence remains a valid prediction and produces joint zero.
    6 The adapter reads only the pinned validation split, not test gold.
    7 """
    8 from __future__ import annotations
    9 
   10 from data.scaffolds.qa import base
   11 from data.scaffolds.qa.evidence_scorers import (hotpot_exact_match, hotpot_f1_score,
   12                                       SpScore, hotpot_joint, hotpot_update_sp, MissingEvidence)
   13 
   14 KIND = "dataset"
   15 
   16 DATASET_ROOT = base.Path("/mnt/20t/xubuqiang/datasets/hotpotqa")
   17 REVISION = "1908d6afbbead072334abe2965f91bd2709910ab"
   18 CONFIG = "fullwiki"
   19 VALIDATION = f"{CONFIG}/validation-00000-of-00001.parquet"
   20 TEST = f"{CONFIG}/test-00000-of-00001.parquet"
   21 
   22 
   23 def _gold(row: dict) -> dict:
   24     facts = row.get("supporting_facts")
   25     if isinstance(facts, dict):
   26         titles = facts.get("title") or []
   27         indices = facts.get("sent_id") or []
   28         if len(titles) != len(indices):
   29             raise ValueError("hotpotqa: supporting_facts title/sent_id lengths differ")
   30         facts = [[title, index] for title, index in zip(titles, indices)]
   31     return {"answer": str(row["answer"]), "supporting_facts": facts}
   32 
   33 EVIDENCE_AWARE = True
   34 ORACLE_SOURCE = "released"
   35 
   36 
   37 def score_prediction(prediction: dict, gold: dict) -> dict:
   38     answer = prediction["answer"]
   39     em = hotpot_exact_match(answer, gold["answer"])
   40     f1, precision, recall = hotpot_f1_score(answer, gold["answer"])
   41     try:
   42         sp = hotpot_update_sp(prediction.get("predicted_evidence"),
   43                               gold.get("supporting_facts"))
   44     except MissingEvidence:
   45         sp = SpScore(em=0.0, f1=0.0, prec=0.0, recall=0.0)
   46     joint_em, joint_f1 = hotpot_joint(em, precision, recall, sp)
   47     return {"reward": joint_em, "correct": bool(joint_em == 1.0),
   48             "official": {"metric": "HotpotQA joint", "em": em, "f1": f1,
   49                           "sp_em": sp.em, "sp_f1": sp.f1,
   50                           "joint_em": joint_em, "joint_f1": joint_f1},
   51             "why": f"joint_em={joint_em:g}, joint_f1={joint_f1:g}"}
   52 
   53 
   54 def score(prediction: str, gold: dict) -> dict:
   55     return score_prediction({"answer": prediction}, gold)
   56 
   57 
   58 def _rows() -> list[dict]:
   59     path = DATASET_ROOT / VALIDATION
   60     if not path.is_file():
   61         raise FileNotFoundError(f"no {VALIDATION} at {path}")
   62     return base.parquet_rows(path)
   63 
   64 
   65 _SPEC = base.QASpec(
   66     name="hotpotqa",
   67     root=DATASET_ROOT,
   68     rows=lambda: {"validation": _rows()},
   69     question=lambda row: str(row["question"]),
   70     gold=_gold,
   71     goal=lambda question, gold: (
   72         f"Question: {question}\n\nWrite the answer to `answer.txt` with `qa submit`, or "
   73         "write `prediction.json` with `qa submit-json @file`: "
   74         "{\"schema_version\":1,\"answer\":\"...\",\"predicted_evidence\":["
   75         "[title,sentence_index]]}. Supporting facts use exact set semantics and are "
   76         "required for a nonzero joint score."),
   77     score=score,
   78     downloads=lambda: [
   79         {
   80             "kind": "hf-dataset",
   81             "name": "hotpot_qa (fullwiki)",
   82             "repo": "hotpot_qa",
   83             "version": REVISION,
   84             "size": "~28 MB (fullwiki validation; the full repository is ~750 MB)",
   85             "files": [VALIDATION],
   86             "sha256": {
   87                 VALIDATION: "78933c0a31a5f7b420d4effdf4cd4eed573b28c6a3da6179dcf7a02b39e51d03",
   88             },
   89             "dst": ".",
   90             "note": "Trap 1: `hotpot_qa` has two configs and they are different "
   91                     "benchmarks -- `distractor` ships ten candidate paragraphs per "
   92                     "question, `fullwiki` ships the whole-Wikipedia setting. Trap 2: only "
   93                     "the `fullwiki/validation` file is fetched; `fullwiki/train` alone is "
   94                     "330 MB and the `distractor` config another 360 MB, none of it scored. "
   95                     "Trap 3: the `supporting_facts` column is the gold evidence a future "
   96                     "answer contract would need -- it is here, the answer channel is what "
   97                     "does not exist.",
   98         },
   99     ],
  100     split="validation",
  101     source="hotpot_qa@1908d6afbbead072334abe2965f91bd2709910ab (config `fullwiki`)",
  102 )

  105 DEFAULT_ENV = _BUILT["DEFAULT_ENV"]
