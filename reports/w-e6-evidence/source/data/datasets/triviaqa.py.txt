    1 r"""TriviaQA as a dataset: open-domain trivia questions graded against an alias set.
    2 
    3     python3 tools/datasets.py --fetch --dataset triviaqa
    4     python3 data/scaffolds/qa/control.py --dataset triviaqa --oracle
    5     python3 data/scaffolds/qa/control.py --dataset triviaqa --oracle --platform
    6     driver.py --dataset triviaqa ...
    7 
    8 Where the tasks come from
    9 -------------------------
   10 Joshi, Choi, Weld, Zettlemoyer, *TriviaQA* (arXiv:1705.03551) -- `mandarjoshi/trivia_qa`,
   11 config **`rc.nocontext`**, split **`validation`**: 17,944 questions, 7.3 MB. That split is
   12 the exam side.
   13 
   14 The **study side** is the benchmark's own `train` split, sampled to 2,000 rows by an even
   15 stride across the file (`STUDY_ROWS`, and the reason is measured in `_rows()`: the file is
   16 ordered by question source, so a prefix would be one source). The division is therefore
   17 TriviaQA's own -- `--split declared` and `--split official` agree -- and the other 136,384
   18 training rows are on disk and unread by design; `--split <file.json>` is how a run asks for
   19 a different study set.
   20 
   21 **Two choices, both forced, both worth stating.**
   22 
   23 1. **`validation`, not `test`.** The benchmark's own `test` split has its gold withheld: in
   24    the mirror copy measured here every row's `answer["value"]` is `"<unk>"` and its
   25    `normalized_aliases` is empty. Scoring on `test` would report the fraction of systems that
   26    answer `"<unk>"`. ReWOO's `run_eval.py` never overrides its loader's `split="test"` default,
   27    so a TriviaQA number attributed to ReWOO has to come from `validation` or from `<unk>`; this
   28    adapter uses `validation` and says so.
   29 2. **`rc.nocontext`, not `rc`.** `nocontext` is the config without retrieved documents, which
   30    is the one a parameterised agent can be scored on without a retrieval tool. It is also the
   31    one ReWOO used (`utils/DataLoader.py::load_trivia_qa`). The full `rc` config is 8 GB.
   32 
   33 The metric and scorer identity
   34 ------------------------------
   35 The official scorer is `evaluation/triviaqa_evaluation.py` in the dataset's own repository:
   36 normalise both sides the same way, and count a hit if the prediction matches **any** alias --
   37 `metric_max_over_ground_truths` takes the `max` over `get_ground_truths(answer)`.
   38 
   39 New evidence records `SCORER_IDENTITY` and the pinned upstream EM reference. This
   40 adapter reports per-row alias-max EM only; it does not implement upstream F1 or
   41 question/document grouping. Historical scores keep their recorded scorer identity.
   42 
   43 `get_ground_truths` is `answer['NormalizedAliases'] + [normalize_answer(a) for a in
   44 answer.get('HumanAnswers', [])]`. Two things about it matter:
   45 
   46 * the official key names are **capitalised** (`NormalizedAliases`, `HumanAnswers`); the
   47   HuggingFace copy lower-cases them and ships **no human answers** for this config, so the
   48   gold set here is the normalised aliases. The code below reads both spellings rather than
   49   one, because a silent `KeyError`-turned-empty-set would score every answer 0.
   50 * the gold set is the **normalised** aliases. Normalising the raw `aliases` instead gives a
   51   different string whenever an alias carries punctuation or an article -- measured on the
   52   first row's aliases, where the two happen to agree, and the rule is what the upstream code
   53   does, not what one row shows.
   54 
   55 What is deliberately not added
   56 ------------------------------
   57 No retrieval tools and no context: `rc.nocontext` is the config that has none, so a score
   58 here measures what the model knows, not what it can look up. A harness that wants to search
   59 is free to, but the dataset does not hand it documents, because the config it pins does not
   60 have any.
   61 """

   85 # ------------------------------------------------------------------ the scorer ---
   86 #
   87 # EM semantics are compared with independently acquired, byte-pinned source in
   88 # `tests/fixtures/triviaqa_upstream`, not a second local transcription. Local
   89 # extensions are string coercion, Hub key spellings, float rewards and empty max=0.

  100 def normalize_answer(s: str) -> str:
  101     """Pinned upstream normalization semantics, with local string coercion.
  102 
  103     Replace underscores, lowercase, replace punctuation with spaces, remove English
  104     articles, then collapse whitespace. For example, `"The A-Team"` becomes `"team"`.
  105     """
  106 
  107     def remove_articles(text: str) -> str:
  108         return re.sub(r"\b(a|an|the)\b", " ", text)
  109 
  110     def white_space_fix(text: str) -> str:
  111         return " ".join(text.split())
  112 
  113     def remove_punc(text: str) -> str:
  114         exclude = set(string.punctuation + "‘’´`")
  115         return "".join(" " if ch in exclude else ch for ch in text)
  116 
  117     def lower(text: str) -> str:
  118         return text.lower()
  119 
  120     def replace_underscore(text: str) -> str:
  121         return text.replace("_", " ")
  122 
  123     return white_space_fix(remove_articles(
  124         remove_punc(lower(replace_underscore(str(s)))))).strip()
  125 
  126 
  127 def get_ground_truths(answer: dict) -> list[str]:
  128     """Upstream alias selection with official and Hub key spellings.
  129 
  130         answer['NormalizedAliases'] + [normalize_answer(a) for a in answer.get('HumanAnswers', [])]
  131 
  132     The HuggingFace columns are `normalized_aliases` / `human_answers` and this config has
  133     no human answers at all, so the second term is empty here. Both spellings are read
  134     because the official JSON and the Hub copy differ, and a missing key must not silently
  135     become an empty gold set.
  136     """
  137     truths = list(answer.get("NormalizedAliases")
  138                   or answer.get("normalized_aliases") or [])
  139     human = answer.get("HumanAnswers") or answer.get("human_answers") or []
  140     return [str(t) for t in truths] + [normalize_answer(h) for h in human]
  141 
  142 
  143 def exact_match_score(prediction: str, ground_truth: str) -> float:
  144     """Upstream normalized exact match, returned as a float reward."""
  145     return float(normalize_answer(prediction) == normalize_answer(ground_truth))
  146 
  147 
  148 def metric_max_over_ground_truths(metric_fn, prediction: str,
  149                                   ground_truths: list[str]) -> float:
  150     """Upstream alias maximum, with a local zero for an empty alias list.
  151 
  152     This `max` is the official rule: one alias is enough. Scoring against `answer["value"]`
  153     alone is a **different, harsher** metric -- the `rewoo` filter, not the official one.
  154     """
  155     if not ground_truths:
  156         return 0.0
  157     return max(metric_fn(prediction, gt) for gt in ground_truths)
  158 
  159 
  160 def score(prediction: str, gold: dict) -> dict:
  161     """The dataset's grade. Called by the shared check (`data/scaffolds/qa/verify/check.py`)."""
  162     accepted = list(gold.get("accepted") or [])
  163     hit = metric_max_over_ground_truths(exact_match_score, prediction or "", accepted) == 1.0
  164     return {
  165         "reward": 1.0 if hit else 0.0,
  166         "correct": hit,
  167         "official": {
  168             "metric": "exact_match over the alias set (metric_max_over_ground_truths)",
  169             "scorer_identity": SCORER_IDENTITY,
  170             "reference": dict(SCORER_REFERENCE),
  171             "accepted": accepted[:20],
  172             "normalized_prediction": normalize_answer(prediction or ""),
  173             "canonical_value": gold.get("value"),
  174         },
  175         "why": (
  176             f"{'matches' if hit else 'no match for'} "
  177             f"{normalize_answer(prediction or '')[:60]!r} among {len(accepted)} alias(es)"
  178             + (f"; canonical value {gold.get('value')!r}"
  179                if gold.get("value") else "")
  180         ),
  181     }

  187 #: The files this adapter knows about. The scored split is read on demand; `train` is read
  188 #: only to build the study side, and only as a sample (see `STUDY_ROWS`).
  189 SPLIT_FILES = {"validation": VALIDATION, "train": TRAIN, "test": TEST}
  190 
  191 #: The two fields this adapter reads. TriviaQA ships `entity_pages` and `search_results`
  192 #: beside them -- the retrieved Wikipedia pages and search results -- which no part of this
  193 #: adapter or its scorer touches, and which are what the decode time is spent on: measured
  194 #: on `train`, **9.46 s** for every column against **0.34 s** for these two.
  195 COLUMNS = ("question", "answer")
  196 
  197 #: How many rows of the official `train` split this adapter loads as study material.
  198 #:
  199 #: The split is 138,384 rows, and loading all of it is not a matter of taste. Measured:
  200 #: `to_pylist()` over every row costs **3.55 s and ~850 MB** of nested `answer` dicts, and
  201 #: a record keeps its whole row, so `records()` for `train` + `validation` (156,328
  202 #: records) is 7.6 s / 1.1 GB and `load()` -- which renders every goal -- did not finish
  203 #: inside 60 s. A study side also costs one harness run per task, so 138k of them is not a
  204 #: round anybody runs.
  205 #:
  206 #: 2,000 is a declared sample, not a truncation of convenience: the rule is in
  207 #: `_rows()`/`data/scaffolds/qa/read.py::parquet_rows(sample=…)` (every `total//2000`-th row in file
  208 #: order), it is deterministic, and the file it samples is pinned by sha256 in
  209 #: `DOWNLOADS.md`.
  210 STUDY_ROWS = 2000
  211

  213 def _rows():
  214     """A lazy mapping: reading a split costs a parquet decode, so only the splits that are
  215     actually asked for are read. `records()` asks for `self.split` and, here, `train`.
  216 
  217     Reading `train` in full used to be refused outright for cost. With the projection and
  218     the arrow-side sample that is 0.33 s and a few thousand Python dicts, so what remains
  219     is the size of the *pool*, which is why the train side is sampled rather than whole.
  220 
  221     **Why an even spread and not the first 2,000 rows.** TriviaQA's parquet is ordered by
  222     question source, measured: row group 0 (the first 1,000 rows) is 100% `tc_*`, while
  223     rows spread across the file carry `qw`, `qg`, `tc`, `bt`, `odql` and more. A plain
  224     prefix would study one question writer's style and then be scored on a validation
  225     split that has all of them -- a study side that cannot transfer, chosen silently by a
  226     file layout. Measured on the two rules: the first 2,000 rows are 2,000 `tc` questions,
  227     the evenly spread 2,000 carry 8 sources with `sfq` the largest at 25%.
  228     """
  229     return _LazySplits()
  230 
  231 
  232 class _LazySplits:
  233     def __init__(self) -> None:
  234         self._loaded: dict[str, list[dict]] = {}
  235 
  236     def get(self, split, default=None):
  237         if split not in SPLIT_FILES:
  238             return default
  239         if split not in self._loaded:
  240             self._loaded[split] = base.parquet_rows(
  241                 DATASET_ROOT / SPLIT_FILES[split],
  242                 columns=COLUMNS,
  243                 sample=STUDY_ROWS if split == "train" else None,
  244             )
  245         return self._loaded[split]
  246

  248 def _question(row: dict) -> str:

  252 def _gold(row: dict) -> dict:
  253     """The alias set, plus the dataset's own canonical `value`.
  254 
  255     `value` is what the control submits: it is the alias the dataset itself calls the
  256     answer, so it is in the accepted set by construction, and the control therefore tests
  257     the scorer rather than the control.
  258     """
  259     answer = {k: base.normalize_missing(v) for k, v in dict(row.get("answer") or {}).items()}
  260     accepted = get_ground_truths(answer)
  261     value = str(answer.get("value") or "").strip()
  262     if not accepted:
  263         raise ValueError(
  264             "triviaqa: a row has no answer aliases. Measured cause: the `test` split of "
  265             "`rc.nocontext` withholds its gold (`value` is `<unk>`, aliases empty), and "
  266             "scoring it would report the fraction of systems answering `<unk>`. Use "
  267             "`validation`.")
  268     return {"answer": value or accepted[0], "value": value or accepted[0],
  269             # The max must consider every alias; a size cap can discard the only hit.
  270             "accepted": accepted, "accepted_total": len(accepted)}
