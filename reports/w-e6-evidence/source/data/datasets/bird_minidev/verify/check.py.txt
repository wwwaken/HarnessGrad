    1 #!/usr/bin/env python3
    2 """BIRD Mini-Dev's official **EX**, with "cannot measure" kept apart from "wrong".
    3 
    4 Provenance: `bird-bench/mini_dev@abd11b6db92a1c9f809b32f7564c7c71b34d67f0`,
    5 `evaluation/evaluation_ex.py`. `calculate_ex` (lines 18-22) is transcribed **verbatim** below
    6 and the test asserts it is a substring of this file, so drift in upstream is a test failure
    7 rather than a slow divergence. The execution semantics around it are upstream's too --
    8 predicted SQL first, then the gold, `fetchall()` each, set equality, and a timeout or an
    9 error on the *prediction* scoring 0 (`evaluation_ex.py::execute_model`, lines 25-43, which
   10 catches `FunctionTimedOut` and `Exception` and returns `res = 0`).
   11 
   12 **One deliberate difference**, and it is why this file is not `execute_model` wholesale:
   13 upstream catches everything in one `try`, so a **gold** query that fails -- a broken
   14 database, a missing table -- is scored as a wrong answer. This platform treats that as
   15 unmeasurable and writes `measured: false`, because a harness's 0 and an environment's 0 are
   16 otherwise indistinguishable (`docs/dataset-onboarding-skill.md` §5, and the same rule
   17 `data/scaffolds/qa/verify/check.py` implements by exiting 3). A missing or empty `answer.sql` is *not*
   18 in that class: the task ran and the agent did not answer, which is a measured zero.
   19 
   20 The reward is per-question EX. The benchmark's other two official columns -- Soft-F1
   21 (`evaluation_f1.py`) and R-VES (`evaluation_ves.py`) -- are separate metrics over the same
   22 query pair and are **not** reported here; the metric is named in the evidence so a published
   23 number cannot be mistaken for one of them.
   24 """
   25 from __future__ import annotations
   26 
   27 import argparse
   28 import json
   29 import signal
   30 import sqlite3
   31 import sys
   32 from pathlib import Path
   33 
   34 #: Upstream's per-query budget (`evaluation_ex.py`, `--meta_time_out` default 30.0).
   35 TIMEOUT_S = 30.0
   36 
   37 
   38 # --- verbatim: bird-bench/mini_dev@abd11b6 evaluation/evaluation_ex.py lines 18-22 ---------
   39 def calculate_ex(predicted_res, ground_truth_res):
   40     res = 0
   41     if set(predicted_res) == set(ground_truth_res):
   42         res = 1
   43     return res
   44 # -------------------------------------------------------------------------------------------
   45 
   46 
   47 class QueryTimedOut(Exception):
   48     """The query exceeded upstream's 30-second budget."""
   49 
   50 
   51 def _alarm(_signum, _frame):
   52     raise QueryTimedOut(f"exceeded {TIMEOUT_S:g}s")
   53 
   54 
   55 def run_query(connection: sqlite3.Connection, sql: str) -> list[tuple]:
   56     """`execute_sql`'s two lines, with upstream's timeout as a real interrupt."""
   57     previous = signal.signal(signal.SIGALRM, _alarm)
   58     signal.setitimer(signal.ITIMER_REAL, TIMEOUT_S)
   59     try:
   60         cursor = connection.cursor()
   61         cursor.execute(sql)
   62         return cursor.fetchall()
   63     finally:
   64         signal.setitimer(signal.ITIMER_REAL, 0)
   65         signal.signal(signal.SIGALRM, previous)
   66 
   67 
   68 def _write(path: str, payload: str) -> None:
   69     out = Path(path)
   70     out.parent.mkdir(parents=True, exist_ok=True)
   71     out.write_text(payload, encoding="utf-8")
   72 
   73 
   74 def _measure(args, reward: float, why: str) -> int:
   75     _write(args.reward_file, f"{reward:g}\n")
   76     _write(args.evidence_file, json.dumps(
   77         {"measured": True, "metric": "EX", "reward": reward, "why": why}, sort_keys=True) + "\n")
   78     print(f"reward {reward:g}: {why}")
   79     return 0
   80 
   81 
   82 def _cannot_measure(args, why: str) -> int:
   83     """No reward file at all: `eval/verification/verifier.py` turns this into an `invalid` task."""
   84     _write(args.evidence_file, json.dumps(
   85         {"measured": False, "metric": "EX", "error": why}, sort_keys=True) + "\n")
   86     print(f"check cannot measure: {why}", file=sys.stderr)
   87     return 3
   88 
   89 
   90 def main() -> int:
   91     parser = argparse.ArgumentParser()
   92     parser.add_argument("--db", required=True)
   93     parser.add_argument("--gold", required=True)
   94     parser.add_argument("--pred", required=True)
   95     parser.add_argument("--reward-file", default="verifier/reward.txt")
   96     parser.add_argument("--evidence-file", default="verifier/evidence.json")
   97     args = parser.parse_args()
   98 
   99     gold_sql = Path(args.gold).read_text(encoding="utf-8").strip() if Path(args.gold).is_file() else ""
  100     if not Path(args.db).is_file():
  101         return _cannot_measure(args, f"no database at {args.db}: this task cannot be graded")
  102     if not gold_sql:
  103         return _cannot_measure(args, "the gold query is empty: this task cannot be graded")
  104 
  105     connection = sqlite3.connect(args.db)
  106     try:
  107         try:
  108             gold_rows = run_query(connection, gold_sql)
  109         except Exception as exc:                    # noqa: BLE001 - a broken gold is not the agent's
  110             return _cannot_measure(
  111                 args, f"the gold query cannot run against {Path(args.db).name} "
  112                       f"({type(exc).__name__}: {exc}): the environment cannot measure this task")
  113 
  114         prediction = (Path(args.pred).read_text(encoding="utf-8").strip()
  115                       if Path(args.pred).is_file() else "")
  116         if not prediction:
  117             return _measure(args, 0.0, "no answer (measured zero): answer.sql is missing or empty")
  118         try:
  119             predicted_rows = run_query(connection, prediction)
  120         except Exception as exc:                    # noqa: BLE001 - upstream scores this 0
  121             return _measure(args, 0.0, f"prediction did not run ({type(exc).__name__}: {exc})")
  122     finally:
  123         connection.close()
  124 
  125     reward = float(calculate_ex(predicted_rows, gold_rows))
  126     return _measure(args, reward, f"exact set equality: {len(predicted_rows)} predicted vs "
  127                                   f"{len(gold_rows)} gold rows")
  128 
  129 
  130 if __name__ == "__main__":
  131     raise SystemExit(main())
