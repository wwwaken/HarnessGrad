    1 #!/usr/bin/env python3
    2 """GAIA's official scorer, **called** rather than transcribed.
    3 
    4 `scorer.py` arrives as a `VERIFY.inputs` entry (`dst: /scorer.py`) copied out of `_src/`, where it
    5 was pinned by sha256 from the official leaderboard Space. This script therefore runs upstream's
    6 own `question_scorer` -- the same call the leaderboard makes (`app.py`:
    7 `question_scorer(task['model_answer'], gold_results[...]["Final answer"])`) -- and adds only the
    8 platform's reward file, the placeholder guard and the evidence.
    9 
   10 Why calling matters here more than usual: the scorer has **three branches that normalize
   11 differently** -- a float (strip `$`, `%`, `,` then convert), a comma/semicolon list (compare
   12 element-wise, keeping punctuation), and a string (remove whitespace, lowercase, strip punctuation).
   13 A hand-written "exact match" agrees with it on the easy third of the pool and silently disagrees on
   14 the rest, which is the failure mode this repository's gate 3 exists to prevent.
   15 
   16 Two outcomes that must not look alike
   17 -------------------------------------
   18 * a missing or empty `answer.txt` is a **measured zero** -- the task ran and the agent did not
   19   answer;
   20 * a gold that is empty or the test split's placeholder `"?"` is **unmeasurable** (`measured: false`,
   21   exit 3, no reward file). That is not hypothetical: 301 of the release's 466 rows carry `"?"`, and
   22   a scorer run over them would compare a submission against a placeholder and report a number.
   23 """
   24 from __future__ import annotations
   25 
   26 import argparse
   27 import contextlib
   28 import io
   29 import json
   30 import sys
   31 from pathlib import Path
   32 
   33 #: The scorer is copied to `/` beside this file, so the script's own directory is enough.
   34 sys.path.insert(0, str(Path(__file__).resolve().parent))
   35 
   36 WORKDIR = Path("/app")
   37 DEFAULT_ANSWER = str(WORKDIR / "answer.txt")
   38 
   39 
   40 def _write(path: str, payload: str) -> None:
   41     out = Path(path)
   42     out.parent.mkdir(parents=True, exist_ok=True)
   43     out.write_text(payload, encoding="utf-8")
   44 
   45 
   46 def _measure(args, reward: float, why: str, extra: dict) -> int:
   47     _write(args.reward_file, f"{reward:g}\n")
   48     evidence = {"measured": True, "metric": "GAIA exact match (official question_scorer)",
   49                 "reward": reward, "why": why}
   50     evidence.update(extra)
   51     _write(args.evidence_file, json.dumps(evidence, sort_keys=True) + "\n")
   52     print(f"reward {reward:g}: {why}")
   53     return 0
   54 
   55 
   56 def _cannot_measure(args, why: str) -> int:
   57     _write(args.evidence_file, json.dumps(
   58         {"measured": False, "metric": "GAIA exact match", "error": why}, sort_keys=True) + "\n")
   59     print(f"check cannot measure: {why}", file=sys.stderr)
   60     return 3
   61 
   62 
   63 def main() -> int:
   64     parser = argparse.ArgumentParser()
   65     parser.add_argument("--answer", default=DEFAULT_ANSWER)
   66     parser.add_argument("--gold", required=True)
   67     parser.add_argument("--reward-file", default="/verifier/reward.txt")
   68     parser.add_argument("--evidence-file", default="/verifier/evidence.json")
   69     args = parser.parse_args()
   70 
   71     try:
   72         from scorer import question_scorer
   73     except Exception as exc:                       # noqa: BLE001 - the environment, not the agent
   74         return _cannot_measure(args, f"the pinned scorer will not import: "
   75                                      f"{type(exc).__name__}: {exc}")
   76 
   77     gold_path = Path(args.gold)
   78     if not gold_path.is_file():
   79         return _cannot_measure(args, f"no gold at {gold_path}")
   80     try:
   81         record = json.loads(gold_path.read_text(encoding="utf-8"))
   82         gold = str(record["gold"])
   83     except Exception as exc:                       # noqa: BLE001
   84         return _cannot_measure(args, f"the gold is not usable: {type(exc).__name__}: {exc}")
   85     if gold.strip() in ("", "?"):
   86         return _cannot_measure(
   87             args, f"gold is the placeholder {gold!r} (the test split's sentinel): this task "
   88                   f"cannot be scored")
   89 
   90     answer_path = Path(args.answer)
   91     answer = answer_path.read_text(encoding="utf-8").strip() if answer_path.is_file() else ""
   92     if not answer:
   93         return _measure(args, 0.0, "no answer (measured zero): answer.txt is missing or empty",
   94                         {"gold": gold, "task_id": record.get("task_id")})
   95 
   96     transcript = io.StringIO()
   97     try:
   98         with contextlib.redirect_stdout(transcript):
   99             correct = bool(question_scorer(answer, gold))
  100     except Exception as exc:                       # noqa: BLE001 - a scorer crash is not a wrong answer
  101         return _cannot_measure(args, f"question_scorer raised {type(exc).__name__}: {exc}")
  102 
  103     return _measure(args, 1.0 if correct else 0.0,
  104                     "exact match after the official normalization" if correct
  105                     else "no match after the official normalization",
  106                     {"answer": answer, "gold": gold, "task_id": record.get("task_id"),
  107                      "scorer_said": transcript.getvalue().strip()[:500]})
  108 
  109 
  110 if __name__ == "__main__":
  111     raise SystemExit(main())
