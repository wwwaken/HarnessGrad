   44 def normalize_answer(s):
   45     """Lower text and remove punctuation, articles and extra whitespace."""
   46     def remove_articles(text):
   47         return re.sub(r"\b(a|an|the)\b", " ", text, flags=re.UNICODE)
   48     def remove_punc(text):
   49         return "".join(ch for ch in text if ch not in set(string.punctuation))
   50     return " ".join(remove_articles(remove_punc(s.lower())).split())

   53 def compute_exact(a_gold, a_pred):
   54     return int(normalize_answer(a_gold) == normalize_answer(a_pred))
   55 
   56 
   57 def compute_f1(a_gold, a_pred):
   58     gold_toks = normalize_answer(a_gold).split() if a_gold else []
   59     pred_toks = normalize_answer(a_pred).split() if a_pred else []
   60     num_same = sum((collections.Counter(gold_toks) & collections.Counter(pred_toks)).values())
   61     if len(gold_toks) == 0 or len(pred_toks) == 0:
   62         return int(gold_toks == pred_toks)
   63     if num_same == 0:
   64         return 0
   65     precision = num_same / len(pred_toks)
   66     recall = num_same / len(gold_toks)
   67     return (2 * precision * recall) / (precision + recall)

   70 def _check_mode(mode: str) -> None:
   71     if mode not in (DEFAULT_MODE, LEGACY_MODE):
   72         raise ValueError("MuSiQue mode must be answer_support or legacy_answer_only")

   75 def compute_support(predicted_indices: list[int], gold_indices: list[int]) -> dict:
   76     """Pinned metrics/support.py::SupportMetric, including the both-empty case."""
   77     predicted, expected = set(predicted_indices), set(gold_indices)
   78     tp = len(predicted & expected)
   79     fp, fn = len(predicted - expected), len(expected - predicted)
   80     precision = tp / (tp + fp) if tp + fp else 0.0
   81     recall = tp / (tp + fn) if tp + fn else 0.0
   82     f1 = 2 * precision * recall / (precision + recall) if precision + recall else 0.0
   83     if not predicted and not expected:
   84         f1 = 1.0
   85     return {"f1": f1, "em": int(fp + fn == 0)}

   88 def _score(answer: str, evidence: list | None, gold: dict, mode: str) -> dict:
   89     _check_mode(mode)
   90     answers = [gold["answer"], *gold["aliases"]]
   91     f1 = max(compute_f1(reference, answer) for reference in answers)
   92     em = max(compute_exact(reference, answer) for reference in answers)
   93     official = {"metric": "MuSiQue-Ans", "answer_f1": f1, "answer_em": em,
   94                 "id": gold["id"], "mode": mode,
   95                 "aggregate_policy": "ordered macro mean; official columns rounded to 3 decimals"}
   96     if mode == LEGACY_MODE:
   97         official.update(metric="MuSiQue-Ans legacy answer-only (support unmeasured)",
   98                         support_f1=None, reward_policy="legacy answer_f1 only")
   99         return {"reward": float(f1), "correct": bool(em), "official": official,
  100                 "why": f"legacy_answer_only: answer_f1={f1:g}, answer_em={em}; support unmeasured"}
  101     if "supporting_facts" not in gold or not isinstance(gold["supporting_facts"], list):
  102         raise ValueError("MuSiQue trusted gold is missing supporting_facts")
  103     # Titles describe public citations; upstream support scoring uses paragraph idx only.
  104     support = compute_support([pair[1] for pair in (evidence or [])],
  105                               [pair[1] for pair in gold["supporting_facts"]])
  106     joint_f1, joint_em = f1 * support["f1"], em * support["em"]
  107     official.update(support_f1=support["f1"], support_em=support["em"],
  108                     joint_f1=joint_f1, joint_em=joint_em,
  109                     reward_policy="platform answer_f1 * support_f1 (not an official metric)",
  110                     joint_em_policy="platform answer_em * support_em (not an official metric)")
  111     return {"reward": float(joint_f1), "correct": bool(joint_em), "official": official,
  112             "why": f"answer_f1={f1:g}, support_f1={support['f1']:g}, answer_em={em}; "
  113                    f"platform joint_f1={joint_f1:g}"}

  130 def aggregate(results: list[dict]) -> dict:
  131     """Match upstream ordered accumulation/rounding for the three official columns.
  132 
  133     The platform reward is the unrounded mean of per-item rewards, never the product
  134     of corpus means. Mixing legacy and default results is refused.
  135     The declared denominator is the provided results: this helper has no gold pool
  136     to detect absent rows. Upstream requires complete, identically ordered IDs.
  137     """
  138     if not results:
  139         raise ValueError("cannot aggregate an empty MuSiQue result list")
  140     from data.scaffolds.aggregation import aggregation_denominator
  141 
  142     denominator = aggregation_denominator("musique", "corpus_metrics",
  143                                           provided_results=len(results))
  144     modes = {result["official"]["mode"] for result in results}
  145     if len(modes) != 1:
  146         raise ValueError("cannot mix MuSiQue scoring modes")
  147     mode = modes.pop()
  148     _check_mode(mode)
  149     output = {"mode": mode, "count": len(results),
  150               "reward_policy": results[0]["official"]["reward_policy"],
  151               "aggregate_policy": "ordered macro mean; official columns rounded to 3 decimals"}
  152     fields = ["answer_f1", "answer_em"]
  153     if mode == DEFAULT_MODE:
  154         fields += ["support_f1", "joint_f1", "joint_em"]
  155     else:
  156         output["support_f1"] = None
  157     for field in [*fields, "reward"]:
  158         total = 0.0
  159         for result in results:
  160             total += result["reward"] if field == "reward" else result["official"][field]
  161         mean = total / denominator
  162         output[field] = round(mean, 3) if field in ("answer_f1", "answer_em", "support_f1") else mean
  163     return output

  166 def _rows():
  167     with (DATASET_ROOT / DATA_FILE).open(encoding="utf-8") as handle:
  168         rows = [json.loads(line) for line in handle if line.strip()]
  169     if len(rows) != 2417 or len({r["id"] for r in rows}) != 2417:
  170         raise ValueError("expected exactly 2417 unique MuSiQue-Ans dev questions")
  171     if not all(r["answerable"] and r["answer"] for r in rows):
  172         raise ValueError("this adapter requires answerable MuSiQue-Ans rows with gold")
  173     return {"dev": rows}

  176 def _question(row):
  177     context = "\n\n".join(f"[{p['idx']}] {p['title']}\n{p['paragraph_text']}"
  178                           for p in row["paragraphs"])
