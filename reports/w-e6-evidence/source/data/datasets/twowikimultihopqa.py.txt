  124 def normalize_answer(s):
  125 
  126     def remove_articles(text):
  127         return re.sub(r'\b(a|an|the)\b', ' ', text)
  128 
  129     def white_space_fix(text):
  130         return ' '.join(text.split())
  131 
  132     def remove_punc(text):
  133         exclude = set(string.punctuation)
  134         return ''.join(ch for ch in text if ch not in exclude)
  135 
  136     def lower(text):
  137         return text.lower()
  138 
  139     return white_space_fix(remove_articles(remove_punc(lower(s))))

  142 def f1_score(prediction, ground_truth):
  143     normalized_prediction = normalize_answer(prediction)
  144     normalized_ground_truth = normalize_answer(ground_truth)
  145 
  146     ZERO_METRIC = (0, 0, 0)
  147 
  148     if normalized_prediction in ['yes', 'no', 'noanswer'] and normalized_prediction != normalized_ground_truth:
  149         return ZERO_METRIC
  150     if normalized_ground_truth in ['yes', 'no', 'noanswer'] and normalized_prediction != normalized_ground_truth:
  151         return ZERO_METRIC
  152 
  153     prediction_tokens = normalized_prediction.split()
  154     ground_truth_tokens = normalized_ground_truth.split()
  155     common = Counter(prediction_tokens) & Counter(ground_truth_tokens)
  156     num_same = sum(common.values())
  157     if num_same == 0:
  158         return ZERO_METRIC
  159     precision = 1.0 * num_same / len(prediction_tokens)
  160     recall = 1.0 * num_same / len(ground_truth_tokens)
  161     f1 = (2 * precision * recall) / (precision + recall)
  162     return f1, precision, recall

  175 def update_answer(metrics, prediction, golds):
  176     max_em, max_f1, max_prec, max_recall = 0, 0, 0, 0
  177 
  178     for gold in golds:
  179         em, f1, prec, recall = eval_answer(prediction, gold)
  180 
  181         max_em = max(max_em, em)
  182         max_f1 = max(max_f1, f1)
  183         max_prec = max(max_prec, prec)
  184         max_recall = max(max_recall, recall)
  185 
  186     metrics['em'] += float(max_em)
  187     metrics['f1'] += max_f1
  188     metrics['prec'] += max_prec
  189     metrics['recall'] += max_recall
  190 
  191     return max_em, max_prec, max_recall

  194 def normalize_sp(sps):
  195     new_sps = []
  196     for sp in sps:
  197         sp = list(sp)
  198         sp[0] = sp[0].lower()
  199         new_sps.append(sp)
  200     return new_sps

  203 def update_sp(metrics, prediction, gold):
  204     cur_sp_pred = normalize_sp(set(map(tuple, prediction)))
  205     gold_sp_pred = normalize_sp(set(map(tuple, gold)))
  206     tp, fp, fn = 0, 0, 0
  207     for e in cur_sp_pred:
  208         if e in gold_sp_pred:
  209             tp += 1
  210         else:
  211             fp += 1
  212     for e in gold_sp_pred:
  213         if e not in cur_sp_pred:
  214             fn += 1
  215     prec = 1.0 * tp / (tp + fp) if tp + fp > 0 else 0.0
  216     recall = 1.0 * tp / (tp + fn) if tp + fn > 0 else 0.0
  217     f1 = 2 * prec * recall / (prec + recall) if prec + recall > 0 else 0.0
  218     em = 1.0 if fp + fn == 0 else 0.0
  219     metrics['sp_em'] += em
  220     metrics['sp_f1'] += f1
  221     metrics['sp_prec'] += prec
  222     metrics['sp_recall'] += recall
  223     return em, prec, recall

  226 def normalize_evi(evidences):
  227 
  228     def white_space_fix(text):
  229         return ' '.join(text.split())
  230 
  231     def remove_punc(text):
  232         exclude = set(string.punctuation)
  233         return ''.join(ch for ch in text if ch not in exclude)
  234 
  235     def lower(text):
  236         return text.lower()
  237 
  238     def recurse(arr):
  239         for i in range(len(arr)):
  240             if isinstance(arr[i], str):
  241                 arr[i] = white_space_fix(remove_punc(lower(arr[i])))
  242             else:
  243                 recurse(arr[i])
  244 
  245     recurse(evidences)
  246 
  247     return evidences
  248

  250 def update_evi(metrics, prediction, gold):
  251     prediction_normalize = normalize_evi(prediction)
  252     gold_normalize = normalize_evi(gold)
  253     #
  254     cur_evi_pred = set(map(tuple, prediction_normalize))
  255     gold_evi_pred = list(map(lambda e: set(map(tuple, e)), gold_normalize))
  256     #
  257     num_matches = 0
  258     num_preds = len(cur_evi_pred)
  259     num_golds = len(gold_evi_pred)
  260 
  261     for pred_evidence in cur_evi_pred:
  262         for gold_evidences in gold_evi_pred:
  263             if pred_evidence in gold_evidences:
  264                 num_matches += 1
  265                 break
  266 
  267     prec = num_preds and num_matches / num_preds
  268     recall = num_golds and num_matches / num_golds
  269     f1 = 2 * prec * recall / (prec + recall) if prec + recall > 0 else 0.0
  270     em = 1.0 if num_matches == num_preds == num_golds else 0.0
  271     
  272     metrics['evi_em'] += em
  273     metrics['evi_f1'] += f1
  274     metrics['evi_prec'] += prec
  275     metrics['evi_recall'] += recall
  276 
  277     return em, prec, recall

  280 def eval(prediction_file, gold_file, alias_file):
  281     aliases = {}
  282 
  283     with open(prediction_file) as f:
  284         prediction = json.load(f)
  285     with open(gold_file) as f:
  286         gold = json.load(f)
  287     with open(alias_file) as f:
  288         for json_line in map(json.loads, f):
  289             aliases[json_line["Q_id"]] = {
  290                 "aliases": set(json_line["aliases"] + json_line["demonyms"])
  291             }
  292 
  293     metrics = {'em': 0, 'f1': 0, 'prec': 0, 'recall': 0,
  294         'sp_em': 0, 'sp_f1': 0, 'sp_prec': 0, 'sp_recall': 0,
  295         'evi_em': 0, 'evi_f1': 0, 'evi_prec': 0, 'evi_recall': 0,
  296         'joint_em': 0, 'joint_f1': 0, 'joint_prec': 0, 'joint_recall': 0}
  297 
  298     for dp in gold:
  299         cur_id = dp['_id']
  300         can_eval_joint = True
  301         # answer prediction task
  302         if cur_id not in prediction['answer']:
  303             print('missing answer {}'.format(cur_id))
  304             can_eval_joint = False
  305         else:
  306             gold_answers = {dp['answer']}  # Gold span
  307 
  308             if dp['answer_id'] in aliases and aliases[dp['answer_id']]["aliases"]:
  309                 gold_answers.update(aliases[dp['answer_id']]["aliases"])
  310 
  311             em, prec, recall = update_answer(
  312                 metrics, prediction['answer'][cur_id], gold_answers)
  313         # sentence-level supporting facts prediction task
  314         if cur_id not in prediction['sp']:
  315             print('missing sp fact {}'.format(cur_id))
  316             can_eval_joint = False
  317         else:
  318             sp_em, sp_prec, sp_recall = update_sp(
  319                 metrics, prediction['sp'][cur_id], dp['supporting_facts'])
  320         # evidence generation task
  321         if cur_id not in prediction['evidence']:
  322             print('missing evidence {}'.format(cur_id))
  323             can_eval_joint = False
  324         else:
  325             gold_evidences = []
  326 
  327             for evidence_idx, (sub_str, rel_str, obj_str) in enumerate(dp['evidences']):
  328                 sub_strs = {sub_str}
  329                 obj_strs = {obj_str}
  330 
  331                 if dp['evidences_id'] != []:
  332                     #
  333                     assert len(dp['evidences_id']) == len(dp['evidences'])
  334                     sub_id, rel_id, obj_id = dp['evidences_id'][evidence_idx]
  335                     
  336                     assert rel_id == rel_str
  337 
  338                     if sub_id in aliases:
  339                         sub_strs.update(aliases[sub_id]["aliases"])
  340                     if obj_id in aliases:
  341                         obj_strs.update(aliases[obj_id]["aliases"])
  342                     
  343                 gold_evidence = []
  344 
  345                 for sub_str, obj_str in itertools.product(sub_strs, obj_strs):
  346                     gold_evidence.append([sub_str, rel_str, obj_str])
  347 
  348                 gold_evidences.append(gold_evidence)
  349 
  350             evi_em, evi_prec, evi_recall = update_evi(
  351                 metrics, prediction['evidence'][cur_id], gold_evidences)
  352 
  353         if can_eval_joint:
  354             joint_prec = prec * sp_prec * evi_prec
  355             joint_recall = recall * sp_recall * evi_recall
  356             #
  357             if joint_prec + joint_recall > 0:
  358                 joint_f1 = 2 * joint_prec * joint_recall / (joint_prec + joint_recall)
  359             else:
  360                 joint_f1 = 0.
  361             joint_em = em * sp_em * evi_em
  362 
  363             metrics['joint_em'] += joint_em
  364             metrics['joint_f1'] += joint_f1
  365             metrics['joint_prec'] += joint_prec
  366             metrics['joint_recall'] += joint_recall
  367 
  368     N = len(gold)
  369 
  370     for k in metrics.keys():
  371         metrics[k] = round(metrics[k] / N * 100, 2)
  372 
  373     print(json.dumps(metrics, indent=4))

  379 def _one_question(prediction: str, golds) -> dict:
  380     """Upstream's `update_answer`, applied to one question and a fresh accumulator.
  381 
  382     The per-question metric *is* this call: `update_answer` writes `metrics['f1'] +=
  383     max_f1` and returns the other three, so a dict seeded at zero comes back holding this
  384     question's answers and nothing else. Re-deriving the max here instead would be a second
  385     implementation of the metric, which is the thing the drift test exists to prevent.
  386     """
  387     metrics = {"em": 0.0, "f1": 0.0, "prec": 0.0, "recall": 0.0}
  388     update_answer(metrics, prediction, golds)
  389     return metrics

  404     return {

  411                      "id": gold.get("id"), "type": gold.get("type")},
  412         "why": (f"legacy answer-only: Ans F1={metrics['f1']:g}, Ans EM={metrics['em']:g}; "
  413                 "Sup/Evi/Joint are not reported in this mode"),

  418                     for metric in ("em", "f1", "prec", "recall"))

  422 def validate_prediction(prediction: dict) -> None:
  423     """Validate the adapter's public JSON contract, including partial predictions.
  424 
  425     Omitted channels are different from submitted empty values. Unknown fields and
  426     malformed channels are contract errors rather than invented scorer zeroes.
  427     """
  428     if not isinstance(prediction, dict) or set(prediction) - PREDICTION_KEYS:
  429         raise ValueError("2Wiki prediction must be an object with only public prediction fields")
  430     if type(prediction.get("schema_version")) is not int or prediction["schema_version"] != 1:
  431         raise ValueError("2Wiki prediction requires schema_version=1")
  432     if "answer" in prediction and not isinstance(prediction["answer"], str):
  433         raise ValueError("answer must be a string")
  434     for channel, length in (("supporting_facts", 2), ("evidence", 3)):
  435         if channel not in prediction:
  436             continue
  437         if not isinstance(prediction[channel], list):
  438             raise ValueError(f"{channel} must be a list")
  439         for item in prediction[channel]:
  440             if not isinstance(item, list) or len(item) != length:
  441                 raise ValueError(f"{channel} entries must be lists of length {length}")
  442             if channel == "supporting_facts":
  443                 valid = (isinstance(item[0], str) and type(item[1]) is int and item[1] >= 0)
  444             else:
  445                 valid = all(isinstance(part, str) for part in item)
  446             if not valid:
  447                 raise ValueError(f"invalid {channel} entry")

  450 def _prediction_metrics(prediction: dict, gold: dict) -> dict:
  451     validate_prediction(prediction)
  452     if not isinstance(gold.get("answer"), str):
  453         raise ValueError("gold has no answer")
  454     if "supporting_facts" not in gold or "evidences" not in gold:
  455         raise ValueError("structured scoring requires support and evidence gold")
  456     metrics = dict.fromkeys(METRIC_KEYS, 0.0)
  457     if "answer" in prediction:
  458         em, prec, recall = update_answer(
  459             metrics, prediction["answer"], {gold["answer"], *gold.get("aliases", [])})
  460     if "supporting_facts" in prediction:
  461         sp_em, sp_prec, sp_recall = update_sp(
  462             metrics, prediction["supporting_facts"], gold["supporting_facts"])
  463     if "evidence" in prediction:
  464         variants = _evidence_variants(gold)
  465         # Upstream normalizes evidence in place; variants are fresh and predictions
  466         # must be copied so a verifier does not mutate its inputs.
  467         evi_em, evi_prec, evi_recall = update_evi(
  468             metrics, deepcopy(prediction["evidence"]), variants)
  469     if all(channel in prediction for channel in ("answer", "supporting_facts", "evidence")):
  470         joint_prec = prec * sp_prec * evi_prec
  471         joint_recall = recall * sp_recall * evi_recall
  472         metrics.update(
  473             joint_em=float(em * sp_em * evi_em), joint_prec=joint_prec,
  474             joint_recall=joint_recall,
  475             joint_f1=(2 * joint_prec * joint_recall / (joint_prec + joint_recall)
  476                       if joint_prec + joint_recall > 0 else 0.0))
  477     return metrics

  490 def score(prediction: str | dict, gold: dict) -> dict:
  491     """Platform entry point: JSON in answer.txt, or labeled legacy answer text."""

  499 def evaluate_predictions(predictions: dict[str, dict], golds: list[dict]) -> dict:
  500     """Official percentage aggregate over every gold, in gold order.
  501 
  502     Extra prediction ids are ignored. Missing rows/channels contribute zero without
  503     shrinking the denominator. Golds are the trusted inline records produced by _gold.
  504     """
  505     if not golds:
  506         raise ValueError("cannot evaluate an empty gold pool")
  507     from data.scaffolds.aggregation import aggregation_denominator
  508 
  509     denominator = aggregation_denominator("twowikimultihopqa", "corpus_metrics",
  510                                           gold_rows=len(golds))
  511     totals = dict.fromkeys(METRIC_KEYS, 0.0)
  512     for gold in golds:
  513         metrics = _prediction_metrics(
  514             predictions.get(gold["id"], {"schema_version": 1}), gold)
  515         for key in totals:
  516             totals[key] += metrics[key]
  517     return {key: round(value / denominator * 100, 2) for key, value in totals.items()}

  523 def _aliases() -> dict[str, list[str]]:
  524     """`{Q-id: [alias, ...]}`, from the official alias table. Loaded once.
  525 
  526     Only the **pool's** half of the platform reads this: the gold travels to the verifier
  527     with the aliases inline, so the check in the verification namespace never opens a file
  528     under `DATASET_ROOT` (it cannot see one; `docs/dataset-onboarding-skill.md` §16).
  529     """
  530     global _ALIASES
  531     if _ALIASES is None:
  532         table: dict[str, list[str]] = {}
  533         with (DATASET_ROOT / ALIAS_FILE).open(encoding="utf-8") as handle:
  534             for line in handle:
  535                 if not line.strip():
  536                     continue
  537                 entry = json.loads(line)
  538                 # Sorted and de-duplicated so the gold a task carries is a function of the
  539                 # row, not of the file's line order.
  540                 table[entry["Q_id"]] = sorted(set(entry["aliases"]) | set(entry["demonyms"]))
  541         _ALIASES = table
  542     return _ALIASES

  545 def _rows() -> dict[str, list[dict]]:
  546     with (DATASET_ROOT / DATA_FILE).open(encoding="utf-8") as handle:
  547         rows = json.load(handle)
  548     if len(rows) != EXPECTED_ROWS or len({row["_id"] for row in rows}) != EXPECTED_ROWS:
  549         raise ValueError(f"expected exactly {EXPECTED_ROWS} unique 2WikiMultihopQA dev rows")
  550     if not all(row["answer"] and row.get("context") for row in rows):
  551         raise ValueError("this adapter requires dev rows with an answer and a context")
  552     return {"dev": rows}

  555 def _question(row: dict) -> str:
  556     """The question plus the ten released paragraphs, in the released order.
  557 
  558     Nothing else: `supporting_facts`, `evidences`, `evidences_id`, `answer_id`,
  559     `entity_ids` and `type` are annotations, and a prompt that carried them would grade
  560     the agent's reading of a label rather than of the paragraphs.
  561     """
  562     context = "\n\n".join(
  563         f"{title}\n" + "\n".join(f"[{index}] {sentence}"
  564                                 for index, sentence in enumerate(sentences))
  565         for title, sentences in row["context"])
  566     return f"Question: {row['question']}\n\nContext paragraphs:\n{context}"

  569 def _gold(row: dict) -> dict:
  570     aliases = _aliases()
  571     evidence_ids = row["evidences_id"]
  572     if evidence_ids and len(evidence_ids) != len(row["evidences"]):
  573         raise ValueError("evidences_id and evidences lengths differ")
  574     evidence_aliases = {}
  575     for (_, relation, _), (subject_id, relation_id, object_id) in zip(
  576             row["evidences"], evidence_ids):
  577         if relation_id != relation:
  578             raise ValueError("evidence relation does not match evidences_id")
  579         for entity_id in (subject_id, object_id):
  580             if entity_id in aliases:
  581                 evidence_aliases[entity_id] = aliases[entity_id]
  582     return {"answer": row["answer"],
  583             "aliases": aliases.get(row.get("answer_id"), []),
  584             "supporting_facts": deepcopy(row["supporting_facts"]),
  585             "evidences": deepcopy(row["evidences"]),
  586             "evidences_id": deepcopy(evidence_ids), "evidence_aliases": evidence_aliases,
  587             "id": row["_id"], "type": row["type"]}

  590 def _evidence_variants(gold: dict) -> list:
  591     """Upstream's subject/object alias product, from trusted inline aliases only.
  592 
  593     Expand at scoring time rather than storing every product for the whole task pool.
  594     The verifier never loads the global alias table or any dataset payload.
  595     """
  596     evidence_ids = gold.get("evidences_id", [])
  597     aliases = gold.get("evidence_aliases", {})
  598     if evidence_ids and len(evidence_ids) != len(gold["evidences"]):
  599         raise ValueError("evidences_id and evidences lengths differ")
  600     variants = []
  601     for index, (subject, relation, obj) in enumerate(gold["evidences"]):
  602         subjects, objects = {subject}, {obj}
  603         if evidence_ids:
  604             subject_id, relation_id, object_id = evidence_ids[index]
  605             if relation_id != relation:
  606                 raise ValueError("evidence relation does not match evidences_id")
  607             subjects.update(aliases.get(subject_id, []))
  608             objects.update(aliases.get(object_id, []))
  609         variants.append([[sub, relation, ob] for sub, ob in
  610                          itertools.product(sorted(subjects), sorted(objects))])
  611     return variants
