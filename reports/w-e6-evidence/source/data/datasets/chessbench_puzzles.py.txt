    1 """ChessBench's 10,000-puzzle tactical slice.
    2 
    3 Source: `google-deepmind/searchless_chess`, ChessBench paper (NeurIPS 2024,
    4 arXiv:2402.04494), `puzzles.csv`.
    5 
    6 Official puzzle accuracy uses interactive stepwise play: after each solver move the
    7 trusted evaluator applies the reference opponent response and presents the updated board.
    8 `StepwisePuzzle` implements that public board / single-UCI-answer contract without exposing
    9 the reference line. It belongs in a trusted controller, never in the actor's filesystem.
   10 
   11 The existing QA task still accepts a space-separated sequence. Its reward is explicitly
   12 the offline strict replay variant, not interactive official accuracy or Elo. Every applied
   13 move is checked for legality, including setup and reference opponent moves. Any legal
   14 immediate mate succeeds and ends play; unused trailing answers are ignored. The pinned
   15 upstream replay, including its unchecked Board.push behavior, is retained separately as a
   16 diagnostic in `official.upstream`; it does not determine reward. Importing this module
   17 does not read the puzzle CSV.
   18 """

   47 def _read_rows() -> list[dict]:

   55 def _presented_fen(row: dict) -> str:
   56     moves = row["Moves"].split()
   57     if not moves:
   58         raise ValueError(f"{row['PuzzleId']}: empty Moves")
   59     board = chess.Board(row["FEN"])
   60     board.push_uci(moves[0])
   61     return board.fen()

   78 def _selected_records(task_ids: list[str]) -> list[dict]:
   79     """Read through the last selected CSV index; decode only selected puzzles.
   80 
   81     Validate the entire selection before opening the CSV. Keep this separate from
   82     QASpec's full-pool cache so a targeted control cannot change later pool loading.
   83     """
   84     ids = list(dict.fromkeys(task_ids))
   85     if not ids:
   86         raise ValueError("ChessBench selection must not be empty")
   87     indices = {}
   88     for task_id in ids:
   89         match = _TASK_ID_RE.fullmatch(task_id)
   90         if match is None or int(match[1]) >= 10000:
   91             raise ValueError(f"unknown ChessBench task id {task_id!r}")
   92         indices[int(match[1])] = task_id
   93     remaining = set(indices)
   94     records = {}
   95     with CSV_PATH.open(newline="", encoding="utf-8") as handle:
   96         for index, raw in enumerate(csv.DictReader(handle)):
   97             if index not in remaining:
   98                 continue
   99             row = _row(raw)
  100             task_id = indices[index]
  101             records[task_id] = {
  102                 "task_id": task_id,
  103                 "index": index,
  104                 "split": "test",
  105                 "row": row,
  106                 "question": _question(row),
  107                 "gold": _gold(row),
  108             }
  109             remaining.remove(index)
  110             if not remaining:
  111                 break
  112     if remaining:
  113         raise ValueError(f"ChessBench CSV is missing selected indices {sorted(remaining)}")
  114     return [records[task_id] for task_id in ids]

  149 def _official_replay(board: chess.Board, moves: list[str], predicted: list[str]) -> bool:
  150     """The official `evaluate_puzzle_from_board` loop, with engine output supplied by us.
  151 
  152     Upstream source: `src/puzzles.py`, lines 74-94 at the pinned repository revision.
  153     `predicted` replaces `engine.play(board).uci()` so the dataset answer contract can carry
  154     a complete scripted sequence instead of a live engine object.
  155     """
  156     solver_index = 0
  157     for move_idx, move in enumerate(moves):
  158         # According to the Lichess puzzle format, index 0 is the opponent setup move and
  159         # index 1 is the first move presented to the player.
  160         if move_idx % 2 == 1:
  161             if solver_index >= len(predicted):
  162                 return False
  163             predicted_move = predicted[solver_index]
  164             solver_index += 1
  165             if move != predicted_move:
  166                 try:
  167                     board.push(chess.Move.from_uci(predicted_move))
  168                 except ValueError:
  169                     return False
  170                 # Lichess accepts any mate-in-one when it differs from the stored line.
  171                 return board.is_checkmate()
  172             try:
  173                 board.push(chess.Move.from_uci(predicted_move))
  174             except ValueError:
  175                 return False
  176         else:
  177             try:
  178                 board.push(chess.Move.from_uci(move))
  179             except ValueError:
  180                 return False
  181     # Like an Engine, only requested moves count; trailing scripted moves are unused.
  182     return True

  185 def score_upstream(prediction: str, gold: dict) -> dict:
  186     """Diagnostic only: preserve the pinned upstream loop's unchecked-push semantics."""
  187     try:
  188         moves = list(gold["moves"])
  189         hit = bool(moves) and _official_replay(
  190             chess.Board(gold["fen"]), moves, prediction.strip().split())
  191     except (ValueError, AssertionError):
  192         hit = False
  193     return {
  194         "metric": UPSTREAM_METRIC,
  195         "reward": 1.0 if hit else 0.0,
  196         "correct": hit,
  197         "protocol": "offline sequence; not interactive official accuracy",
  198     }

  201 class StepwisePuzzle:
  202     """Trusted session; only `public_task()` results may cross into the actor.
  203 
  204     Construct with verifier-owned gold. Present `public_task()`; receive exactly
  205     {"uci": "e2e4"}; call `submit()` and present its result before asking again.
  206     Responses reveal the resulting board, turn number and verdict, never the expected
  207     solver move, future opponent response, line length, original FEN or reference line.
  208     A runner must isolate this object and gold from the actor, and retain submissions
  209     itself; an actor-authored transcript is not evidence of interactive evaluation.
  210     """
  211 
  212     def __init__(self, gold: dict):
  213         self._board = None
  214         self._moves = list(gold["moves"])
  215         self._index = 1
  216         self._solver_turn = 0
  217         self._status = "awaiting_move"
  218         self._verdict = "awaiting_move"
  219         try:
  220             self._board = chess.Board(gold["fen"])
  221         except ValueError:
  222             self._finish("invalid_reference")
  223             return
  224         if not self._board.is_valid() or len(self._moves) < 2:
  225             self._finish("invalid_reference")
  226             return
  227         move = self._legal_move(self._moves[0], reference=True)
  228         if move is not None:
  229             self._board.push(move)
  230             if self._board.is_game_over():
  231                 self._finish("invalid_reference")
  232 
  233     def _finish(self, verdict: str) -> None:
  234         self._verdict = verdict
  235         self._status = "solved" if verdict in {
  236             "reference_complete", "checkmate", "alternate_mate_in_one"} else "failed"
  237 
  238     def _legal_move(self, uci: str, *, reference: bool = False):
  239         try:
  240             move = chess.Move.from_uci(uci)
  241         except (ValueError, TypeError):
  242             self._finish("invalid_reference" if reference else "invalid_move")
  243             return None
  244         # Board.push() is deliberately unchecked in python-chess, even for null moves.
  245         if move not in self._board.legal_moves:
  246             self._finish("invalid_reference" if reference else "illegal_move")
  247             return None
  248         return move
  249 
  250     def public_task(self) -> dict:
  251         return {
  252             "contract": STEPWISE_CONTRACT,
  253             "fen": self._board.fen() if self._board is not None else None,
  254             "solver_turn": self._solver_turn,
  255             "status": self._status,
  256             "verdict": self._verdict,
  257         }
  258 
  259     def submit(self, answer: dict) -> dict:
  260         if self._status != "awaiting_move":
  261             raise ValueError("puzzle session is already terminal")
  262         if (not isinstance(answer, dict) or set(answer) != {"uci"}
  263                 or not isinstance(answer["uci"], str)):
  264             self._finish("invalid_answer")
  265             return self.public_task()
  266         move = self._legal_move(answer["uci"])
  267         if move is None:
  268             return self.public_task()
  269         expected = self._moves[self._index]
  270         self._board.push(move)
  271         self._solver_turn += 1
  272         if self._board.is_checkmate():
  273             self._finish("checkmate" if answer["uci"] == expected
  274                          else "alternate_mate_in_one")
  275         elif answer["uci"] != expected:
  276             self._finish("wrong_move")
  277         else:
  278             self._index += 1
  279             if self._index < len(self._moves):
  280                 reply = self._legal_move(self._moves[self._index], reference=True)
  281                 if reply is None:
  282                     return self.public_task()
  283                 self._board.push(reply)
  284                 self._index += 1
  285             if self._index == len(self._moves):
  286                 self._finish("reference_complete")
  287             elif self._board.is_game_over():
  288                 self._finish("invalid_reference")
  289         return self.public_task()
  290 
  291     def result(self) -> dict:
  292         hit = self._status == "solved"
  293         return {"metric": STRICT_METRIC, "reward": float(hit), "correct": hit,
  294                 "verdict": self._verdict, "solver_moves_played": self._solver_turn}

  297 def score_strict(prediction: str, gold: dict) -> dict:
  298     """Replay requested solver tokens through the same legal stepwise state machine."""
  299     session = StepwisePuzzle(gold)
  300     for token in prediction.strip().split():
  301         if session.public_task()["status"] != "awaiting_move":
  302             break
  303         session.submit({"uci": token})
  304     if session.public_task()["status"] == "awaiting_move":
  305         session._finish("missing_move")
  306     return session.result()

  309 def score(prediction: str, gold: dict) -> dict:
  310     strict = score_strict(prediction, gold)
  311     upstream = score_upstream(prediction, gold)
  312     return {
  313         **strict,
  314         "official": {
  315             "metric": STRICT_METRIC,
  316             "protocol": "offline strict sequence; not interactive official accuracy",
  317             "puzzle_id": gold["puzzle_id"],
  318             "strict": strict,
  319             "upstream": upstream,
  320         },
  321         "why": f"strict legal replay: {strict['verdict']}",
  322     }
