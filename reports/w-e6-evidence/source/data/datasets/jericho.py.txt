    1 r"""Jericho as a dataset: 56 Z-machine text adventures, scored by the games' own points.
    2 
    3     python3 tools/datasets.py --fetch --dataset jericho
    4     python3 data/datasets/jericho/control.py --oracle
    5     python3 data/datasets/jericho/control.py --oracle --platform
    6     driver.py --dataset jericho ...
    7 
    8 Where the tasks come from
    9 -------------------------
   10 Two artifacts, both pinned:
   11 
   12 * the **game files** from `BYU-PCCL/z-machine-games`, the archive Jericho's own quick-start
   13   tells you to download. The suite is 8.5 MB of `.z3`/`.z4`/`.z5`/`.z8` ROMs; the archive is
   14   110 MB because it also carries the `z-machine-games-master/z-machine-games-master`
   15   collection (another 100 MB of ROMs this dataset never reads).
   16 * the **task list and reference solutions** from the installed `jericho` package's own
   17   `jericho/game_info.py`: 57 games, each with `rom`, the default `seed`, a full
   18   `walkthrough` and `minimal_actions`. That module is the benchmark's own table, so the task
   19   set is not this adapter's invention.
   20 
   21 **56 of the 57 are usable**: `sherbet.z5` is absent from the BYU-PCCL archive (verified by
   22 listing the extracted suite). The 56th is excluded rather than replaced, and `provision_status`
   23 reports it, because a game with no ROM is a task nobody can run.
   24 
   25 The metric, and where it comes from
   26 -----------------------------------
   27 Jericho's documented usage is:
   28 
   29     observation, reward, done, info = env.step('open mailbox')
   30     print('Total Score', info['score'], 'Moves', info['moves'])
   31     print('Scored', info['score'], 'out of', env.get_max_score())
   32 
   33 so a game's result is the **points the game itself awards**, and there are three ways to
   34 reduce it to a number. This adapter uses the fraction:
   35 
   36     reward = info['score'] / env.get_max_score()      (0.0 - 1.0)
   37 
   38 and records the raw pair in the evidence. The alternatives, and why not:
   39 
   40 * **normalised by the walkthrough's score** -- the usual "normalized score" in the Jericho
   41   paper. It differs from `get_max_score()` only for games where the reference solution does
   42   not collect every point, and it needs the walkthrough replayed per task to know the
   43   denominator; the environment's own maximum is a number the game states about itself, which
   44   is a better authority than a solution's outcome. Both are recorded, so a reader can pick.
   45 * **absolute points** -- not comparable across games: 350 for Zork 1, 100 for `ludicorp`.
   46   A mean over absolute points is dominated by whichever games happen to have large maxima.
   47 
   48 The fraction makes "0" mean "made no progress" and "1" mean "the game is finished", and it is
   49 the same reduction the environment's own documentation demonstrates.
   50 
   51 The oracle
   52 ----------
   53 `FrotzEnv.get_walkthrough()`, at the game's **official seed** (`game_info[key]['seed']`, which
   54 is what `get_walkthrough()` is valid for -- the docs say so and the seeds are checked below).
   55 The walkthrough is the benchmark's reference solution, exactly as ALFWorld's `walkthrough` is,
   56 and it lives in the installed package rather than in a file this dataset ships, so the
   57 provisioner cannot corrupt it.
   58 
   59 What is deliberately not added
   60 ------------------------------
   61 * **No `done` gate.** A game ends when the game says so; `jer done` exists so a harness can
   62   stop and is recorded, not required. §2's rule, third dataset running.
   63 * **No move budget.** Unlike ALFWorld and ScienceWorld, Jericho's games have no step limit
   64   and the benchmark does not impose one: Zork 1's walkthrough is 396 moves, and a limit that
   65   cut it off would score a correct oracle below 1. The budget is therefore **the length of
   66   the log**; a harness that never stops is bounded by the platform's 300 s timeout, which is
   67   the honest place for that bound.
   68 """

  125 def _official_games() -> list[dict]:
  126     """The benchmark's own game table, filtered to the ROMs that are actually present.
  127 
  128     `jericho.game_info` is imported **lazily**: `data/scaffolds/registry.py::published` imports every
  129     adapter to decide what the chooser offers, and an import that pulls in `jericho` (a
  130     ctypes-wrapped C library) at that point is a second thing that can fail on a machine that
  131     has not installed it.
  132     """
  133     import jericho.game_info as info
  134 
  135     suite = games_dir()
  136     records = []
  137     for key in sorted(dir(info)):
  138         if key.startswith("_"):
  139             continue
  140         entry = getattr(info, key)
  141         if not isinstance(entry, dict) or "rom" not in entry:
  142             continue
  143         rom = str(entry["rom"])
  144         path = suite / rom
  145         if not path.is_file():
  146             continue
  147         walkthrough = [a for a in str(entry.get("walkthrough") or "").split("/") if a]
  148         records.append({
  149             "task_id": Path(rom).stem,
  150             "key": key,
  151             "rom": rom,
  152             "path": path,
  153             "seed": int(entry.get("seed") or 0),
  154             "walkthrough": walkthrough,
  155         })
  156     if not records:
  157         raise FileNotFoundError(
  158             f"no game ROMs under {suite}; provision them with "
  159             f"`python3 tools/datasets.py --fetch --dataset jericho`")
  160     return records

  163 def missing_games() -> list[str]:
  164     """Official games whose ROM the archive does not carry. Reported, not silently dropped."""
  165     import jericho.game_info as info
  166 
  167     suite = games_dir()
  168     out = []
  169     for key in sorted(dir(info)):
  170         if key.startswith("_"):
  171             continue
  172         entry = getattr(info, key)
  173         if isinstance(entry, dict) and "rom" in entry and not (suite / str(entry["rom"])).is_file():
  174             out.append(str(entry["rom"]))
  175     return out

  178 def _goal(record: dict) -> str:
  179     """The instruction: what the interface is, and the budget that actually exists.
  180 
  181     Nothing here hints at the solution. The game's own opening text is deliberately **not**
  182     included: printing it in the goal would be free information the agent should get by
  183     looking, and it would go stale the moment the seed changes the starting room.
  184     """
  185     return (
  186         f"# Task\n"
  187         f"\n"
  188         f"You are playing a classic interactive-fiction game, `{record['task_id']}`, "
  189         f"described one room at a time. You act by typing commands.\n"
  190         f"\n"
  191         f"## How to act\n"
  192         f"\n"
  193         f"- See where you are and what you can see:\n"
  194         f"      python3 jer look\n"
  195         f"- Take one action (ordinary parser commands; the game decides whether it\n"
  196         f"  understands, and an action it does not understand still costs a move):\n"
  197         f"      python3 jer do \"open mailbox\"\n"
  198         f"- See your score and how many moves you have spent:\n"
  199         f"      python3 jer score\n"
  200         f"- See what you are carrying:\n"
  201         f"      python3 jer inventory\n"
  202         f"- When you are finished, say so:\n"
  203         f"      python3 jer done\n"
  204         f"- Every action is appended to `{LOG_FILE}`.\n"
  205         f"\n"
  206         f"## How it is scored\n"
  207         f"\n"
  208         f"- Your actions are replayed from the beginning in a fresh copy of the game at its\n"
  209         f"  official seed, and the **game's own score** is read at the end, as a fraction of\n"
  210         f"  the maximum the game can award. Finishing the game means reaching 1.0.\n"
  211         f"- There is no move limit: play until the game ends or you stop.\n"
  212         f"- Nothing you print is read. Only the actions you take are.\n"
  213     )

  216 def _setup(record: dict) -> dict:
  217     """What the agent receives: the CLI, its library, the game's identity, and the ROM.
  218 
  219     The ROM has to be here and not only in `VERIFY.inputs`, which cost a platform control run
  220     to learn: measured, `--oracle --platform` scored 0 with "the log holds no actions", because
  221     the harness's `jer do ...` could not start the game -- the harness and the check share one
  222     workdir, and the ROM had only been written for the check. The platform's own dataset
  223     pattern (ALFWorld, ScienceWorld) puts everything both sides need in `SETUP` **and** in
  224     `VERIFY.inputs`, and that is now what this does.
  225 
  226     The ROM is not a secret -- it is the game, and the agent cannot play without it. The
  227     *answer* is the walkthrough. The agent sees a stripped package projection;
  228     the installed reference-bearing package is reserved for trusted verification.
  229     """
  230     import json
  231 
  232     return {"files": [
  233         {"path": LOG_FILE, "content": ""},
  234         {"path": GAME_FILE,
  235          "content": json.dumps({"task_id": record["task_id"], "rom": record["rom"],
  236                                 "seed": record["seed"]})},
  237         {"path": "jer_env.py", "from": str(LIB_DIR / "jer_env.py")},
  238         {"path": "jer", "from": str(CLI_DIR / "jer")},
  239         {"path": record["rom"], "from": str(record["path"])},
  240     ]}

  250     import json

  257             "--game", GAME_FILE,

  259             "--log", LOG_FILE,

  355 def load() -> tuple[list[dict], dict[str, str]]:
  356     """`(tasks, scorable)`. Every task is graded by its `VERIFY` entry."""
  357     records = __getattr__("_RECORDS")
  358     if _LOAD_ERROR is not None:
  359         raise ValueError(
  360             f"jericho: cannot read the game suite ({_LOAD_ERROR}). Provision it with "
  361             f"`python3 tools/datasets.py --fetch --dataset jericho` and install "
  362             f"`jericho==3.3.1`.")
  363     if not records:
  364         raise ValueError(f"jericho: no games under {games_dir()}")
  365     tasks = [{"task_id": r["task_id"], "goal": _goal(r), "inputs": []} for r in records]
  366     return tasks, {r["task_id"]: "" for r in records}

  369 def _is_eval(task_id: str) -> bool:
  370     """A deterministic third, by hash of the task id (ALFWorld's and Terminal-Bench's rule).
  371 
  372     The benchmark declares no train/eval division, so the platform needs one imposed; hashing
  373     the id means adding or removing a game does not reshuffle the other side.
  374     """
  375     import hashlib
  376     return hashlib.sha256(task_id.encode()).digest()[0] % 3 == 0

  383 def __getattr__(name: str):
  384     """`ENV`/`SETUP`/`VERIFY`/`SPLIT`, built on first use.
  385 
  386     Not at import: `data/scaffolds/registry.py::published` imports this module to decide what the
  387     chooser offers, and the maps carry one entry per game with a ROM path and a JSON blob.
  388     """
  389     if name == "_RECORDS":
  390         if "_RECORDS" not in globals():
  391             try:
  392                 globals()["_RECORDS"] = _official_games()
  393             except Exception as exc:  # reported by load(), as with eager provisioning
  394                 globals()["_RECORDS"] = []
  395                 globals()["_LOAD_ERROR"] = f"{type(exc).__name__}: {exc}"
  396         return globals()["_RECORDS"]
  397     if name == "ENV":
  398         env = _env()
  399         return {r["task_id"]: dict(env) for r in __getattr__("_RECORDS")}
  400     if name == "SETUP":
  401         return _map(_setup)
  402     if name == "VERIFY":
  403         return _map(_verifier)
  404     if name == "SPLIT":
  405         records = __getattr__("_RECORDS")
  406         if (os.environ.get("HG_JERICHO_SPLIT") or "hash").lower() == "none":
  407             return {"train": [], "eval": [r["task_id"] for r in records]}
  408         ids = [r["task_id"] for r in records]
  409         return {"train": [t for t in ids if not _is_eval(t)],
  410                 "eval": [t for t in ids if _is_eval(t)]}
  411     raise AttributeError(f"module {__name__!r} has no attribute {name!r}")
