    1 #!/usr/bin/env python3
    2 r"""The SpreadsheetBench check: recalculate the agent's workbook and compare it.
    3 
    4 The comparison is the official one, transcribed from
    5 `RUCKBReasoning/SpreadsheetBench`'s `evaluation/evaluation.py`:
    6 
    7     load both workbooks with `openpyxl.load_workbook(..., data_only=True)`
    8     for each `Sheet!A1:B2` range in the task's `answer_position`, compare every cell
    9     as `transform_value(gt) == transform_value(proc)`
   10 
   11 `transform_value` (verbatim):
   12 
   13     numbers -> `round(float(v), 2)`
   14     `datetime` -> Excel's serial day count, rounded
   15     `datetime.time` -> `str(v)[:-3]` (to the minute)
   16     a string that parses as a float -> that float
   17     `""` and `None` compare equal
   18 
   19 **Why the recalculation step exists.** `data_only=True` returns the value Excel last *cached*
   20 next to each formula. A workbook produced by a program that computed its own answers has those
   21 caches; one whose cells hold formulas that were never evaluated has `None` in every one of them,
   22 and would be scored 0 for a reason no evidence would show. So the agent's workbook is
   23 recalculated first, with LibreOffice in headless mode -- the same thing the repository's own
   24 `evaluation/open_spreadsheet.py` does, and the reason `soffice` has to be installed.
   25 
   26 **What is not compared.** Cell fill and font colour: the official `cell_level_compare` has both
   27 comparisons present and commented out, and this file matches the code that runs.
   28 
   29 **A check that cannot measure** -- unreadable workbooks, comparator/runtime failures, no
   30 LibreOffice when the workbook holds formulas -- writes **no reward file**, writes evidence
   31 saying `measured: false`, and exits 3. A bare 0 is indistinguishable from a wrong workbook.
   32 An absent result artifact and a completed value/sheet mismatch remain measured zero.
   33 An answer range rejected by the official parser also keeps its official-equivalent zero;
   34 this compatibility rule does not establish that any cells were compared.
   35 """
   36 from __future__ import annotations

   50 DEFAULT_EVIDENCE_FILE = "/verifier/evidence.json"

   53 # ------------------------------------------------------- the official scorer ---
   54 # `evaluation/evaluation.py`, transcribed. The function bodies are upstream's; the only
   55 # changes are `print` -> return values, the module-level imports it assumed, and raising
   56 # measurement failures instead of returning the same False as a completed mismatch.

   59 def transform_value(v):
   60     """`evaluation.py::transform_value`, verbatim."""
   61     if isinstance(v, (int, float)):
   62         v = round(float(v), 2)
   63     elif isinstance(v, datetime.time):
   64         v = str(v)[:-3]
   65     elif isinstance(v, datetime.datetime):
   66         v = round(datetime_to_float(v), 0)
   67     elif isinstance(v, str):
   68         try:
   69             v = round(float(v), 2)
   70         except ValueError:
   71             pass
   72     return v
   73 
   74 
   75 def datetime_to_float(dt):
   76     """`evaluation.py::datetime_to_float`, verbatim."""
   77     excel_start_date = datetime.datetime(1899, 12, 30)
   78     delta = dt - excel_start_date
   79     return delta.days + delta.seconds / 86400.0
   80 
   81 
   82 def compare_cell_value(v1, v2):
   83     """`evaluation.py::compare_cell_value`, verbatim."""
   84     v1 = transform_value(v1)
   85     v2 = transform_value(v2)
   86     if (v1 == "" and v2 is None) or (v1 is None and v2 == ""):
   87         return True
   88     if (v1 == "" and v2 == "") or (v1 is None and v2 is None):
   89         return True
   90     if type(v1) != type(v2):
   91         return False
   92     if v1 == v2:
   93         return True
   94     else:
   95         return False
   96

   98 def col_num2name(n):
   99     """`evaluation.py::col_num2name`, verbatim."""
  100     name = ""
  101     while n > 0:
  102         n, remainder = divmod(n - 1, 26)
  103         name = chr(65 + remainder) + name
  104     return name
  105 
  106 
  107 def col_name2num(name):
  108     """`evaluation.py::col_name2num`, verbatim."""
  109     num = 0
  110     for c in name:
  111         num = num * 26 + (ord(c) - ord("A") + 1)
  112     return num
  113

  115 def parse_cell_range(range_str):
  116     """`evaluation.py::parse_cell_range`, verbatim."""
  117     start_cell, end_cell = range_str.split(":")
  118     start_col, start_row = "", ""
  119     for char in start_cell:
  120         if char.isdigit():
  121             start_row += char
  122         else:
  123             start_col += char
  124     end_col, end_row = "", ""
  125     for char in end_cell:
  126         if char.isdigit():
  127             end_row += char
  128         else:
  129             end_col += char
  130     return (col_name2num(start_col), int(start_row)), (col_name2num(end_col), int(end_row))
  131 
  132 
  133 def generate_cell_names(range_str):
  134     """`evaluation.py::generate_cell_names`, verbatim."""
  135     if ":" not in range_str:
  136         return [range_str]
  137     (start_col, start_row), (end_col, end_row) = parse_cell_range(range_str)
  138     columns = [col_num2name(i) for i in range(start_col, end_col + 1)]
  139     return [f"{col}{row}" for col in columns for row in range(start_row, end_row + 1)]
  140

  142 def cell_level_compare(wb_gt, wb_proc, sheet_name, cell_range):
  143     """`evaluation.py::cell_level_compare`, with the comparisons that are commented out
  144     upstream left out here too. Returns `(ok, message)`."""
  145     # Invalid reference metadata is not an agent's missing-sheet mismatch.
  146     ws_gt = wb_gt[sheet_name]
  147     if sheet_name not in wb_proc:
  148         return False, f"worksheet {sheet_name!r} not found in the result"
  149     ws_proc = wb_proc[sheet_name]
  150     for cell_name in generate_cell_names(cell_range):
  151         cell_gt = ws_gt[cell_name]
  152         cell_proc = ws_proc[cell_name]
  153         if not compare_cell_value(cell_gt.value, cell_proc.value):
  154             return False, (f"value difference at {cell_gt.coordinate}: golden has "
  155                            f"{cell_gt.value!r}, result has {cell_proc.value!r}")
  156     return True, ""

  159 def compare_workbooks(gt_file, proc_file, instruction_type, answer_position):
  160     """`evaluation.py::compare_workbooks`, with measurement failures raised.
  161 
  162     `instruction_type` is accepted and unused, exactly as upstream: the comparison does not
  163     branch on it (upstream only uses it to group the report).
  164     """
  165     import openpyxl
  166 
  167     if not os.path.exists(proc_file):
  168         # main() handles an absent submission; disappearance during comparison is a failure.
  169         raise FileNotFoundError(f"result workbook unavailable for comparison: {proc_file}")
  170     try:
  171         wb_gt = openpyxl.load_workbook(filename=gt_file, data_only=True)
  172         wb_proc = openpyxl.load_workbook(filename=proc_file, data_only=True)
  173     except Exception as exc:  # noqa: BLE001 - loading failed, no comparison was completed
  174         raise RuntimeError(f"workbook load failed: {type(exc).__name__}: {exc}") from exc
  175 
  176     sheet_cell_ranges = answer_position.split(",")
  177     result_list = []
  178     msg_list = []
  179     for sheet_cell_range in sheet_cell_ranges:
  180         # Compatibility decision (w-spread2, option a): upstream evaluation() catches
  181         # range-parse ValueError and records int(False). Preserve that measured zero
  182         # for released metadata such as 283-32; it is not a local runtime failure.
  183         # Unexpected parser exceptions still propagate to the unmeasured path.
  184         _cell = sheet_cell_range.split("!")[-1].strip().strip("'")
  185         # Only a **range** goes through `parse_cell_range`; a single cell (`B6`) is returned
  186         # as-is by `generate_cell_names` and never parsed. Measured: a first version validated
  187         # every position, which rejected `B6` and marked `51-12` unmeasurable.
  188         if ":" in _cell:
  189             try:
  190                 _ = parse_cell_range(_cell)
  191             except ValueError as exc:
  192                 return False, (
  193                     f"official-equivalent measured zero: the task's answer_position "
  194                     f"{sheet_cell_range!r} is not a cell range the official parser can read: "
  195                     f"{exc}")
  196         if "!" in sheet_cell_range:
  197             sheet_name, cell_range = sheet_cell_range.split("!")
  198             sheet_name = sheet_name.lstrip("'").rstrip("'")
  199         else:
  200             sheet_name = wb_gt.sheetnames[0]
  201             cell_range = sheet_cell_range
  202         sheet_name = sheet_name.lstrip("'").rstrip("'")
  203         cell_range = cell_range.lstrip("'").rstrip("'")
  204         result, msg = cell_level_compare(wb_gt, wb_proc, sheet_name, cell_range)
  205         result_list.append(result)
  206         msg_list.append(msg)
  207     first_bad = next((m for r, m in zip(result_list, msg_list) if not r), "")
  208     return all(result_list), first_bad

  214 def soffice_path() -> str | None:
  215     """The LibreOffice binary, from `PATH` or the two conventional names."""
  216     for candidate in ("soffice", "libreoffice"):
  217         found = shutil.which(candidate)
  218         if found:
  219             return found
  220     return None

  223 def recalculate(workbook: Path, *, scratch: Path, soffice: str) -> tuple[bool, str]:
  224     """Re-evaluate `workbook`'s formulas and cache the results, in place.
  225 
  226     LibreOffice's converter writes a file whose cells carry the computed values, which is what
  227     `data_only=True` then reads. A private `UserInstallation` keeps concurrent checks from
  228     fighting over one profile lock -- measured: without it, two LibreOffice runs on one host
  229     can make the second fail.
  230     """
  231     profile = scratch / "lo-profile"
  232     profile.mkdir(parents=True, exist_ok=True)
  233     outdir = scratch / "conv"
  234     outdir.mkdir(parents=True, exist_ok=True)
  235     env = dict(os.environ, HOME=str(scratch),
  236                TMPDIR=str(scratch))
  237     cmd = [soffice, "--headless", "--calc", "--norestore", "--nolockcheck",
  238            f"-env:UserInstallation=file://{profile}",
  239            "--convert-to", "xlsx:Calc MS Excel 2007 XML",
  240            "--outdir", str(outdir), str(workbook)]
  241     try:
  242         proc = subprocess.run(cmd, capture_output=True, text=True, timeout=180, env=env)
  243     except subprocess.TimeoutExpired:
  244         return False, "LibreOffice timed out after 180s"
  245     if proc.returncode != 0:
  246         return False, f"LibreOffice exited {proc.returncode}: {proc.stderr.strip()[:300]}"
  247     converted = outdir / (workbook.stem + ".xlsx")
  248     if not converted.is_file():
  249         return False, (f"LibreOffice produced no file (stderr: {proc.stderr.strip()[:200]})")
  250     shutil.move(str(converted), str(workbook))
  251     return True, ""
  252

  254 def workbook_has_formulas(path: Path) -> bool:
  255     """Does any cell hold a formula? Decides whether recalculation is needed."""
  256     import openpyxl
  257 
  258     try:
  259         wb = openpyxl.load_workbook(filename=path, data_only=False)
  260     except Exception as exc:  # noqa: BLE001 - False must mean a readable, formula-free file
  261         raise RuntimeError(f"workbook load failed during formula detection: "
  262                            f"{type(exc).__name__}: {exc}") from exc
  263     for ws in wb.worksheets:
  264         for row in ws.iter_rows():
  265             for cell in row:
  266                 if isinstance(cell.value, str) and cell.value.startswith("="):
  267                     return True
  268     return False

  274 def parse_args(argv):
  275     parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
  276     parser.add_argument("--workdir", default=".")
  277     parser.add_argument("--answer", default="answer.txt")
  278     parser.add_argument("--result", default="result.xlsx")
  279     parser.add_argument("--golden", default="golden.xlsx")
  280     parser.add_argument("--task", default="task.json")
  281     parser.add_argument("--reward-file", default=DEFAULT_REWARD_FILE)
  282     parser.add_argument("--evidence-file", default=DEFAULT_EVIDENCE_FILE)
  283     args = parser.parse_args(argv)
  284     args.workdir = Path(args.workdir)
  285     args.reward_file = Path(args.reward_file)
  286     args.evidence_file = Path(args.evidence_file)
  287     return args

  290 def write_reward(path: Path, score: float) -> None:
  291     path.parent.mkdir(parents=True, exist_ok=True)
  292     path.write_text(f"{score:g}\n")
  293 
  294 
  295 def write_evidence(path: Path, evidence: dict) -> None:
  296     try:
  297         path.parent.mkdir(parents=True, exist_ok=True)
  298         path.write_text(json.dumps(evidence, indent=1, sort_keys=True,
  299                                    ensure_ascii=False) + "\n")
  300     except OSError as exc:
  301         print(f"could not write evidence: {exc}", file=sys.stderr)
  302

  304 def main(argv) -> int:
  305     args = parse_args(argv)
  306     workdir = Path(args.workdir)
  307     task_path = workdir / args.task
  308     golden_path = workdir / args.golden
  309     answer_path = workdir / args.answer
  310     evidence: dict = {"answer_file": str(answer_path), "golden": str(golden_path)}
  311 
  312     if not task_path.is_file():
  313         evidence.update(measured=False, error=f"no task file at {task_path}")
  314         write_evidence(args.evidence_file, evidence)
  315         print(f"check cannot measure: {evidence['error']}", file=sys.stderr)
  316         return 3
  317     if not golden_path.is_file():
  318         evidence.update(measured=False, error=f"no golden workbook at {golden_path}")
  319         write_evidence(args.evidence_file, evidence)
  320         print(f"check cannot measure: {evidence['error']}", file=sys.stderr)
  321         return 3
  322 
  323     spec = json.loads(task_path.read_text())
  324     evidence.update(task_id=spec.get("task_id"),
  325                     answer_position=spec.get("answer_position"),
  326                     instruction_type=spec.get("instruction_type"))
  327 
  328     raw_answer = answer_path.read_text(encoding="utf-8").strip() if answer_path.is_file() else ""
  329     # The answer is the **name of the workbook** the agent left behind. An empty answer means
  330     # the conventional name; anything else is taken as a relative path inside the workdir, so a
  331     # harness cannot name a file outside the task.
  332     named = raw_answer.splitlines()[0].strip() if raw_answer else ""
  333     named = named or args.result
  334     candidate = (workdir / named).resolve()
  335     if not str(candidate).startswith(str(workdir.resolve()) + os.sep):
  336         evidence.update(measured=False,
  337                         error=f"the answer names {named!r}, which is outside the workdir")
  338         write_evidence(args.evidence_file, evidence)
  339         print(f"check cannot measure: {evidence['error']}", file=sys.stderr)
  340         return 3
  341     evidence.update(named_result=named, result_exists=candidate.is_file())
  342 
  343     if not candidate.is_file():
  344         # The agent left no workbook: a measured zero, not an unmeasured task. It had a
  345         # contract, and the artifact it names is absent.
  346         evidence.update(measured=True, reward=0.0,
  347                         why=f"the answer names {named!r} but no such file exists",
  348                         score=0.0, correct=False)
  349         write_evidence(args.evidence_file, evidence)
  350         write_reward(args.reward_file, 0.0)
  351         print(f"reward 0: {evidence['why']}")
  352         return 1
  353 
  354     phase = "formula detection"
  355     try:
  356         # Recalculate before comparing: data_only=True reads cached formula values.
  357         if workbook_has_formulas(candidate):
  358             phase = "LibreOffice recalculation"
  359             soffice = soffice_path()
  360             if soffice is None:
  361                 raise RuntimeError("the result workbook holds formulas and no LibreOffice is "
  362                                    "installed, so its values cannot be computed; the check "
  363                                    "will not score an unevaluated workbook")
  364             with tempfile.TemporaryDirectory(prefix="sb-recalc-") as scratch:
  365                 ok, why = recalculate(candidate, scratch=Path(scratch), soffice=soffice)
  366             evidence["recalculated"] = ok
  367             if not ok:
  368                 raise RuntimeError(f"could not recalculate the workbook: {why}")
  369         else:
  370             evidence["recalculated"] = False
  371 
  372         phase = "comparison"
  373         ok, message = compare_workbooks(str(golden_path), str(candidate),
  374                                         spec.get("instruction_type"),
  375                                         str(spec["answer_position"]))
  376     except Exception as exc:  # noqa: BLE001 - a failed tool/comparator is not a zero
  377         evidence.update(measured=False, error=f"{phase} failed: {type(exc).__name__}: {exc}",
  378                         traceback=traceback.format_exc()[-2000:])
  379         write_evidence(args.evidence_file, evidence)
  380         print(f"check cannot measure: {evidence['error']}", file=sys.stderr)
  381         return 3
  382 
  383     evidence.update(measured=True, reward=1.0 if ok else 0.0, correct=bool(ok),
  384                     score=1.0 if ok else 0.0,
  385                     why=("the named cells hold the reference values" if ok
  386                          else message if message.startswith("official-equivalent measured zero:")
  387                          else f"workbook differs: {message or 'comparison failed'}"))
  388     write_evidence(args.evidence_file, evidence)
  389     write_reward(args.reward_file, 1.0 if ok else 0.0)
  390     print(f"reward {1.0 if ok else 0.0:g}: {evidence['why']}")
  391     return 0 if ok else 1
  392 
  393 
  394 if __name__ == "__main__":
  395     try:
  396         raise SystemExit(main(sys.argv[1:]))
  397     except SystemExit:
  398         raise
  399     except Exception:  # noqa: BLE001 - a crash must not look like a measured zero
  400         traceback.print_exc()
  401         raise SystemExit(4)
