    1 """What every plain-text QA dataset in this repository shares, in one place.
    2 
    3 The batch this exists for -- `gsm8k`, `triviaqa`, `sotuqa`, `strategyqa`,
    4 `sportsunderstanding`, `physicsquestions`, `fever`, `hotpotqa` -- differs only in the
    5 part a program cannot decide: *which table*, *which field is the question*, and *what
    6 the official scorer says*. Everything else is identical and is written once here:
    7 
    8     the data root is outside the platform tree
    9     the pin is the revision the files were fetched from
   10     a task id is stable across machines and stable when a file is re-extracted
   11     the split is imposed by hash, so adding a task does not reshuffle the others
   12     the agent answers by writing one file, and the check re-derives the grade from
   13       the official scorer and a **re-materialized** gold file
   14 
   15 The answer contract, and why it is a file
   16 -----------------------------------------
   17 `answer.txt` is the platform's own convention (`eval/run/runner.py` reads it, every base
   18 harness writes it), so a QA task needs no new channel: the goal says to write it, the
   19 harness writes it, and the check reads it. `data/scaffolds/qa/cli/qa` exists so a harness that
   20 only knows how to run shell commands can still answer, and so the format the scorer
   21 expects is reachable from inside the task.
   22 
   23 The gold travels as `VERIFY.inputs` content, never to the agent
   24 ---------------------------------------------------------------
   25 A task dict is written to `task.json` and handed to the harness, and
   26 `data/scaffolds/registry.py::FORBIDDEN_IN_TASKS` refuses `expected` **in a task** for exactly
   27 this reason. The gold therefore lives in the verifier spec: the platform
   28 re-materializes `VERIFY.inputs` immediately before the check runs, so the file the
   29 check reads is the platform's copy and not anything the agent could have edited. That
   30 is the same mechanism that keeps ALFWorld's game file trustworthy, used here for one
   31 string.
   32 
   33 What this module deliberately does not do
   34 -----------------------------------------
   35 It does not decide whether a dataset's official metric is *reachable*. Some in this
   36 batch are not, and a dataset whose contract cannot carry the official number raises
   37 instead of reporting a nearby one -- see `data/datasets/fever.py` and `data/datasets/hotpotqa.py`, and
   38 `docs/dataset-onboarding-sketch` §2 for why dropping a requirement is the dangerous
   39 direction.
   40 """

  123 def is_eval(task_id: str) -> bool:
  124     """A deterministic third of the tasks, by hash of the task id.
  125 
  126     Hashed rather than sliced off the sorted list so that adding or removing a task does
  127     not reshuffle which side every other task is on -- a split that moves under the
  128     benchmark is a split whose past scores cannot be compared with its future ones. The
  129     same rule `data/datasets/alfworld.py` and `data/datasets/terminal_bench.py` use.
  130     """
  131     return hashlib.sha256(task_id.encode()).digest()[0] % 3 == 0

  134 class QASpec:
  135     """One QA dataset's decisions. Everything else is `Scaffold`'s.
  136 
  137     `rows` is a callable, not a list, because a dataset adapter is imported on machines
  138     where its data was never provisioned (`data/scaffolds/registry.py::published`) and must not
  139     read a byte at import time.
  140 
  141     The shared QA controls read the adapter module's optional `NOOP_EXPECTED`
  142     declaration: a float, or a callable taking the record dict and returning a float.
  143     It sets the expected do-nothing reward in both local and platform controls and
  144     defaults to 0.0 when absent. The oracle arm always expects 1.0.
  145     """
  146 
  147     def __init__(
  148         self,
  149         *,
  150         name: str,
  151         root: Path,
  152         rows: Callable[[], dict[str, list[dict]]],
  153         question: Callable[[dict], str],
  154         gold: Callable[[dict], dict],
  155         goal: Callable[[dict, dict], str],
  156         score: Callable[[str, dict], dict],
  157         downloads: Callable[[], list[dict]],
  158         split: str,
  159         source: str,
  160         extra_setup: Sequence[dict] = (),
  161         study: Sequence[str] = (),
  162     ) -> None:
  163         self.name = name
  164         self.root = Path(root)
  165         self._rows = rows
  166         self._question = question
  167         self._gold = gold
  168         self._goal = goal
  169         self.score = score
  170         self._downloads = downloads
  171         self.split = split
  172         #: Other splits of the same benchmark that the benchmark itself designates as
  173         #: **study material** -- `gsm8k`'s `train`, for the one dataset where the benchmark
  174         #: publishes both sides. Empty for every QA adapter whose table is a single scored
  175         #: split, which is most of them: there the platform's hash rule (`is_eval`) is the
  176         #: only division that exists and is imposed rather than discovered.
  177         #:
  178         #: When this is non-empty the benchmark has stated the division, so the platform
  179         #: stops imposing one: `build()` derives the two sides from the split each task id
  180         #: carries, and `OFFICIAL_EVAL` becomes the scored split's ids instead of `None`.
  181         self.study = tuple(study)
  182         #: One or two sentences for the module docstring / DOWNLOADS: where this table
  183         #: came from and what was chosen about it.
  184         self.source = source
  185         self.extra_setup = list(extra_setup)
  186         #: Filled by `build()`. Declared here so an adapter can expose `_RECORDS` to
  187         #: the control before (or without) the extra call.
  188         self._records: list[dict] = []
  189         #: `None`, or the reason `load()` will refuse. Set by `build()` when the data
  190         #: is not on disk -- see `build()` for why that must not be an import failure.
  191         self.load_error: str | None = None

  193     # ---------------------------------------------------------------- the table ---
  194 
  195     def ensure(self) -> list[dict]:
  196         """`records()`, computed once and remembered.
  197 
  198         **The table is read lazily, and that is load-bearing.** A first version built
  199         every task's goal and every `VERIFY` entry at import time: measured on triviaqa,
  200         importing the adapter took **18 s** and every one of the four control runs paid
  201         it, because `load()` re-derived all 17,944 records each time. The platform only
  202         needs the task list when it runs or lints the dataset, so the expensive part is
  203         deferred to `load()`, and `load()` is not re-run per subprocess.
  204         """
  205         if self._records:
  206             return self._records
  207         if self.load_error is None:
  208             try:
  209                 self._records = self.records()
  210             except Exception as exc:  # noqa: BLE001 - carried, not swallowed
  211                 self.load_error = f"{type(exc).__name__}: {exc}"
  212                 self._records = []
  213         return self._records
  214 
  215     def records(self) -> list[dict]:
  216         """One record per row of the pool, in file order: the scored split first, then any
  217         study splits the benchmark designates.
  218 
  219         A record carries the question, the gold **as data** (a dict -- a list of
  220         aliases, a label, an acceptable-answer set), and the task id. Nothing else: the
  221         goal text is derived later, so the same record can be rendered for a human
  222         without recomputing the table.
  223 
  224         The split stays in the task id (`<dataset>/<split>/<n>`), which is what lets
  225         `build()` read the benchmark's own division back off the ids instead of imposing
  226         one -- and what keeps a `train` and a `test` row at the same index from colliding.
  227         """
  228         try:
  229             tables = self._rows()
  230         except FileNotFoundError as exc:
  231             raise ValueError(
  232                 f"{self.name}: {exc}. Provision it with "
  233                 f"`python3 tools/datasets.py --fetch --dataset {self.name}`") from exc
  234         records = []
  235         for split in (self.split, *self.study):
  236             rows = tables.get(split)
  237             if rows is None:
  238                 raise ValueError(
  239                     f"{self.name}: the loader returned splits {sorted(tables)}, and "
  240                     f"{split!r} is not one of them")
  241             for index, row in enumerate(rows):
  242                 # A sampled list is shorter than the file it came from, so the task id is
  243                 # the row's **file** position when the reader reports one
  244                 # (`data/scaffolds/qa/read.py::SOURCE_INDEX`). Numbering by position instead would
  245                 # let `train/7` keep its id and change its question the moment the sample
  246                 # rule changed -- same id, same `split_sha`, different task.
  247                 source_index = row.get(SOURCE_INDEX, index) if isinstance(row, dict) else index
  248                 if isinstance(row, dict) and SOURCE_INDEX in row:
  249                     row = {k: v for k, v in row.items() if k != SOURCE_INDEX}
  250                 gold = self._gold(row)
  251                 records.append({
  252                     "task_id": f"{self.name}/{split}/{source_index}",
  253                     "index": source_index,
  254                     "split": split,
  255                     "row": row,
  256                     "question": self._question(row),
  257                     "gold": gold,
  258                 })
  259         if not records:
  260             raise ValueError(f"{self.name}: {self.split} has no rows")
  261         return records
  262 
  263     # ------------------------------------------------------------- the platform ---
  264 
  265     def load(self) -> tuple[list[dict], dict[str, str]]:
  266         """`(tasks, scorable)`. Every task is graded by its `VERIFY` entry, so the
  267         `scorable` map carries the human-readable gold only as a fallback -- it must not
  268         be empty, because `load()` has no other way to say "this task has a grade"."""
  269         records = self.ensure()
  270         if self.load_error:
  271             raise ValueError(f"{self.name}: cannot read {self.root} ({self.load_error})")
  272         if not records:
  273             raise ValueError(f"{self.name}: no tasks under {self.root}")
  274         # The **declared** division, not the hash rule: once a benchmark states its own
  275         # two sides, a hash that happens to put every task on one side says nothing about
  276         # whether the dataset can run. `division()` is the one place that decides which of
  277         # the two is in force, and this guard reads its answer.
  278         division = self.division()
  279         if not division["train"] or not division["eval"]:
  280             raise ValueError(
  281                 f"{self.name}: the division put every task on one side "
  282                 f"({len(records)} tasks, source={division['source']}); a dataset with "
  283                 f"one empty side cannot run")
  284         self._goals = {r["task_id"]: self._goal(r["question"], r["gold"]) for r in records}
  285         tasks = [{"task_id": r["task_id"], "goal": self._goals[r["task_id"]],
  286                   "inputs": []}
  287                  for r in records]
  288         return tasks, {r["task_id"]: str(r["gold"].get("answer", "")) for r in records}
  289 
  290     def division(self) -> dict:
  291         """The two sides of this pool, and **which of the two rules** produced them.
  292 
  293         `"declared"` means the benchmark publishes both sides and `study` names the study
  294         one; `"hash"` means the benchmark publishes only a scored split, so the platform
  295         imposes the hash third (`is_eval`). One function rather than a rule copied into
  296         `build()` and `load()` separately, because the day those two disagree a dataset
  297         runs on a division its own `SPLIT` does not describe -- and the failure is silent.
  298         """
  299         records = self.ensure()
  300         ids = [r["task_id"] for r in records]
  301         if self.study:
  302             eval_ids = [r["task_id"] for r in records if r["split"] == self.split]
  303             # `set(eval_ids)` is hoisted out of the comprehension on purpose: written
  304             # inline it is rebuilt once **per task**, which on TriviaQA (19,944 tasks,
  305             # 17,944 of them eval) measured **9.75 s** in this one call -- paid at import
  306             # by `build()`. A set membership test is O(1); building the set is not.
  307             eval_set = set(eval_ids)
  308             return {"train": [t for t in ids if t not in eval_set],
  309                     "eval": eval_ids, "source": "declared", "official_eval": eval_ids}
  310         return {"train": [t for t in ids if not is_eval(t)],
  311                 "eval": [t for t in ids if is_eval(t)],
  312                 "source": "hash", "official_eval": None}

  322     def verifier(self, record: dict) -> dict:
  323         """The grade: a check that re-derives the official metric from `answer.txt`.
  324 
  325         `inputs` use `path` (relative to the workdir) rather than `dst`, because a
  326         `files` environment has no container to put `/gold` into. The gold is inline
  327         `content`, so it is re-materialized by the platform before the check runs and
  328         the copy the check reads is not the one the agent could have found.
  329 
  330         **Everything the check imports travels with it.** That is not tidiness: the
  331         verification namespace puts a tmpfs over the platform tree (`eval/containers/sandbox.py`
  332         hides the work root's ancestors), so a check that imported `data.scaffolds.qa.verify.check`
  333         from the platform would fail to find it -- or, worse, would find whatever the
  334         subject left at that path. So the check, the parquet reader and the dataset's own
  335         scorer are copied into `qa/` in the workdir and imported from there. The
  336         scorer itself must import nothing but the standard library for this to work,
  337         which the adapters satisfy by keeping their scoring half self-contained.
  338         """
  339         gold_content = json.dumps(record["gold"], ensure_ascii=False, sort_keys=True)
  340         scorer_input = {"task_id": record["task_id"], "path": GOLD_FILE,
  341                         "sha256": hashlib.sha256(gold_content.encode("utf-8")).hexdigest()}
  342         return {
  343             "kind": self.VERIFIER_KIND,
  344             "scorer_input": scorer_input,
  345             "argv": [
  346                 "python3", f"{CHECK_DIR}/{CHECK_NAME}",
  347                 "--dataset", self.name,
  348                 "--answer", ANSWER_FILE,
  349                 "--prediction", PREDICTION_FILE,
  350                 "--gold", GOLD_FILE,
  351                 "--task-id", record["task_id"],
  352                 "--gold-sha256", scorer_input["sha256"],
  353                 "--reward-file", REWARD_REL,
  354                 "--evidence-file", EVIDENCE_REL,
  355             ],
  356             "inputs": [
  357                 {"path": GOLD_FILE,
  358                  "content": gold_content},
  359                 {"path": f"{CHECK_DIR}/{CHECK_NAME}",
  360                  "from": str(VERIFY_DIR / CHECK_NAME)},
  361                 # The scaffold, materialized under the same package names the platform
  362                 # uses (`data.scaffolds.qa`), so the adapter's own `from data.scaffolds.qa import base` is
  363                 # the same line in both contexts. A shim that made the import *different*
  364                 # in the check would be one more thing that can disagree; this is the
  365                 # same module, copied.
  366                 {"path": f"{CHECK_DIR}/read.py", "from": str(QA_DIR / "read.py")},
  367                 # The scaffold, materialized under the package names the platform uses,
  368                 # so the adapter's own `from data.scaffolds.qa import base` is the same line in
  369                 # both contexts. `data/__init__.py` is what makes `data` an ordinary
  370                 # package: without it `data` is a namespace package, and a copy of it in
  371                 # the workdir would *merge* with the platform's rather than shadow it --
  372                 # so the import would resolve into the tree the sandbox hides.
  373                 {"path": f"{DATASET_PACKAGE}/__init__.py",
  374                  "from": str(ROOT / DATASET_PACKAGE / "__init__.py")},
  375                 {"path": f"{DATASET_PACKAGE}/scaffolds/__init__.py",
  376                  "from": str(ROOT / "data/scaffolds/__init__.py")},
  377                 {"path": f"{DATASET_PACKAGE}/scaffolds/qa/__init__.py", "from": str(QA_DIR / "__init__.py")},
  378                 {"path": f"{DATASET_PACKAGE}/scaffolds/qa/read.py", "from": str(QA_DIR / "read.py")},
  379                 {"path": f"{DATASET_PACKAGE}/scaffolds/qa/base.py", "from": str(QA_DIR / "base.py")},
  380                 {"path": f"{DATASET_PACKAGE}/scaffolds/qa/evidence_scorers.py",
  381                  "from": str(QA_DIR / "evidence_scorers.py")},
  382                 {"path": f"{DATASET_PACKAGE}/scaffolds/qa/prediction.py",
  383                  "from": str(QA_DIR / "prediction.py")},
  384                 {"path": f"{DATASET_PACKAGE}/datasets/__init__.py",
  385                  "from": str(ROOT / "data/datasets/__init__.py")},
  386                 # The dataset's own adapter, under its real module name.
  387                 {"path": f"{DATASET_PACKAGE}/datasets/{self.name}.py",
  388                  "from": str(ROOT / DATASET_PACKAGE / "datasets" / f"{self.name}.py")},
  389             ],
  390             "reward_file": REWARD_FILE,
  391             "evidence_file": EVIDENCE_FILE,
  392             "timeout_s": 3600,
  393         }

  395     def provision_status(self) -> dict:
  396         """What is present locally, for `tools/datasets.py --check`.
  397 
  398         Deliberately does not read the table: this is the answer to "did the
  399         provisioning run", which has to work on a machine where it did not, and on a
  400         machine where pyarrow is not installed.
  401         """
  402         # What counts as present is the **downloaded artifact**, not the files a loader
  403         # reads out of it: an `http` entry's `dst` is what `--fetch` writes, and it is the
  404         # only path checkable without knowing how an archive was extracted. A first
  405         # version looked for the member names under the root and reported every
  406         # tarball-backed dataset as unprovisioned, so `--check` said `MISS` and the lint
  407         # skipped a dataset whose data was right there.
  408         artifacts: list[Path] = []
  409         for entry in self._downloads():
  410             dst = entry.get("dst")
  411             if dst:
  412                 path = Path(str(dst))
  413                 artifacts.append(path if path.is_absolute() else self.root / path)
  414         present = [str(p) for p in artifacts if p.exists()]
  415         return {
  416             "root": str(self.root),
  417             "present": bool(artifacts) and len(present) == len(artifacts),
  418             "split_dir": str(self.root),
  419             "files": [str(p) for p in artifacts],
  420             "present_files": present,
  421             "tasks": len(self.ensure()),
  422         }

  429     def build(self) -> dict:
  430         """Everything a `data/datasets/<name>.py` module exports, as one dict.
  431 
  432         Returned rather than copied into the caller's globals because the caller has to
  433         spell each name anyway (`KIND = built["KIND"]`), and a `globals().update(...)`
  434         that adds `DATASET_ROOT` by accident would put the lint's path rule on a
  435         variable nobody declared.
  436 
  437         **A missing data root is not an import failure.** `data/scaffolds/registry.py::published`
  438         imports every module in `data/` to decide what the Launch form offers, and
  439         `tools/dataset_lint.py` imports one to lint it -- both must work on a machine
  440         where `--fetch` has not run. So a table that cannot be read is recorded in
  441         `load_error` and re-raised by `load()`, which is the call that actually needs the
  442         bytes. This is the same split `data/datasets/alfworld.py` makes with `_LOAD_ERROR`.
  443         """
  444         records = self.ensure()
  445         ids = [r["task_id"] for r in records]
  446         division = self.division()
  447         exported = {
  448             "KIND": "dataset",
  449             "DATASET_ROOT": self.root,
  450             "DEFAULT_ENV": {"kind": "files"},
  451             "ENV": {tid: {"kind": "files"} for tid in ids},
  452             "SPLIT": {"train": division["train"], "eval": division["eval"]},
  453             # The benchmark's own evaluation set, in the one place this scaffold can know
  454             # it: the scored split's ids when the benchmark publishes a study split too
  455             # (`gsm8k`), and `None` when the pool *is* the scored split (`triviaqa` and the
  456             # rest). Never absent -- `data/scaffolds/registry.py::official_split` refuses silence.
  457             "OFFICIAL_EVAL": division["official_eval"],
  458             #: Which rule produced `SPLIT`: `"declared"` (the benchmark said so) or
  459             #: `"hash"` (the platform imposed it). Read by `load()`'s emptiness guard and
  460             #: by `tests/quality/test_qaspec_division.py`, which is the test that fails if
  461             #: the two rules are ever computed in two different places again.
  462             "SPLIT_SOURCE": division["source"],
  463         }
  464         # `SETUP` and `VERIFY` are **not** built here. Each is a dict with one entry per
  465         # task and each entry renders the goal or serialises the gold, so building them at
  466         # import made importing triviaqa cost 18 s. The caller that needs them (the
  467         # platform's `run_one`, or the lint) asks for them by name; see the adapter's
  468         # module `__getattr__`.
  469         # The one-side-empty check is made at load time, not here: with no data both
  470         # sides are empty, and that is a provisioning problem, not a split problem.
  471         return exported
  472 
  473     def setup_map(self) -> dict:
  474         """`SETUP`: one entry per task. Built on demand -- see `build()`."""
  475         return {r["task_id"]: self.setup(r) for r in self.ensure()}
  476 
  477     #: The `kind` every entry of `VERIFY` carries, declared once. `verify_map()`
  478     #: materializes one entry per task on purpose (building it at import cost 18 s on
  479     #: triviaqa), so a caller that only needs the *set* of kinds must not be forced to
  480     #: pay for 949 identical dicts -- that turned a taxonomy report into a 19-second
  481     #: command and made the dataset-classification suite take fifteen minutes.
  482     #: `tests/quality/test_dataset_classes.py` pins this against a materialized map so
  483     #: the declaration cannot drift away from what `verifier()` actually builds.
  484     VERIFIER_KIND = "command"
  485 
  486     def verify_map(self) -> dict:
  487         """`VERIFY`: one entry per task. Built on demand -- see `build()`."""
  488         return {r["task_id"]: self.verifier(r) for r in self.ensure()}
  489 
  490     def control_dir(self) -> Path:
  491         return Path(__file__).resolve().parent / "oracle_harness"
  492 
  493 
  494 def module_getattr(spec: "QASpec", name: str):
  495     """The adapter's module-level `__getattr__`, shared by every QA dataset.
  496 
  497     Returns `SETUP`, `VERIFY`, `_RECORDS` or `_LOAD_ERROR` on demand and raises
  498     `AttributeError` for anything else, which is what a module-level `__getattr__` must
  499     do. The two maps are built lazily because each holds one entry per task and building
  500     them at import cost 18 s on triviaqa (`QASpec.ensure`); `_RECORDS` is exposed because
  501     the control and the tests read it, and it is the same cached list `load()` uses.
  502     """
  503     if name == "SETUP":
  504         return spec.setup_map()
  505     if name == "VERIFY":
  506         return spec.verify_map()
  507     if name == "_RECORDS":
  508         return spec.ensure()
  509     if name == "_LOAD_ERROR":
  510         return spec.load_error
  511     raise AttributeError(f"module {spec.name!r} has no attribute {name!r}")
  512

  514 def split_env_override(name: str) -> str | None:
  515     """`HG_<NAME>_SPLIT=none` runs every task as eval, as ALFWorld's adapter allows.
  516 
  517     Present for the same reason: the benchmark has no train/eval division and the one
  518     here is imposed, so a reader who wants the whole set scored must be able to say so
  519     without editing the module.
  520     """
  521     return os.environ.get(f"HG_{name.upper()}_SPLIT")
