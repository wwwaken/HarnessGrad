class Strip(ast.NodeTransformer):
    def visit_Dict(self, node):
        self.generic_visit(node)
        pairs = [(k, v) for k, v in zip(node.keys, node.values)
                 if not (isinstance(k, ast.Constant) and k.value == 'immutable')]
        node.keys = [k for k, v in pairs]
        node.values = [v for k, v in pairs]
        return node
    def visit_Expr(self, node):
        return None if isinstance(node.value, ast.Constant) and isinstance(node.value.value, str) else self.generic_visit(node)
