for root in ('base_harness', 'eval', 'data', 'harnessgrad', 'tools'):
    for path in sorted(Path(root).rglob('*.py')):
        if root == 'data' and path.parts[-2:] == ('verify', 'check.py'):
            continue
        trees = [(ast.parse(path.read_text(encoding='utf-8')), 0, None)]
        if str(path) == 'tools/diagnose_appworld.py':
            value = next(n.value for n in trees[0][0].body if isinstance(n, ast.Assign)
                         and any(isinstance(t, ast.Name) and t.id == 'ACTOR' for t in n.targets))
            trees.append((ast.parse(value.value), value.lineno - 1, 'ACTOR'))
        for tree, offset, embedded in trees:
            for node in sorted(ast.walk(tree), key=lambda n: getattr(n, 'lineno', 0)):
                if not isinstance(node, ast.Call):
                    continue
                name = getattr(node.func, 'attr', None) or getattr(node.func, 'id', None)
                if name not in ('run', 'Popen', 'check_output', 'check_call', 'call'):
